## 核心架构问题分析 [CRITICAL PRIORITY]

### 问题1: CardInstance与Ability单向绑定设计缺陷 [CRITICAL]

**问题描述：**
当前架构中CardInstance与Ability采用单向绑定设计，存在严重的功能限制和性能问题。

**具体问题：**
- [ ] CardInstance无法直接访问自己的能力列表
- [ ] 每次查询卡牌能力都需要通过AbilityManager.GetAbilitiesForCard()
- [ ] 临时能力（AbilityBuffEffect）无法正常注册和管理
- [ ] 游戏逻辑中频繁的能力查询造成性能开销
- [ ] UI显示卡牌能力信息需要额外的服务调用

**影响范围：**
- 游戏性能：频繁的AbilityManager查询
- 功能完整性：AbilityBuffEffect无法工作
- 代码复杂度：需要在多处手动查询能力
- 用户体验：UI无法实时显示完整的卡牌能力信息

### 问题2: AbilityBuffEffect架构层级不匹配 [HIGH]

**问题描述：**
BuffManager无法直接访问AbilityManager来注册临时能力，导致"获得能力"类Buff无法正常工作。

**具体问题：**
- [ ] BuffManager只有CardInstance的字符串ID，无法获取完整对象
- [ ] AbilityManager.RegisterAbility()需要完整的CardInstance对象
- [ ] 临时能力的生命周期管理复杂且不可靠
- [ ] 架构层级依赖关系混乱

**技术根因：**
```
BuffManager (Domain/Buffs)
    ↓ 需要访问但无法访问
AbilityManager (Domain/Abilities)
    ↓ 需要访问
CardInstance (Domain/Entities)
```

### 问题3: 游戏运行时能力查询性能问题 [MEDIUM]

**问题描述：**
当前架构中多个场景需要频繁查询卡牌能力，造成不必要的性能开销。

**查询场景分析：**
- [ ] 事件触发时：EventProcessor需要查询所有相关能力
- [ ] 可玩性检查：GameService.CheckCardPlayability需要查询能力条件
- [ ] UI显示：前端需要显示卡牌的完整能力信息
- [ ] 战斗计算：攻击、防御计算需要考虑能力效果
- [ ] 关键字检查：CardInstanceExtensions中的各种关键字检查

**性能影响：**
- 每次查询都需要Dictionary查找和List复制
- 高频操作（如UI更新）造成重复查询
- 缺少缓存机制导致重复计算

## 架构改进方案设计

### 方案A: CardInstance双向绑定设计 [推荐方案]

**设计目标：**
让CardInstance能够直接访问自己的能力，同时保持架构的SOLID原则。

**实现方案：**
- [ ] 在CardInstance中添加Abilities属性（只读）
- [ ] 通过AbilityManager维护双向绑定关系
- [ ] 实现能力变更时的自动同步机制
- [ ] 添加能力缓存和失效机制

**技术实现：**
```csharp
// CardInstance.cs
public IReadOnlyList<Ability> Abilities { get; private set; } = [];

// AbilityManager.cs
public void RegisterAbility(Ability ability)
{
    // 现有逻辑...

    // 新增：更新CardInstance的能力列表
    if (ability.SourceCard != null)
    {
        UpdateCardInstanceAbilities(ability.SourceCard);
    }
}

private void UpdateCardInstanceAbilities(CardInstance cardInstance)
{
    var abilities = GetAbilitiesForCard(cardInstance.InstanceId);
    cardInstance.SetAbilities(abilities); // 内部方法
}
```

**优势：**
- 直接访问，无需查询开销
- 保持数据一致性
- 支持临时能力的完整生命周期
- 提升UI响应性能

**风险：**
- 需要维护双向同步
- 内存占用略有增加
- 需要仔细处理并发访问

```csharp
// AbilityManager.cs - 性能优化
public class AbilityManager
{
    // 新增：能力变更事件，用于批量更新优化
    public event Action<CardInstance>? CardAbilitiesChanged;

    // 新增：批量操作支持
    private readonly HashSet<CardInstance> _pendingUpdates = new();
    private readonly object _updateLock = new();

    // 修改：支持批量更新的注册方法
    public void RegisterAbilities(IEnumerable<Ability> abilities)
    {
        lock (_updateLock)
        {
            foreach (var ability in abilities)
            {
                if (ability?.SourceCard == null) continue;

                // 注册逻辑...
                _pendingUpdates.Add(ability.SourceCard);
            }

            // 批量更新
            FlushPendingUpdates();
        }
    }

    // 新增：刷新待更新的CardInstance
    private void FlushPendingUpdates()
    {
        foreach (var cardInstance in _pendingUpdates)
        {
            UpdateCardInstanceAbilities(cardInstance);
            CardAbilitiesChanged?.Invoke(cardInstance);
        }
        _pendingUpdates.Clear();
    }
}
```

#### BuffManager修改设计
```csharp
// BuffManager.cs - 修改AbilityBuffEffect处理
public class BuffManager : IBuffManager
{
    // 新增：事件发送器
    private readonly IEventProcessor _eventProcessor;

    // 修改：AbilityBuffEffect注册处理
    private async void HandleAbilityBuffRegistration(Buff buff)
    {
        try
        {
            var abilityEffects = buff.Effects.OfType<AbilityBuffEffect>().ToList();
            if (abilityEffects.Count == 0) return;

            foreach (var abilityEffect in abilityEffects)
            {
                // 创建并发送事件
                var registrationEvent = new AbilityBuffRegistrationEvent(
                    targetCardInstanceId: buff.TargetCardInstanceId,
                    abilityConfigs: abilityEffect.GrantedAbilityConfigs,
                    sourceBuffId: buff.Id,
                    isRemoval: false
                );

                // 通过事件系统发送
                await _eventProcessor.ProcessEventAsync(registrationEvent, null);

                _logger.LogDebug("Sent AbilityBuffRegistrationEvent for Buff {BuffId}: {AbilityCount} abilities",
                    buff.Id, abilityEffect.GrantedAbilityConfigs.Count);
            }
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error sending AbilityBuffRegistrationEvent for Buff {BuffId}", buff.Id);
        }
    }

    // 修改：AbilityBuffEffect移除处理
    private async void HandleAbilityBuffRemoval(Buff buff)
    {
        try
        {
            var abilityEffects = buff.Effects.OfType<AbilityBuffEffect>().ToList();
            if (abilityEffects.Count == 0) return;

            foreach (var abilityEffect in abilityEffects)
            {
                // 创建并发送移除事件
                var removalEvent = new AbilityBuffRegistrationEvent(
                    targetCardInstanceId: buff.TargetCardInstanceId,
                    abilityConfigs: abilityEffect.GrantedAbilityConfigs,
                    sourceBuffId: buff.Id,
                    isRemoval: true
                );

                await _eventProcessor.ProcessEventAsync(removalEvent, null);

                _logger.LogDebug("Sent AbilityBuffRemovalEvent for Buff {BuffId}: {AbilityCount} abilities",
                    buff.Id, abilityEffect.GrantedAbilityConfigs.Count);
            }
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error sending AbilityBuffRemovalEvent for Buff {BuffId}", buff.Id);
        }
    }
}
```

```csharp
        try
        {
            // 从GameState获取CardInstance
            var cardInstance = GetCardInstanceById(evt.TargetCardInstanceId);
            if (cardInstance == null)
            {
                _logger.LogWarning("CardInstance {CardId} not found for AbilityBuffEffect", evt.TargetCardInstanceId);
                return;
            }

            if (evt.IsRemoval)
            {
                await RemoveTemporaryAbilities(evt.SourceBuffId);
            }
            else
            {
                await RegisterTemporaryAbilities(cardInstance, evt.AbilityConfigs, evt.SourceBuffId);
            }
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error handling AbilityBuffRegistrationEvent for buff {BuffId}", evt.SourceBuffId);
        }
    }

    // 新增：注册临时能力
    private async Task RegisterTemporaryAbilities(CardInstance cardInstance, List<AbilityConfig> abilityConfigs, string sourceBuffId)
    {
        var temporaryAbilities = new List<Ability>();

        foreach (var config in abilityConfigs)
        {
            // 标记为临时能力
            config.Metadata["IsTemporary"] = true;
            config.Metadata["SourceBuffId"] = sourceBuffId;
            config.Metadata["RegistrationTime"] = DateTime.UtcNow;

            // 使用现有的AbilityManager流程创建能力
            var ability = _abilityManager.CreateAbilityFromConfig(config, cardInstance);
            if (ability != null)
            {
                temporaryAbilities.Add(ability);
                _abilityManager.RegisterAbility(ability);
            }
        }

        // 记录临时能力，便于后续清理
        _temporaryAbilitiesByBuffId[sourceBuffId] = temporaryAbilities;

        _logger.LogInformation("Registered {Count} temporary abilities for CardInstance {CardId} from Buff {BuffId}",
            temporaryAbilities.Count, cardInstance.InstanceId, sourceBuffId);
    }

    // 新增：移除临时能力
    private async Task RemoveTemporaryAbilities(string sourceBuffId)
    {
        if (!_temporaryAbilitiesByBuffId.TryGetValue(sourceBuffId, out var temporaryAbilities))
        {
            _logger.LogDebug("No temporary abilities found for Buff {BuffId}", sourceBuffId);
            return;
        }

        foreach (var ability in temporaryAbilities)
        {
            _abilityManager.UnregisterAbility(ability);
        }

        _temporaryAbilitiesByBuffId.Remove(sourceBuffId);

        _logger.LogInformation("Removed {Count} temporary abilities from Buff {BuffId}",
            temporaryAbilities.Count, sourceBuffId);
    }

    // 新增：获取CardInstance的辅助方法
    private CardInstance? GetCardInstanceById(string cardInstanceId)
    {
        // 从当前游戏状态中查找CardInstance
        // 这里需要根据实际的GameState结构实现
        // 示例实现：
        var gameState = GetCurrentGameState();

        // 搜索所有可能的位置
        var allCards = new List<CardInstance>();
        allCards.AddRange(gameState.Player1.Hand);
        allCards.AddRange(gameState.Player1.Field);
        allCards.AddRange(gameState.Player1.Graveyard);
        allCards.AddRange(gameState.Player2.Hand);
        allCards.AddRange(gameState.Player2.Field);
        allCards.AddRange(gameState.Player2.Graveyard);

        return allCards.FirstOrDefault(c => c.InstanceId.ToString() == cardInstanceId);
    }
}
```

```csharp
// AbilityManager.cs - 缓存机制
public class AbilityManager
{
    // 新增：查询结果缓存
    private readonly MemoryCache _abilityQueryCache;
    private readonly CacheOptions _cacheOptions;

    // 新增：缓存配置
    public class CacheOptions
    {
        public TimeSpan DefaultExpiration { get; set; } = TimeSpan.FromMinutes(5);
        public int MaxCacheSize { get; set; } = 1000;
        public bool EnableCaching { get; set; } = true;
    }

    // 修改：带缓存的能力查询
    public List<Ability> GetAbilitiesForCard(string cardInstanceId)
    {
        if (!_cacheOptions.EnableCaching)
        {
            return GetAbilitiesForCardInternal(cardInstanceId);
        }

        var cacheKey = $"abilities_card_{cardInstanceId}";

        if (_abilityQueryCache.TryGetValue(cacheKey, out List<Ability>? cachedAbilities))
        {
            _logger?.LogDebug("[AbilityManager] Cache hit for card {CardId}", cardInstanceId);
            return cachedAbilities.ToList(); // 返回副本
        }

        var abilities = GetAbilitiesForCardInternal(cardInstanceId);

        // 缓存结果
        _abilityQueryCache.Set(cacheKey, abilities.ToList(), _cacheOptions.DefaultExpiration);
        _logger?.LogDebug("[AbilityManager] Cached abilities for card {CardId}: {Count} abilities",
            cardInstanceId, abilities.Count);

        return abilities;
    }

    // 新增：实际查询逻辑（原GetAbilitiesForCard方法）
    private List<Ability> GetAbilitiesForCardInternal(string cardInstanceId)
    {
        return _abilitiesByCardId.TryGetValue(cardInstanceId, out var abilities)
            ? abilities.ToList()
            : new List<Ability>();
    }

    // 新增：缓存失效方法
    public void InvalidateCardCache(string cardInstanceId)
    {
        var cacheKey = $"abilities_card_{cardInstanceId}";
        _abilityQueryCache.Remove(cacheKey);
        _logger?.LogDebug("[AbilityManager] Invalidated cache for card {CardId}", cardInstanceId);
    }

    // 修改：注册能力时失效缓存
    public void RegisterAbility(Ability ability)
    {
        // 现有逻辑...

        // 新增：失效相关缓存
        if (ability?.SourceCard != null)
        {
            InvalidateCardCache(ability.SourceCard.InstanceId);
        }
    }
}
```

```csharp
            }

            _disposed = true;

    // 新增：弱引用版本（如果需要）
    private readonly List<WeakReference<Ability>> _weakAbilities = new();

    public IEnumerable<Ability> GetValidAbilities()
    {
        var validAbilities = new List<Ability>();
        for (int i = _weakAbilities.Count - 1; i >= 0; i--)
        {
            if (_weakAbilities[i].TryGetTarget(out var ability))
            {
                validAbilities.Add(ability);
            }
            else
            {
                _weakAbilities.RemoveAt(i); // 清理失效引用
            }
        }
        return validAbilities;
    }
}
```
