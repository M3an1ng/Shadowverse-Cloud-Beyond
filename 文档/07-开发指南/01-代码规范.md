## 核心概念

### 代码规范原则
- **一致性**: 整个项目保持统一的代码风格
- **可读性**: 代码应该自解释，易于理解
- **可维护性**: 便于后续修改和扩展
- **SOLID原则**: 遵循面向对象设计原则

### 质量标准
- **单一职责**: 每个类和方法只负责一个功能
- **命名清晰**: 使用描述性的名称
- **注释适度**: 关键逻辑添加必要注释
- **错误处理**: 完善的异常处理机制

        /// 检查卡牌是否可以进化
        /// </summary>

            }
        }
        
        #endregion
    }

// ✅ 正确的异步方法实现
public class CardRepository : ICardRepository
{
    public async Task<CardDefinition?> GetCardDefinitionByIdAsync(int cardId)
    {
        try
        {
            return await _context.CardDefinitions
                .AsNoTracking()
                .FirstOrDefaultAsync(c => c.Id == cardId);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to get card definition for ID {CardId}", cardId);
            throw;
        }
    }

    public async Task<List<CardDefinition>> GetAllCardsAsync()
    {
        return await _context.CardDefinitions
            .AsNoTracking()
            .OrderBy(c => c.Cost)
            .ThenBy(c => c.Name)
            .ToListAsync();
    }
}

#### 异常处理模式
```csharp
// ✅ 正确的异常处理
public class EconomyService : IEconomyService
{
    public async Task<bool> SpendCoinsAsync(Guid userId, int amount, CoinTransactionType type)
    {
        if (amount <= 0)
        {
            _logger.LogWarning("Invalid coin amount: {Amount} for user {UserId}", amount, userId);
            return false;
        }

        using var transaction = await _context.Database.BeginTransactionAsync();
        try
        {
            var userEconomy = await GetOrCreateUserEconomyAsync(userId);
            
            if (userEconomy.Coins < amount)
            {
                _logger.LogWarning("Insufficient coins: Required {Required}, Available {Available} for user {UserId}", 
                    amount, userEconomy.Coins, userId);
                return false;
            }

            userEconomy.Coins -= amount;
            await _context.SaveChangesAsync();
            await transaction.CommitAsync();

            _logger.LogInformation("Successfully spent {Amount} coins for user {UserId}", amount, userId);
            return true;
        }
        catch (Exception ex)
        {
            await transaction.RollbackAsync();
            _logger.LogError(ex, "Failed to spend coins for user {UserId}", userId);
            throw new EconomyServiceException($"Failed to spend coins for user {userId}", ex);
        }
    }
}

// 自定义异常类
public class EconomyServiceException : Exception
{
    public EconomyServiceException(string message) : base(message) { }
    public EconomyServiceException(string message, Exception innerException) : base(message, innerException) { }
}
```

// ✅ 正确的异步处理
class SignalRManager {
    constructor() {
        this.connection = null;
        this.isConnected = false;
    }

    async connect(gameId, playerId) {
        try {
            const hubUrl = `/gamehub?gameId=${encodeURIComponent(gameId)}&playerId=${encodeURIComponent(playerId)}`;
            
            this.connection = new signalR.HubConnectionBuilder()
                .withUrl(hubUrl)
                .withAutomaticReconnect()
                .configureLogging(signalR.LogLevel.Information)
                .build();

            this.registerEventHandlers();
            
            await this.connection.start();
            this.isConnected = true;
            
            console.log('SignalR connection established');
        } catch (error) {
            console.error('Failed to establish SignalR connection:', error);
            this.isConnected = false;
            throw error;
        }
    }

    async sendGameAction(actionType, actionData) {
        if (!this.isConnected) {
            throw new Error('SignalR connection not established');
        }

        try {
            await this.connection.invoke(actionType, actionData);
        } catch (error) {
            console.error(`Failed to send ${actionType}:`, error);
            throw error;
        }
    }

    registerEventHandlers() {
        this.connection.on('ReceiveGameState', (gameState) => {
            this.handleGameStateUpdate(gameState);
        });

        this.connection.on('ReceiveGameEvent', (gameEvent) => {
            this.handleGameEvent(gameEvent);
        });

        this.connection.on('ReceiveError', (error) => {
            this.handleError(error);
        });
    }
}

// ✅ 防抖处理
function debounce(func, wait) {
    let timeout;
    return function executedFunction(...args) {
        const later = () => {
            clearTimeout(timeout);
            func(...args);
        };
        clearTimeout(timeout);
        timeout = setTimeout(later, wait);
    };
}

// 搜索输入防抖
const debouncedSearch = debounce(async (searchTerm) => {

// ✅ 批量DOM更新
function updateMultipleElements(updates) {
    // 使用DocumentFragment减少重排
    const fragment = document.createDocumentFragment();
    
    updates.forEach(update => {
        const element = document.createElement(update.tag);
        element.textContent = update.text;
        element.className = update.className;
        fragment.appendChild(element);
    });
    
    document.getElementById('container').appendChild(fragment);
}
