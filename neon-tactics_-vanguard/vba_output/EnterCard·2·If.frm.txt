' 事件功能：回合条件处理 - 回合期间
' 说明：设置效果在特定玩家回合期间触发的条件
' ==================================================
Private Sub DuringTurn_Click()
    ' ========== 互斥逻辑处理 ==========
    ' 与解锁回合条件互斥
    If DuringTurn.Value Then UnlockTurn.Value = False
    
    ' ========== 玩家选择 ==========
    ' 询问自己或对手，并记录在按钮名称上
    If DuringTurn.Caption = Translate("DuringTurnButton") Then
        Dim options() As String
        options = Split(SentenceTranslate("DuringTurnButton|Allied|Enemy"), "|")
        Dim i%
        i = Val(ShowChoice(options, 50, 2))
        
        DuringTurn.Caption = options(i) & "-" & Translate("Turn")
    End If
    
    ' ========== 状态恢复 ==========
    ' 如果未按下，则按钮名称恢复默认
    If Not DuringTurn.Value Then DuringTurn.Caption = Translate("DuringTurnButton")
    
    ' ========== 效果预览更新 ==========
    Call EffectPreview
End Sub

' ==================================================
' 事件功能：比较条件处理 - 确认比较
' 说明：确认并生成完整的比较条件表达式
' ==================================================
Private Sub Compare_Click()
    ' ========== 条件验证 ==========
    ' 检查必要的条件是否已设置
    If SetX.Caption = Translate("ClickToInput") Or _
        (Equal.Value = False And Less.Value = False And More.Value = False And UnEqual.Value = False) Or _
        (SetConst.Value = False And SetY.Caption = Translate("ClickToInput")) Then
        Compare.Value = False
        Exit Sub
    End If
    
    ' ========== 比较表达式生成 ==========
    ' 生成比较条件字符串
    If Compare.Caption = Translate("CompareButton") Then
        Dim Str As String
        Compare.Caption = Empty
        
        ' 获取比较运算符
        If Equal.Value Then Str = "="
        If Less.Value Then Str = "<"
        If More.Value Then Str = ">"
        If UnEqual.Value Then Str = "<>"
        
        ' 构建比较表达式
        If Str <> "" And SetX.Caption <> Translate("ClickToInput") Then
            If SetConst.Value Then
                Compare.Caption = Compare.Caption & SetX.Caption & Str & SetConst.Caption & "|"
            ElseIf SetY.Caption <> Translate("ClickToInput") Then
                Compare.Caption = Compare.Caption & SetX.Caption & Str & SetY.Caption & "|"
            End If
        End If
        
        ' ========== 状态重置 ==========
        ' 生成后重置所有比较条件状态
        SetX.Caption = Translate("ClickToInput")
        SetY.Caption = Translate("ClickToInput")
        SetConst.Value = False
        Equal.Value = False
        Less.Value = False
        More.Value = False
        UnEqual.Value = False
    End If
    
    ' ========== 状态恢复 ==========
    ' 如果未按下，则按钮名称恢复默认
    If Not Compare.Value Then Compare.Caption = Translate("CompareButton")
    
    ' ========== 效果预览更新 ==========
    Call EffectPreview
End Sub

' ==================================================
' 函数功能：效果预览生成
' 说明：根据当前设置的条件生成预览字符串
' ==================================================
Function EffectPreview()
    ' ========== 条件字符串初始化 ==========
    IfEffect = Translate("If") & "|"
    
    ' ========== 回合条件添加 ==========
    If DuringTurn.Value Then IfEffect = IfEffect & DuringTurn.Caption & "|"
    If UnlockTurn.Value Then IfEffect = IfEffect & UnlockTurn.Caption & "|"
    
    ' ========== 触发次数条件添加 ==========
    If Times.Value Then IfEffect = IfEffect & Times.Caption & "|"
    
    ' ========== 比较条件添加 ==========
    If Compare.Value Then IfEffect = IfEffect & Compare.Caption
    
    ' ========== 直接比较条件处理 ==========
    ' 处理尚未确认的比较条件
    Dim Str As String
    If Equal.Value Then Str = "="
    If Less.Value Then Str = "<"
    If More.Value Then Str = ">"
    If UnEqual.Value Then Str = "<>"
    
    If Str <> "" And SetX.Caption <> Translate("ClickToInput") Then
        If SetConst.Value Then
            IfEffect = IfEffect & SetX.Caption & Str & SetConst.Caption & "|"
        ElseIf SetY.Caption <> Translate("ClickToInput") Then
            IfEffect = IfEffect & SetX.Caption & Str & SetY.Caption & "|"
        End If
    End If
    
    ' ========== 职业条件添加 ==========
    If Combo.Value Then IfEffect = IfEffect & Combo.Caption & "|"
    If Rally.Value Then IfEffect = IfEffect & Rally.Caption & "|"
    If EarthRite.Value Then IfEffect = IfEffect & EarthRite.Caption & "|"
    If Overflow.Value Then IfEffect = IfEffect & Overflow.Caption & "|"
    If Skybound.Value Then IfEffect = IfEffect & Skybound.Caption & "|"
    If SuperSkybound.Value Then IfEffect = IfEffect & SuperSkybound.Caption & "|"
    If Necromancy.Value Then IfEffect = IfEffect & Necromancy.Caption & "|"
    If Sacrifice.Value Then IfEffect = IfEffect & Sacrifice.Caption & "|"
    If NoDuplicates.Value Then IfEffect = IfEffect & NoDuplicates.Caption & "|"
    
    ' ========== 字符串格式处理 ==========
    ' 删去末尾分隔符
    If Len(IfEffect) = 0 Then IfEffect = "|"
    IfEffect = Left(IfEffect, Len(IfEffect) - 1)
    
    ' ========== 效果翻译和显示 ==========
    ' 翻译条件字符串并更新显示
    Effect.Caption = IfEffect
    IfEffect = SentenceTranslate(IfEffect)
End Function
