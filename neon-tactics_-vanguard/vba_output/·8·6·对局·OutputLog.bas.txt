Attribute VB_Name = "·8·6·对局·OutputLog"
Function OutputLog(InputEffect As String, ThisCard As String)
    Dim i%, j%
    Dim Str As String
    Dim Arr As Variant
    
    Arr = Down("Log")
    For i = 1 To UBound(Arr, 1)
        If Arr(i, 1) = "" Then
            Str = Down("Turn")
            Arr(i, 1) = Str
            Arr(i, 2) = ThisCard
            Arr(i, 3) = InputEffect
            Exit For
        End If
    Next i
    Call Up("Log", Arr)
End Function
