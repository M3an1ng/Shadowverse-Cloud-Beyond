Attribute VB_Name = "·5·4·步骤·InputSteps"
Function InputSteps()
    Dim i%, j%
    Dim Str As String
    Dim Arr As Variant
    Dim Steps() As String
    Dim Action() As String
    
    Str = QueueAction
    
    'Call OutputLog(Str, "-")
    
    Steps = Split(Str, "||")
    For i = 0 To UBound(Steps, 1)
        Action = Split(Steps(i), "|")
        
        If Action(0) = "Play" Then
            Call PlayFunction(Action(1))
        ElseIf Action(0) = "Strike" Then
            Call StrikeFunction(Action(1), Action(2))
        ElseIf Action(0) = "Engage" Then
            Call EngageFunction(Action(1))
        ElseIf Action(0) = "Keyword&Fuse" Then
            Call FuseFunction(Action(1), Action(2))
        ElseIf Action(0) = "Evolve" Then
            Call EvolveFunction(Action(1))
        ElseIf Action(0) = "SuperEvolve" Then
            Call SuperEvolveFunction(Action(1))
        ElseIf Action(0) = "GetPP" Then
            Call Up("EnemyPP", Val(Down("EnemyPP")) + 1)
        End If
    Next i
    
    QueueAction = Empty
End Function


Function EraseQueueAction(InputAction As String) As String
    Dim i%
    Dim Steps() As String
    Dim Action() As String
    
    Steps = Split(QueueAction, "||")
    For i = 0 To UBound(Steps)
        Action = Split(Steps(i), "|")
        If Action(0) = InputAction Then
            Steps(i) = "\"
            EraseQueueAction = Action(1)
            Exit For
        Else
            Steps(i) = "\"
        End If
    Next i
    QueueAction = Replace(Join(Steps, "||"), "\||", "")
End Function
