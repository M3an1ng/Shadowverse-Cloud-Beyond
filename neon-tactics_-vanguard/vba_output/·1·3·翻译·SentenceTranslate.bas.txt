Attribute VB_Name = "·1·3·翻译·SentenceTranslate"
' ==================================================
' 函数功能：句子翻译函数，处理包含特殊分隔符的文本翻译
' 参数：InputText - 需要翻译的输入文本
' 返回：翻译后的文本，保留原始分隔符结构
' 特性：智能识别和保留分隔符，仅翻译非分隔符部分
' ==================================================
Function SentenceTranslate(InputText As String) As String
    Dim Str As String
    Dim ReplaceText As String
    ReplaceText = InputText   ' 保存原始输入文本的副本
    
    ' ========== 特殊分隔符预处理 ==========
    ' 将"||"替换为临时占位符"?"，避免与后续处理冲突
    ReplaceText = Replace(ReplaceText, "||", "?")
    ' 将"|"替换为临时占位符","，避免与后续处理冲突
    ReplaceText = Replace(ReplaceText, "|", ",")

    ' ========== 正则表达式初始化 ==========
    Dim regex As Object
    Set regex = CreateObject("VBScript.RegExp")   ' 创建正则表达式对象

    ' 设置正则表达式模式：匹配所有分隔符字符
    ' 包括：| - + & $ % @ < > = ? , { }
    regex.Pattern = "(\||\-|\+|&|\$|%|@|<|>|=|\?|\,|{|})"
    regex.Global = True   ' 全局匹配模式

    ' ========== 文本分割处理 ==========
    Dim matches As Object
    Dim parts() As String
    Dim i As Long, count As Long, j As Long

    ' 获取所有分隔符匹配项
    Set matches = regex.Execute(ReplaceText)

    ' 初始化结果数组（最大可能长度）
    ReDim parts(0 To matches.count * 2)

    ' 使用正则表达式在分隔符前后添加双竖线标记
    Dim tempText As String
    tempText = regex.Replace(ReplaceText, "||$1||")

    ' 按双竖线分割字符串，将文本和分隔符分离
    parts = Split(tempText, "||")

    ' ========== 逐部分翻译处理 ==========
    For i = LBound(parts) To UBound(parts)
        ' 检查是否为非空且不是分隔符的文本部分
        If Len(parts(i)) > 0 Then
            If Not regex.test(parts(i)) Then
                ' 调用Translate函数翻译非分隔符文本部分
                Str = Translate(parts(i))
                If parts(i) = Str And i + 2 <= UBound(parts) Then
                    If parts(i + 1) = "&" And Val(parts(i)) = 0 Then
                        parts(i) = Translate(parts(i) & "&" & parts(i + 2))
                        parts(i + 1) = ""
                        parts(i + 2) = ""
                    Else
                        parts(i) = Str
                    End If
                Else
                    parts(i) = Str
                End If
            End If
            ' 如果是分隔符，则保留原样不翻译
        End If
    Next i
    
    ' ========== 结果重组 ==========
    ' 将翻译后的各部分重新组合成完整字符串
    SentenceTranslate = Join(parts, "")
    
    ' ========== 恢复原始分隔符 ==========
    ' 将临时占位符"?"恢复为原始分隔符"||"
    SentenceTranslate = Replace(SentenceTranslate, "?", "||")
    ' 将临时占位符","恢复为原始分隔符"|"
    SentenceTranslate = Replace(SentenceTranslate, ",", "|")
End Function
