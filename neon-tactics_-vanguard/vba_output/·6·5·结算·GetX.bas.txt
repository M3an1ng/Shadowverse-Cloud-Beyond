Attribute VB_Name = "·6·5·结算·GetX"
' ==================================================
' 函数功能：计算卡牌属性值
' 参数：
'   InputX - 计算规则字符串
'   ThisCard - 当前卡牌ID
'   InputEffect - 效果字符串
' 返回：String - 计算结果和计算规则的组合字符串
' ==================================================
Function GetX(InputX As String, ThisCard As String, InputEffect As String) As String
    ' ========== 变量声明 ==========
    Dim result As Integer                    ' 最终计算结果
    Dim ResultX As Integer                   ' 临时计算结果
    Dim Calculate As String                  ' 计算方式标识
    Dim Xrr() As String                      ' 输入规则分割数组
    
    Dim i%, j%, k%                           ' 循环计数器
    Dim Str As String                        ' 临时字符串
    Dim Str2 As String                       ' 临时字符串2
    Dim Arr As Variant                       ' 临时数组
    Dim Brr As Variant                       ' 临时数组2
    
    Dim EffectSplit() As String              ' 效果分割数组
    Dim Enemy As String                      ' 敌方标识

    ' ========== 初始化设置 ==========
    ' 根据当前卡牌确定敌方标识
    If Left(ThisCard, 1) = "F" Then Enemy = "S" Else Enemy = "F"
    
    ' 分割输入规则
    Xrr = Split(InputX, "|")

    ' ========== 规则遍历处理 ==========
    ' 逐个解析计算规则
    For k = 0 To UBound(Xrr)
        ' 检查是否为计算方式标识
        If Xrr(k) = "HalveCount" Or Xrr(k) = "DoubleCount" Or Xrr(k) = "Difference" Or Xrr(k) = "Sum" Then
            Calculate = Xrr(k)
            k = k + 1
        End If
            
        ' ========== 基础属性计算 ==========
        ' PP相关属性
        If InStr(Xrr(k), "MaxPP") > 0 Then
            If InStr(Xrr(k), "Enemy-") > 0 Then ResultX = Down("MaxPP", Enemy & 1000) Else ResultX = Down("MaxPP", ThisCard)
        ElseIf InStr(Xrr(k), "PP") Then
            If InStr(Xrr(k), "Enemy-") > 0 Then ResultX = Down("PP", Enemy & 1000) Else ResultX = Down("PP", ThisCard)
        
        ' EP相关属性
        ElseIf InStr(Xrr(k), "SEP") > 0 Then
            If InStr(Xrr(k), "Enemy-") > 0 Then ResultX = Down("SEP", Enemy & 1000) Else ResultX = Down("SEP", ThisCard)
        ElseIf InStr(Xrr(k), "EP") Then
            If InStr(Xrr(k), "Enemy-") > 0 Then ResultX = Down("EP", Enemy & 1000) Else ResultX = Down("EP", ThisCard)
        
        ' Shadows属性
        ElseIf InStr(Xrr(k), "Shadows") > 0 Then
            If InStr(Xrr(k), "Enemy-") > 0 Then ResultX = Down("Shadows", Enemy & 1000) Else ResultX = Down("Shadows", ThisCard)
        
        ' 防御相关属性
        ElseIf InStr(Xrr(k), "LeaderDefense") > 0 Then
            If InStr(Xrr(k), "Enemy-") > 0 Then ResultX = Down("Defense", Enemy & 1000) Else ResultX = Down("Defense", ThisCard)
        ElseIf InStr(Xrr(k), "MaxDefense") > 0 Then
            If InStr(Xrr(k), "Enemy-") > 0 Then ResultX = Down("MaxDefense", Enemy & 1000) Else ResultX = Down("MaxDefense", ThisCard)
        
        ' ========== 游戏状态计算 ==========
        ' 回合数
        ElseIf Xrr(k) = "Turns" Then
            ResultX = Val(Mid(Down("Turn"), 2, 10))
        
        ' 冠名计数
        ElseIf Left(Xrr(k), Len("NamingCount")) = "NamingCount" Then
            Str = Mid(Xrr(k), Len("NamingCount") + 2, 100)
            ResultX = Down(Str, ThisCard)
        
        ' ========== 特殊效果计算 ==========
        ' 魔力增幅数
        ElseIf Xrr(k) = "Spellboost" Then
            ' 从效果字符串中提取Spellboost值
            EffectSplit = SplitEffect(DownByMatchID(ThisCard, "Effect"))
            For i = 0 To UBound(EffectSplit)
                If Left(EffectSplit(i), Len("Spellboost&")) = "Spellboost&" Then
                    ResultX = Val(Mid(EffectSplit(i), 1 + Len("Spellboost&"), 10))
                End If
            Next i
            
            ' 从卡牌效果中提取额外的Spellboost值
            EffectSplit = GetEffect(ThisCard)
            For i = 1 To UBound(EffectSplit)
                If EffectSplit(i) = "" Then Exit For
                If Left(EffectSplit(i), Len("Spellboost|")) = "Spellboost|" Then
                    Str = Mid(EffectSplit(i), 1 + Len("Spellboost|"), 100)
                    If Left(Str, 2) = "X=" Then ResultX = ResultX + Val(Mid(Str, 1 + Len("X="), 10))
                    Exit For
                End If
            Next i
        
        ' 土之印计数
        ElseIf Xrr(k) = "EarthSigil" Then
            Arr = Down("Field", ThisCard)
            For i = 1 To UBound(Arr, 1)
                ' 查找EarthSigil效果
                j = InStr(Arr(i, 5), "EarthSigil&")
                If j > 0 Then
                    ResultX = Val(Mid(Arr(i, 5), j + Len("EarthSigil&"), 10))
                    Exit For
                End If
            Next i
        
        ' ========== 融合相关计算 ==========
        ElseIf Left(Xrr(k), 5) = "Fuse&" Then
            Brr = SplitEffect(DownByMatchID(ThisCard, "Effect"))
            Str = Empty
            ' 提取所有融合效果
            For i = 0 To UBound(Brr)
                If Left(Brr(i), Len("Fuse|")) = "Fuse|" Then Str = Str & "&" & Mid(Brr(i), 1 + Len("Fuse|"), 100)
            Next i
            
            If Str <> "" Then
                Arr = Split(Str, "&")
                ' 根据融合类型计算
                For i = 1 To UBound(Arr)
                    If Xrr(k) = "Fuse&Cards" Then
                        ResultX = ResultX + 1
                    ElseIf Xrr(k) = "Fuse&Types" Then
                        ' 检查类型是否重复
                        For j = 0 To i
                            If Arr(i) = Arr(j) Then Exit For
                        Next j
                        If i = j Then ResultX = ResultX + 1
                    ElseIf Xrr(k) = "Fuse&Costs" Then
                        ResultX = ResultX + DownByID(CStr(Arr(i)), "Cost")
                    End If
                Next i
            End If
            
        ' ========== 卡牌属性计算 ==========
        ElseIf Xrr(k) = "Cost" Or _
                Xrr(k) = "Attack" Or _
                Xrr(k) = "Defense" Or _
                Xrr(k) = "Countdown" Or _
                Xrr(k) = "OrigCost" Or _
                Xrr(k) = "OrigAttack" Or _
                Xrr(k) = "OrigDefense" Or _
                Xrr(k) = "OrigCountdown" Then
            k = k + 1
            Arr = GetGoal(Xrr(k), ThisCard, InputEffect)
            Xrr(k) = "Goal&" & PreviousGoal
            ' 累加所有目标的属性值
            For i = 1 To UBound(Arr)
                ResultX = ResultX + GetModel(Xrr(k - 1), DownByMatchID(CStr(Arr(i)), "ID"), DownByMatchID(CStr(Arr(i)), "Effect"))
            Next i
        
        ' ========== 伤害计算 ==========
        ElseIf Xrr(k) = "Damaged" Then
            k = k + 1
            Arr = GetGoal(Xrr(k), ThisCard, InputEffect)
            Xrr(k) = "Goal&" & PreviousGoal
            ' 累加所有目标的伤害值
            For i = 1 To UBound(Arr)
                Str = DownByMatchID(CStr(Arr(i)), "Effect")
                j = InStr(Str, "Damaged&")
                If j > 0 Then ResultX = ResultX + Val(Mid(Str, j + Len("Damaged&"), 10))
            Next i
                
        ' ========== 计数计算 ==========
        ' 张数/种类数
        ElseIf Xrr(k) = "Cards" Or Xrr(k) = "Types" Then
            k = k + 1
            Arr = GetGoal(Xrr(k), ThisCard, InputEffect)
            Xrr(k) = "Goal&" & PreviousGoal
            For i = 1 To UBound(Arr)
                If Xrr(k - 1) = "Cards" Then
                    ResultX = ResultX + 1
                ElseIf Xrr(k - 1) = "Types" Then
                    ' 检查类型是否重复
                    For j = 0 To i
                        If DownByMatchID(CStr(Arr(i)), "ID") = DownByMatchID(CStr(Arr(j)), "ID") Then Exit For
                    Next j
                    If i = j Then ResultX = ResultX + 1
                End If
            Next i
        
        ' ========== 回合统计计算 ==========
        ElseIf InStr(Xrr(k), "ThisTurn") > 0 Or InStr(Xrr(k), "ThisMatch") > 0 Then
            Brr = Split(Xrr(k), "-")
            Arr = Down("Steps")
            Str = Down("Turn")
            
            ' 筛选符合条件的步骤记录
            For i = 1 To UBound(Arr, 1)
                If Arr(i, 2) <> Brr(1) Then Arr(i, 1) = Empty
                If Arr(i, 1) <> Str And Brr(0) = "ThisTurn" Then Arr(i, 1) = Empty
            Next i
            Arr = NonEmptyArray(Arr)
            
            ' 筛选目标匹配的记录
            k = k + 1
            Call GetGoal("AnyArea-" & Xrr(k), ThisCard, InputEffect, Xrr(k))
            Xrr(k) = "Goal"
            For i = 1 To UBound(Arr, 1)
                If InStr(PreviousGoal, Arr(i, 3)) = 0 Then
                    Arr(i, 1) = Empty
                Else
                    Xrr(k) = Xrr(k) & "&" & Arr(i, 3)
                End If
            Next i
            Arr = NonEmptyArray(Arr)
            
            ' 计算次数/种类数
            If Brr(2) = Types Then
                For i = 1 To UBound(Arr, 1)
                    For j = 1 To i
                        If DownByMatchID(CStr(Arr(i)), "ID") = DownByMatchID(CStr(Arr(j)), "ID") Then Exit For
                    Next j
                    If i < j Then Arr(i, 1) = Empty
                Next i
                Arr = NonEmptyArray(Arr)
            End If
            If Arr(1, 1) <> "" Then result = UBound(Arr, 1) Else result = 0
            
        ' ========== 直接数值 ==========
        ElseIf Val(Xrr(k)) > 0 Then
            result = Val(Xrr(k))
        End If
        
        ' ========== 最终计算处理 ==========
        If ResultX > 0 Then
            If Calculate = "HalveCount" Then
                result = -Int(-(ResultX / 2))
            ElseIf Calculate = "DoubleCount" Then
                result = ResultX * 2
            ElseIf Calculate = "Difference" Then
                If result = 0 Then result = ResultX Else result = Abs(result - ResultX)
            ElseIf Calculate = "Sum" Then
                If result = 0 Then result = ResultX Else result = result + ResultX
            Else
                result = ResultX
            End If
            ResultX = 0
        End If
    Next k
    
    ' ========== 返回结果 ==========
    GetX = result & "=" & Join(Xrr, "|")
End Function
