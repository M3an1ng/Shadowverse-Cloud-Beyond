Attribute VB_Name = "·4·5·辅助·GetModel"
' ==================================================
' 函数功能：获取卡牌属性模型值
' 参数：Model - 模型类型（OrigCountdown、Countdown、OrigAttack、Attack等）
'        ID - 卡牌ID
'        Effect - 效果字符串
'        MatchID - 可选，匹配ID（用于特殊角色处理）
' 返回：Integer - 计算后的属性值
' 说明：根据模型类型计算卡牌的各种属性值，包括攻击力、生命值、费用、倒数等
' ==================================================
Function GetModel(Model As String, ID As String, Effect As String, Optional MatchID As String = "") As Integer
    ' ========== 变量声明 ==========
    Dim i As Integer            ' 返回值
    Dim j As Integer, k As Integer   ' 循环计数器和位置索引
    Dim Str As String           ' 临时字符串
    Dim Arr() As String         ' 效果分割数组
    Dim Brr() As String         ' 临时分割数组
    
    ' ========== 倒数相关模型 ==========
    ' 原始倒数值（从卡牌数据获取）
    If Model = "OrigCountdown" Then
        i = Val(DownByID(ID, "CountDown"))
        
    ' 当前倒数值（从效果字符串解析）
    ElseIf Model = "Countdown" Then
        Arr = SplitEffect(Effect)
        For j = 0 To UBound(Arr)
            If Left(Arr(j), Len("Countdown&")) = "Countdown&" Then
                i = Val(Mid(Arr(j), 1 + Len("Countdown&"), 10))   ' 提取倒数值
                Exit For
            End If
        Next j
        
    ' ========== 攻击力相关模型 ==========
    ' 原始攻击力（从卡牌数据获取）
    ElseIf Model = "OrigAttack" Then
        i = Val(DownByID(ID, "Attack"))
        
    ' 当前攻击力（考虑效果修正）
    ElseIf Model = "Attack" Then
        i = Val(DownByID(ID, "Attack"))   ' 获取基础攻击力
        Arr = SplitEffect(Effect)
        
        ' 遍历所有效果，处理攻击力修正
        For j = 0 To UBound(Arr)
            ' 查找包含"/"但不包含"{"的效果（排除条件效果）
            If InStr(Arr(j), "/") > 0 And InStr(Arr(j), "{") = 0 Then
                Brr = Split(Arr(j), "|")
                Str = Brr(UBound(Brr))   ' 获取效果数值部分
                
                ' 根据操作符类型计算攻击力
                If Left(Str, 1) = "$" Then
                    i = Val(Mid(Str, 2, 10))        ' 设为固定值
                ElseIf Left(Str, 1) = "*" Then
                    i = i * Val(Mid(Str, 2, 10))    ' 倍数修正
                ElseIf Left(Str, 1) = "%" Then
                    i = -1 * Int(-1 * i / Val(Mid(Str, 2, 10)))   ' 除法修正（进位取整）
                Else
                    i = i + Val(Str)                ' 加减修正
                End If
            End If
        Next j
        If i < 0 Then i = 0   ' 攻击力不能为负
        
    ' ========== 生命值相关模型 ==========
    ' 原始生命值
    ElseIf Model = "OrigDefense" Then
        ' 特殊角色处理（F1000或S1000）
        If MatchID = "F1000" Or MatchID = "S1000" Then
            i = Down("MaxDefense", MatchID)   ' 从界面获取最大生命值
        Else
            i = Val(DownByID(ID, "Defense"))  ' 从卡牌数据获取生命值
        End If
        
    ' 当前生命值（考虑效果修正和伤害）
    ElseIf Model = "Defense" Then
        ' 特殊角色处理（F1000或S1000）
        If MatchID = "F1000" Or MatchID = "S1000" Then
            i = Down("Defense", MatchID)   ' 从界面获取当前生命值
        Else
            i = Val(DownByID(ID, "Defense"))   ' 获取基础生命值
            Arr = SplitEffect(Effect)
            
            ' 处理生命值修正效果
            For j = 0 To UBound(Arr)
                k = InStr(Arr(j), "/")
                ' 查找包含"/"但不包含"{"的效果（排除条件效果）
                If k > 0 And InStr(Arr(j), "{") = 0 Then
                    ' 根据操作符类型计算生命值
                    If Mid(Arr(j), k + 1, 1) = "$" Then
                        i = Val(Mid(Arr(j), k + 2, 10))        ' 设为固定值
                    ElseIf Mid(Arr(j), k + 1, 1) = "*" Then
                        i = i * Val(Mid(Arr(j), k + 2, 10))    ' 倍数修正
                    ElseIf Mid(Arr(j), k + 1, 1) = "%" Then
                        i = -1 * Int(-1 * i / Val(Mid(Arr(j), k + 2, 10)))   ' 除法修正（进位取整）
                    Else
                        i = i + Val(Mid(Arr(j), k + 1, 10))    ' 加减修正
                    End If
                End If
            Next j
            
            ' 处理伤害效果（减少生命值）
            For j = 0 To UBound(Arr)
                If Left(Arr(j), Len("Damaged&")) = "Damaged&" Then
                    i = i - Val(Mid(Arr(j), 1 + Len("Damaged&"), 10))   ' 减去伤害值
                End If
            Next j
        End If
        
    ' ========== 费用相关模型 ==========
    ' 原始费用
    ElseIf Model = "OrigCost" Then
        i = Val(DownByID(ID, "Cost"))
        
    ' 当前费用（考虑效果修正）
    ElseIf Model = "Cost" Then
        i = Val(DownByID(ID, "Cost"))   ' 获取基础费用
        Arr = SplitEffect(Effect)
        
        ' 处理费用修正效果
        For j = 0 To UBound(Arr)
            k = InStr(Arr(j), "Cost")
            ' 查找包含"Cost"但不包含"{"的效果
            If k > 0 And InStr(Arr(j), "{") = 0 Then
                Str = Mid(Arr(j), k + Len("Cost"), 10)   ' 提取费用修正部分
                
                ' 根据操作符类型计算费用
                If Left(Str, 1) = "$" Then
                    i = Val(Mid(Str, 2, 10))        ' 设为固定值
                ElseIf Left(Str, 1) = "*" Then
                    i = i * Val(Mid(Str, 2, 10))    ' 倍数修正
                ElseIf Left(Str, 1) = "%" Then
                    i = -1 * Int(-1 * i / Val(Mid(Str, 2, 10)))   ' 除法修正（进位取整）
                Else
                    i = i + Val(Str)                ' 加减修正
                End If
            End If
        Next j
        If i < 0 Then i = 0   ' 费用不能为负
    End If
    
    GetModel = i   ' 返回计算结果
End Function
