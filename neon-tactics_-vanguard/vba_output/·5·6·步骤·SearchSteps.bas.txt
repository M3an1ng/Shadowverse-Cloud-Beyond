Attribute VB_Name = "·5·6·步骤·SearchSteps"
Function SearchSteps(MatchID As String, InputAction As String, Optional InputTurn As String = "") As Boolean
    ' ========== 变量声明 ==========
    Dim i%, j%, k%, m%, n%  ' 循环计数器
    Dim Arr As Variant          ' Steps数据数组
    SearchSteps = False
    
    ' ========== 查找最近的重要动作 ==========
    ' 获取Steps区域的所有步骤记录
    Arr = Down("Steps")
    
    For i = 1 To UBound(Arr, 1)
        If InputTurn = "" Or InputTurn = Arr(i, 1) Then
            If InputAction = Arr(i, 2) Then
                If InStr(Arr(i, 3), MatchID) > 0 Or MatchID = "AnyCard" Then
                    SearchSteps = True
                    Exit Function
                End If
            End If
        End If
    Next i
End Function
