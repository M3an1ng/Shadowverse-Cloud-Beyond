Attribute VB_Name = "·3·5·录卡·SelectDeck"
' ==================================================
' 过程功能：选择并加载已保存的卡组
' 说明：从Deck工作表中选择一个卡组并加载到原卡组界面
' ==================================================
Sub SelectDeck()
    ' ========== 优化设置 ==========
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    Application.DisplayAlerts = False
    
    ' ========== 变量声明 ==========
    Dim Decks As Variant         ' Deck工作表数据数组
    Dim i As Integer, j As Integer, k As Integer   ' 循环计数器
    Dim options() As String      ' 选择对话框选项数组
    Dim Str As String            ' 选项字符串
    Dim DeckName As String       ' 选择的卡组名称
    Dim OrigDeck As Variant      ' 原卡组数据数组
    
    ' ========== Deck工作表数据初始化 ==========
    ' 读取Deck工作表全部数据（42行×100列）
    Decks = Range(Sheets("Deck").Cells(1, 1), Sheets("Deck").Cells(42, 100))
    
    ' ========== 原卡组数据初始化 ==========
    OrigDeck = Down("OrigDeck")   ' 获取当前原卡组结构
    ' 重新初始化原卡组数组（清空现有数据）
    ReDim OrigDeck(1 To UBound(OrigDeck, 1), 1 To UBound(OrigDeck, 2))
    
    ' ========== 卡组选择对话框 ==========
    ' 构建选择对话框的选项字符串
    Str = Translate("Select") & Translate("Deck")   ' 基础提示：选择卡组
    
    ' 遍历所有卡组，构建选项列表
    For k = 1 To UBound(Decks, 2)
        If Decks(1, k) <> "" Then
            Str = Str & "|" & Decks(1, k)   ' 添加卡组名称到选项
        End If
        k = k + 3   ' 每组卡组占用3列，跳过3列
    Next k
    
    ' 显示选择对话框并获取用户选择
    options = Split(Str, "|")               ' 按竖线分割选项字符串
    i = Val(ShowChoice(options, 100, 4))    ' 显示选择对话框，按钮宽度100，4列
    
    ' ========== 用户选择处理 ==========
    If i > 0 Then
        DeckName = options(i)   ' 获取用户选择的卡组名称
    Else
        Exit Sub   ' 用户取消选择，直接退出过程
    End If
    
    ' ========== 卡组数据加载 ==========
    ' 遍历Deck工作表查找目标卡组
    For k = 1 To UBound(Decks, 2)
        ' 检查当前列组是否匹配目标卡组名称
        If Decks(1, k) = DeckName Then
            ' ========== 卡牌数据复制 ==========
            ' 将Deck工作表中的卡组数据复制到原卡组数组
            For i = 1 To UBound(OrigDeck, 1)
                OrigDeck(i, 1) = Decks(i + 2, k)       ' 复制费用数据（第1列）
                OrigDeck(i, 2) = Decks(i + 2, k + 1)   ' 复制卡牌名称数据（第2列）
                OrigDeck(i, 3) = Decks(i + 2, k + 2)   ' 复制数量数据（第3列）
            Next i
            
            ' ========== 界面数据更新 ==========
            Call Up("OrigDeck", OrigDeck)           ' 更新原卡组界面数据
            Call Up("Class", Decks(1, k + 2))       ' 更新职业信息
            Call Up("DeckName", Decks(1, k))        ' 更新卡组名称
            
            Exit For   ' 找到并加载后退出循环
        End If
        
        k = k + 3   ' 每组卡组占用3列，跳过3列
    Next k
    
CleanUp:
    ' ========== 恢复应用程序状态 ==========
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
End Sub
