Attribute VB_Name = "·9·2·文件·ShowImageFromPath"
' ==================================================
' 函数功能：从指定路径显示图片
' 参数：searchPath - 搜索路径, imageName - 图片文件名
' 返回：Boolean - 显示是否成功
' 说明：在指定路径中递归查找图片文件并使用窗体显示
' ==================================================
Function ShowImageFromPath(searchPath As String, imageName As String) As Boolean
    On Error GoTo ErrorHandler
    
    ' ========== 路径存在性检查 ==========
    ' 检查指定路径是否存在
    If Dir(searchPath, vbDirectory) = "" Then
        ShowImageFromPath = False
        Exit Function
    End If
    
    ' ========== 图片文件查找 ==========
    ' 递归查找图片文件
    Dim foundImagePath As String
    foundImagePath = FindImageRecursive(searchPath, imageName)
    
    ' ========== 查找结果验证 ==========
    ' 如果没找到图片，返回False
    If foundImagePath = "" Then
        ShowImageFromPath = False
        Exit Function
    End If
    
    ' ========== 图片显示处理 ==========
    ' 使用预创建窗体显示图片
    Call DisplayImageForm(foundImagePath, imageName)
    
    ShowImageFromPath = True
    Exit Function
    
ErrorHandler:
    ShowImageFromPath = False
End Function

' ==================================================
' 函数功能：递归查找图片文件
' 参数：folderPath - 文件夹路径, imageName - 图片文件名
' 返回：String - 找到的图片完整路径
' 说明：在文件夹及其子文件夹中递归搜索指定名称的图片文件
' 支持多种图片格式：jpg, jpeg, png, bmp, gif, tiff
' ==================================================
Private Function FindImageRecursive(folderPath As String, imageName As String) As String
    ' ========== 变量声明 ==========
    Dim fso As Object
    Dim folder As Object
    Dim subFolder As Object
    Dim file As Object
    
    ' ========== 创建文件系统对象 ==========
    Set fso = CreateObject("Scripting.FileSystemObject")
    
    ' 检查文件夹是否存在
    If Not fso.FolderExists(folderPath) Then
        FindImageRecursive = ""
        Exit Function
    End If
    
    Set folder = fso.GetFolder(folderPath)
    
    ' ========== 当前文件夹搜索 ==========
    For Each file In folder.Files
        ' 检查文件名是否匹配（不区分大小写）
        If LCase(file.Name) = LCase(imageName) Or _
           LCase(file.Name) = LCase(imageName & ".jpg") Or _
           LCase(file.Name) = LCase(imageName & ".jpeg") Or _
           LCase(file.Name) = LCase(imageName & ".png") Or _
           LCase(file.Name) = LCase(imageName & ".bmp") Or _
           LCase(file.Name) = LCase(imageName & ".gif") Or _
           LCase(file.Name) = LCase(imageName & ".tiff") Then
            
            FindImageRecursive = file.Path
            Exit Function
        End If
    Next file
    
    ' ========== 子文件夹递归搜索 ==========
    For Each subFolder In folder.SubFolders
        Dim result As String
        result = FindImageRecursive(subFolder.Path, imageName)
        If result <> "" Then
            FindImageRecursive = result
            Exit Function
        End If
    Next subFolder
    
    ' ========== 未找到处理 ==========
    FindImageRecursive = ""
End Function

' ==================================================
' 过程功能：在窗体中显示图片
' 参数：imagePath - 图片完整路径, CardName - 卡片名称
' 说明：使用预创建的窗体加载并显示图片，保持原始比例
' ==================================================
Private Sub DisplayImageForm(imagePath As String, CardName As String)
    On Error GoTo ErrorHandler
    
    Dim finalImagePath As String
    finalImagePath = GetCompatibleImagePath(imagePath)
    
    If finalImagePath = "" Then
        MsgBox "无法加载或转换图片: " & imagePath, vbExclamation
        Exit Sub
    End If
    
    With ImageDisplayForm
        .Caption = CardName
        .imgPreview.PictureSizeMode = fmPictureSizeModeZoom
        .imgPreview.Picture = LoadPicture(finalImagePath)
        .Show vbModeless
    End With
    
    ' 清理临时文件（如果是转换生成的）
    If finalImagePath <> imagePath Then
        ' 可以在这里添加延迟删除逻辑，或者不删除缓存以提高性能
    End If
    
    Exit Sub
    
ErrorHandler:
    MsgBox "无法加载图片: " & imagePath & vbCrLf & "错误: " & Err.Description, vbExclamation
    ImageDisplayForm.Hide
End Sub

' 获取兼容的图片路径，自动转换不兼容的格式
Private Function GetCompatibleImagePath(originalPath As String) As String
    On Error GoTo ErrorHandler
    
    ' 检查文件是否存在
    If Dir(originalPath) = "" Then
        GetCompatibleImagePath = ""
        Exit Function
    End If
    
    ' 获取文件扩展名
    Dim fileExt As String
    fileExt = LCase(Right(originalPath, 4))
    
    ' 对于非PNG文件，直接尝试加载
    If fileExt <> ".png" Then
        ' 测试是否能直接加载
        TestLoadPicture originalPath
        GetCompatibleImagePath = originalPath
        Exit Function
    End If
    
    ' 对于PNG文件，尝试直接加载
    If CanLoadPicture(originalPath) Then
        GetCompatibleImagePath = originalPath
        Exit Function
    End If
    
    ' 如果直接加载失败，转换为BMP
    GetCompatibleImagePath = ConvertPngToBmp(originalPath)
    
    Exit Function
    
ErrorHandler:
    GetCompatibleImagePath = ""
End Function

' 测试图片是否能加载
Private Function CanLoadPicture(imagePath As String) As Boolean
    On Error GoTo ErrorHandler
    
    ' 创建临时图片对象测试加载
    Dim testControl As Object
    Set testControl = CreateObject("Forms.Image.1")
    testControl.Picture = LoadPicture(imagePath)
    
    CanLoadPicture = True
    Exit Function
    
ErrorHandler:
    CanLoadPicture = False
End Function

' 测试加载图片（不处理错误）
Private Sub TestLoadPicture(imagePath As String)
    Dim testPic As Object
    Set testPic = LoadPicture(imagePath)
End Sub

' 使用WIA转换PNG到BMP
Private Function ConvertPngToBmp(pngPath As String) As String
    On Error GoTo ErrorHandler
    
    ' 需要引用 "Windows Image Acquisition Library v2.0"
    Dim wia As Object
    Dim img As Object
    Dim bmpPath As String
    
    ' 生成BMP文件路径（临时文件）
    bmpPath = Left(pngPath, Len(pngPath) - 4) & "_temp.bmp"
    
    ' 如果临时文件已存在，先删除
    If Dir(bmpPath) <> "" Then Kill bmpPath
    
    ' 使用WIA转换
    Set wia = CreateObject("WIA.ImageFile")
    wia.LoadFile pngPath
    
    Set img = CreateObject("WIA.ImageProcess")
    
    Dim filter As Object
    Set filter = img.FilterInfos("Convert").FilterObject
    filter.FormatID = "{B96B3CAB-0728-11D3-9D7B-0000F81EF32E}" ' BMP格式GUID
    
    img.Filters.Add filter
    Set wia = img.Apply(wia)
    wia.SaveFile bmpPath
    
    ConvertPngToBmp = bmpPath
    Exit Function
    
ErrorHandler:
    ' WIA不可用，尝试备用方法
    ConvertPngToBmp = ConvertPngToBmp_Alternative(pngPath)
End Function

' 备用转换方法（如果WIA不可用）
Private Function ConvertPngToBmp_Alternative(pngPath As String) As String
    On Error GoTo ErrorHandler
    
    ' 使用简单的文件复制和重命名（这实际上不会转换格式，但可以测试）
    Dim bmpPath As String
    bmpPath = Left(pngPath, Len(pngPath) - 4) & "_copy.bmp"
    
    ' 复制文件并更改扩展名
    FileCopy pngPath, bmpPath
    
    ' 测试是否能加载这个"BMP"文件
    If CanLoadPicture(bmpPath) Then
        ConvertPngToBmp_Alternative = bmpPath
    Else
        Kill bmpPath ' 删除无效文件
        ConvertPngToBmp_Alternative = ""
    End If
    
    Exit Function
    
ErrorHandler:
    ConvertPngToBmp_Alternative = ""
End Function

' ==================================================
' 函数功能：从完整路径中提取文件名
' 参数：fullPath - 完整文件路径
' 返回：String - 文件名
' 说明：从完整路径字符串中提取文件名部分
' ==================================================
Private Function GetFileName(fullPath As String) As String
    GetFileName = Mid(fullPath, InStrRev(fullPath, "\") + 1)
End Function
