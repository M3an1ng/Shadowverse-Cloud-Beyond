Attribute VB_Name = "·7·8·对局·MatchFuse"
' ==================================================
' 过程功能：处理卡牌融合操作的入口函数
' 说明：检查融合条件并执行融合操作
' 包含回合检查、手牌选择检查等逻辑
' ==================================================
Sub MatchFuse()
    ' ========== 优化设置 ==========
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    Application.DisplayAlerts = False
    
    ' ========== 变量声明 ==========
    Dim Arr As Variant
    Dim MatchID As String
    
    ' ========== 回合权限检查 ==========
    ' 不是自己的回合直接退出
    If Left(Down("Turn"), 1) <> Left(Down("FirstMove"), 1) Then GoTo CleanUp
    
    ' ========== 手牌选择检查 ==========
    ' 是否选择了手牌
    Arr = IsSelected("Hand")
    If IsEmpty(Arr) Then GoTo CleanUp
    
    ' ========== 手牌信息获取 ==========
    ' 下载手牌信息
    Arr = Down("Hand", Left(Down("FirstMove"), 1) & 1000)
    ' 获取MatchID，如果为空则刷新并退出
    If Selection.Column > UBound(Arr, 1) Or Arr(1, 1) = Empty Then
        Call UIRefresh
        GoTo CleanUp
    End If
    MatchID = Arr(Selection.Column, 1)
    
    ' ========== 融合状态检查 ==========
    ' 查询是否已融合，已融合则直接退出
    Arr = SplitEffect(DownByMatchID(MatchID, "Effect"))
    For i = 0 To UBound(Arr)
        If Arr(i) = "HaveFused" Then GoTo CleanUp
    Next i
    
    ' ========== 融合目标获取 ==========
    ' 获取所有可融合的目标选项
    Arr = GetEffect(MatchID)
    For i = 1 To UBound(Arr)
        If Left(Arr(i), Len("Fuse|")) = "Fuse|" Then
            Call GetGoal("Select&0-Allied-Hand-Other-" & Mid(Arr(i), 1 + Len("Fuse|"), 100), MatchID)
            If PreviousGoal = "" Then GoTo CleanUp
            Exit For
        End If
    Next i
    If i > UBound(Arr) Then GoTo CleanUp
    
    ' ========== 执行融合 ==========
    Call FuseFunction(MatchID, PreviousGoal)
    
CleanUp:
    ' ========== 恢复应用程序状态 ==========
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
End Sub

' ==================================================
' 函数功能：执行具体的融合逻辑
' 参数：MatchID - 要进行融合的卡牌MatchID
' 说明：处理融合目标选择、融合记录生成、融合效果触发等
' 包括重复选项过滤和融合状态标记
' ==================================================
Function FuseFunction(MatchID As String, FuseCards As String) As Boolean
    ' ========== 变量声明 ==========
    Dim i%, j%
    Dim Str As String
    Dim MatchIDrr() As String
    
    ' ========== 融合记录生成 ==========
    ' 生成融合记录并更新卡牌效果
    MatchIDrr = Split(FuseCards, "&")
    Str = "Fuse|"
    For i = 0 To UBound(MatchIDrr)
        Str = Str & DownByMatchID(MatchIDrr(i), "ID") & "&"
    Next i
    Str = DownByMatchID(MatchID, "Effect") & Left(Str, Len(Str) - 1) & "||HaveFused||"
    Call UpByMatchID(MatchID, "Effect", Str)
    
    ' ========== 能力触发处理 ==========
    ' 处理被融合卡牌的能力触发
    For i = 0 To UBound(MatchIDrr)
        Call UpByMatchID(MatchIDrr(i), "Area", "Fused")
    Next i
    
    ' 处理融合关键字能力触发
    Call GetAction("Keyword&Fuse", MatchID, True)
    Call GetAction("Keyword&Fuse", MatchID, UBound(MatchIDrr) + 1)
    
    ' 处理被融合卡牌的能力触发
    For i = 0 To UBound(MatchIDrr)
        Call GetAction("Fused", MatchIDrr(i), True)
        Call GetAction("Fused", MatchIDrr(i))
    Next i
    
    ' ========== 清理和刷新 ==========
    ' 清空能力队列
    Call EraseQueue
    
    ' 刷新界面和步骤区域
    Call UIRefresh
    Call UpStepsArea
End Function
