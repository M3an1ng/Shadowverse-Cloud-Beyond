'减半计数
Private Sub HalveCount_Click()
    If XEffect <> "" Then HalveCount.Value = False
    '互斥逻辑：选择当前按钮时取消其他系数选择
    If HalveCount.Value Then
        DoubleCount.Value = False
        Difference.Value = False
        Sum.Value = False
    End If
End Sub

'命名计数
Private Sub NamingCount_Click()
    Dim Arr As Variant
    Dim i%, j%
    Dim Str As String
    
    '构建命名计数选项字符串
    Str = "NamingCount"
    Arr = Down("Count")
    For i = 1 To UBound(Arr, 1)
        If Arr(i, 1) = "" Then Exit For
        '排除Class类型
        If Mid(Arr(i, 1), 2, 100) <> "Class" Then
            Str = Str & "|" & Mid(Arr(i, 1), 2, 100)
        End If
        i = i + 1
    Next i
    
    '显示选择对话框
    Dim options() As String
    options = Split(SentenceTranslate(Str), "|")
    i = Val(ShowChoice(options, 50, 4))
    If i = 0 Then Exit Sub
    
    '记录命名计数效果
    XEffect = XEffect & "NamingCount&" & Translate(options(i))
    Call UnloadForm
End Sub

'窗体激活事件 - 初始化
Private Sub UserForm_Activate()
    '清空效果字符串
    XEffect = Empty
    
    '设置默认选择：友方、卡牌计数、本回合
    Allied.Value = True
    Cards.Value = True
    ThisTurn.Value = True
    
    If CardEffect <> "" Then EnterCard·5·X.Caption = SentenceTranslate(CardEffect)
End Sub

' ==================================================
' 函数功能：卸载窗体并处理效果字符串
' 说明：根据选择的系数类型进行相应处理，完成效果字符串构建
' ==================================================
Function UnloadForm()
    Dim i%

    '差值或求和计数处理
    If InStr(XEffect, "Difference") > 0 Or InStr(XEffect, "Sum") > 0 Then
        'XEffect = XEffect & "}"
    '减半计数处理
    ElseIf HalveCount.Value Then
        XEffect = "Halve|" & XEffect
    '加倍计数处理
    ElseIf DoubleCount.Value Then
        XEffect = "DoubleCount|" & XEffect
    '差值计数处理
    ElseIf Difference.Value Then
        XEffect = "Difference|" & XEffect & "|"
        '重置默认选择
        Allied.Value = True
        Cards.Value = True
        ThisTurn.Value = True
        Difference.Value = False
        EnterCard·5·X.Caption = SentenceTranslate(XEffect)
        
        '询问差值常数
        i = Val(InputBox("输入差值：" & Chr(13) & "提示：输入0代表不输入常数", Translate("Difference"), "0"))
        
        If i = 0 Then Exit Function Else XEffect = XEffect & i
    '求和计数处理
    ElseIf Sum.Value Then
        XEffect = "Sum|" & XEffect & "|"
        '重置默认选择
        Allied.Value = True
        Cards.Value = True
        ThisTurn.Value = True
        Sum.Value = False
        EnterCard·5·X.Caption = SentenceTranslate(XEffect)
        
        '询问求和常数
        i = Val(InputBox("输入差值：" & Chr(13) & "提示：输入0代表不输入常数", Translate("Difference"), "0"))
        
        If i = 0 Then Exit Function Else XEffect = XEffect & i
    End If
    
    '卸载当前窗体
    Unload EnterCard·5·X
End Function
