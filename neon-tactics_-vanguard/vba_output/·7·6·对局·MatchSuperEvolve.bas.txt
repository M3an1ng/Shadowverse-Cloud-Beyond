Attribute VB_Name = "·7·6·对局·MatchSuperEvolve"
' ==================================================
' 过程功能：处理从者超进化操作的入口函数
' 说明：检查超进化条件并执行超进化操作
' 包含回合检查、超进化点检查、进化状态验证等逻辑
' ==================================================
Sub MatchSuperEvolve()
    ' ========== 优化设置 ==========
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    Application.DisplayAlerts = False
    
    ' ========== 变量声明 ==========
    Dim Arr As Variant
    Dim Turn As String
    Dim MatchID As String
    
    ' ========== 回合权限检查 ==========
    ' 不是自己的回合直接退出
    Turn = Down("Turn")
    If Left(Turn, 1) <> Left(Down("FirstMove"), 1) Then GoTo CleanUp
    If Left(Turn, 1) = "F" Then
        If Val(Mid(Turn, 2, 10)) < 7 Then GoTo CleanUp
    Else
        If Val(Mid(Turn, 2, 10)) < 6 Then GoTo CleanUp
    End If
    
    ' ========== 战场选择检查 ==========
    ' 是否选择了战场
    Arr = IsSelected("AlliedField")
    If IsEmpty(Arr) Then GoTo CleanUp
    
    ' ========== 进化状态检查 ==========
    ' 是否已进化或超进化（检查按钮文本状态）
    If Sheets("Main").Shapes("EvolveBtn").TextFrame.Characters.Text = Translate("Evolved") Then GoTo CleanUp
    If Sheets("Main").Shapes("SuperEvolveBtn").TextFrame.Characters.Text = Translate("SuperEvolved") Then GoTo CleanUp
    
    ' ========== 战场信息获取 ==========
    ' 下载战场信息
    Arr = Down("Field", Left(Down("FirstMove"), 1) & 1000)
    ' 获取MatchID，如果为空则刷新并退出
    If Selection.Column - 2 > UBound(Arr, 1) Or Arr(1, 1) = "" Then
        Call UIRefresh
        GoTo CleanUp
    End If
    MatchID = Arr(Selection.Column - 2, 1)
    
    ' ========== 随从检查 ==========
    If DownByID(DownByMatchID(MatchID, "ID"), "Type") <> "Follower" Then GoTo CleanUp
    
    ' ========== 已进化状态检查 ==========
    ' 检查是否已经进化（注释标明不严谨实现）
    If InStr(Arr(Selection.Column - 2, 5), "Evolved||") > 0 Then GoTo CleanUp
    
    ' ========== 超进化点检查 ==========
    ' 是否有EP（进化点）
    Arr = GetEffect(MatchID)
    For i = 1 To UBound(Arr)
        If Arr(i) = "NoSEP" Then Exit For
    Next i
    If i > UBound(Arr) Then
        If Val(Down("AlliedSEP")) <= 0 Then GoTo CleanUp
        Arr = SuperEvolveFunction(MatchID)
    Else
        Arr = SuperEvolveFunction(MatchID, True)
    End If
    
    ' ========== 执行超进化 ==========
    'Arr = SuperEvolveFunction(MatchID)
    If Not Arr Then GoTo CleanUp

    ' ========== 界面刷新 ==========
    Call UIRefresh
    Call UpStepsArea
    
CleanUp:
    ' ========== 恢复应用程序状态 ==========
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
End Sub

' ==================================================
' 函数功能：执行具体的超进化逻辑
' 参数：MatchID - 要超进化的从者MatchID
' 说明：处理超进化效果、消耗超进化点、触发超进化时能力等
' 包括超进化失败的回滚处理
' ==================================================
Function SuperEvolveFunction(MatchID As String, Optional NoSEP As Boolean = False) As Boolean
    ' ========== 变量声明 ==========
    Dim i%, j%
    Dim Arr As Variant
    Dim Flag As Boolean
    Dim Effect As String
    Dim EffectSaved As String
    Dim Effectrr() As String
    Dim EvolveFlag As Boolean
    Dim SuperEvolveFlag As Boolean
    SuperEvolveFunction = True
    
    ' ========== 初始化设置 ==========
    ' 清空能力队列
    QueueAbility = Empty
    QueueMatchID = Empty
    
    ' ========== 基础超进化效果应用 ==========
    ' 应用+3/+3与突进效果，标记为已超进化
    EffectSaved = DownByMatchID(MatchID, "Effect")
    Effect = EffectSaved & "{Rush}||SuperEvolved||"
    Call UpByMatchID(MatchID, "Effect", Effect)
    Call GetAbility("||+3/+3|ThisCard", MatchID)
    
    ' ========== 超进化点消耗 ==========
    ' 消耗1点SEP（超进化点）
    If Not NoSEP Then Call Up("SEP", Down("SEP", MatchID) - 1, MatchID)
    Call UpSteps("Evolve", MatchID, DownByID(DownByMatchID(MatchID, "ID"), "Name"))
    Call UpSteps("SuperEvolve", MatchID, DownByID(DownByMatchID(MatchID, "ID"), "Name"))
    
    ' ========== 超进化效果处理 ==========
    ' 遍历并触发超进化和进化相关效果
    Flag = True
    Effectrr = GetEffect(MatchID)
    For i = 1 To UBound(Effectrr)
        ' 处理超进化效果（注释标明不严谨实现）
        If (Left(Effectrr(i), Len("SuperEvolve")) = "SuperEvolve" Or InStr(Effectrr(i), "Change|SuperEvolve") > 0) And InStr(Effect, "Remove|SuperEvolve") = 0 Then
            Flag = GetTimePoint(Effectrr(i), MatchID, "SuperEvolve")
            SuperEvolveFlag = True
        ' 处理进化效果（超进化时也会触发进化效果）
        ElseIf Left(Effectrr(i), Len("Evolve")) = "Evolve" And InStr(Effect, "Remove|Evolve") = 0 Then
            Flag = GetTimePoint(Effectrr(i), MatchID, "SuperEvolve")
            EvolveFlag = True
        End If
    Next i
    
    ' ========== 超进化失败处理 ==========
    ' 如果选择失败，回滚超进化操作
    If Not Flag Then
        If Not NoSEP Then Call Up("SEP", Down("SEP", MatchID) + 1, MatchID)
        Call EraseSteps("Evolve")
        Call UpByMatchID(MatchID, "Effect", EffectSaved)
        QueueAbility = Empty
        QueueMatchID = Empty
        SuperEvolveFunction = False
        Exit Function
    End If
    
    ' ========== 界面更新 ==========
    ' 更改超进化按钮文本状态
    Sheets("Main").Shapes("SuperEvolveBtn").TextFrame.Characters.Text = Translate("SuperEvolved")
    
    ' ========== 能力触发处理 ==========
    ' 触发进场时超进化效果
    Call GetAction("SuperEvolve", MatchID, True)
    
    '奥义
    Arr = Down("Hand", MatchID)
    For i = 1 To UBound(Arr)
        Effectrr = SplitEffect(CStr(Arr(i, 5)))
        For j = 0 To UBound(Effectrr)
            If Left(Effectrr(j), Len("Skybound&")) = "Skybound&" Then
                k = Val(Mid(Effectrr(j), 1 + Len("Skybound&"), 10))
                k = k + 1
                Effectrr(j) = "Skybound&" & k
                Exit For
            End If
        Next j
        Arr(i, 5) = Join(Effectrr, "||")
        If Arr(i, 5) <> "" Then Arr(i, 5) = Arr(i, 5) & "||"
        Call UpByMatchID(CStr(Arr(i, 1)), "Effect", CStr(Arr(i, 5)))
    Next i
    
    ' 其他卡牌响应超进化事件
    Call GetAction("Evolve", MatchID, "0")
    Call GetAction("SuperEvolve", MatchID, "0")
    If EvolveFlag Then Call GetAction("Keyword&Evolve", MatchID)
    If SuperEvolveFlag Then Call GetAction("Keyword&SuperEvolve", MatchID)
    
    ' ========== 清理和刷新 ==========
    ' 清空能力队列
    Call EraseQueue
End Function
