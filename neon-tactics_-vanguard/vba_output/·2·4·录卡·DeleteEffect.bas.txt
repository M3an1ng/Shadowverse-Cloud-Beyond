Attribute VB_Name = "·2·4·录卡·DeleteEffect"
' ==================================================
' 函数功能：删除效果（卡牌）记录，处理相关引用
' 参数：idValue - 要删除的卡牌ID
'        newID - 替换ID（为空时检查引用，非空时替换引用）
' 返回：Boolean - 删除操作是否成功
' 说明：安全删除卡牌记录，检查并处理其他单元格对该ID的引用
' ==================================================
Function DeleteEffect(idValue As String, newID As String) As Boolean
    On Error GoTo ErrorHandler
    
    ' ========== 输入参数验证 ==========
    ' 验证ID长度是否符合标准（7位数字）
    If Len(idValue) <> Len("1000000") Then Exit Function
    If Len(newID) <> Len("1000000") And newID <> "" Then Exit Function
    
    ' ========== 常量定义 ==========
    Const SHEET_NAME As String = "Effect"     ' 效果数据工作表名称
    Const ID_HEADER As String = "ID"          ' ID列标题
    
    ' ========== 变量声明 ==========
    Dim ws As Worksheet           ' 工作表对象
    Dim headerRow As Range        ' 标题行范围
    Dim idCol As Range            ' ID列范围
    Dim idCell As Range           ' 目标ID单元格
    Dim replaceFound As Boolean   ' 是否找到需要替换的引用
    Dim lastRow As Long           ' 最后数据行号
    Dim col As Range              ' 列循环变量
    Dim cell As Range             ' 单元格循环变量
    
    ' ========== 工作表初始化 ==========
    Set ws = ThisWorkbook.Sheets(SHEET_NAME)   ' 设置Effect工作表
    
    ' 查找ID列的标题位置（第一行）
    Set headerRow = ws.rows(1)
    Set idCol = headerRow.Find(What:=ID_HEADER, LookAt:=xlWhole, MatchCase:=False)
    
    ' 验证ID列标题是否存在
    If idCol Is Nothing Then
        MsgBox "错误：未找到ID列标题", vbExclamation
        DeleteEffect = False
        Exit Function
    End If
    
    ' ========== 引用检查与替换 ==========
    replaceFound = False
    lastRow = ws.Cells(ws.rows.count, idCol.Column).End(xlUp).Row   ' 获取ID列最后一行
    
    ' 遍历所有非ID列，检查是否存在对目标ID的引用
    For Each col In ws.UsedRange.Columns
        If col.Column <> idCol.Column Then   ' 跳过ID列本身
            For Each cell In col.Cells
                ' 跳过标题行，只检查数据行
                If cell.Row > 1 And cell.Row <= lastRow Then
                    ' 检查单元格内容是否包含目标ID
                    If InStr(1, cell.Value, idValue) > 0 Then
                        ' 如果newID为空，说明是删除操作，但存在引用则不允许删除
                        If Len(newID) = 0 Then
                            MsgBox "错误：该卡牌尚存在引用，无法删除", vbExclamation
                            DeleteEffect = False
                            Exit Function
                        End If
                        ' 找到引用并标记，同时进行替换操作
                        replaceFound = True
                        cell.Value = Replace(cell.Value, idValue, newID)   ' 替换引用
                    End If
                End If
            Next cell
        End If
    Next col
    
    ' ========== ID记录删除 ==========
    ' 在ID列中精确查找目标ID
    Set idCell = ws.Columns(idCol.Column).Find( _
        What:=idValue, _
        LookIn:=xlValues, _
        LookAt:=xlWhole, _
        MatchCase:=False)
    
    ' 如果找到目标ID记录，删除整行
    If Not idCell Is Nothing Then
        Application.DisplayAlerts = False   ' 禁用删除确认提示
        idCell.EntireRow.Delete              ' 删除整行记录
        Application.DisplayAlerts = True    ' 恢复确认提示
        DeleteEffect = True                  ' 返回操作成功
        Exit Function
    End If
    
    ' ========== 未找到目标ID处理 ==========
    DeleteEffect = False   ' 未找到指定ID，返回操作失败
    Exit Function

ErrorHandler:
    ' ========== 错误处理 ==========
    Application.DisplayAlerts = True   ' 确保恢复系统提示设置
    DeleteEffect = False               ' 返回操作失败
    MsgBox "操作时出错: " & Err.Description, vbCritical   ' 显示错误信息
End Function
