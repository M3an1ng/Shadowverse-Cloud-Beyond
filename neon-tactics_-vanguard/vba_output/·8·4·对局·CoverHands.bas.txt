Attribute VB_Name = "·8·4·对局·CoverHands"
Sub CoverHands()
    Dim i As Integer
    Dim Arr As Variant
    Dim ws As Worksheet
    Dim copyRange As Range
    
    ' 设置工作表对象，避免重复引用
    Set ws = Sheets("Main")
    
    Arr = Down("FirstMove")
    Arr = Down("Hand", CStr(Arr))
    i = UBound(Arr, 1)
    
    If Arr(1, 1) <> "" Then
        If ws.Range("A14").Interior.ThemeColor = xlThemeColorDark1 Then
            ' 显示手牌分支
            ws.Range(ws.Cells(14, 1), ws.Cells(16, i)).Interior.ThemeColor = xlThemeColorLight1
            ws.Shapes("CoverBtn").TextFrame.Characters.Text = "显示手牌"
            
            Arr = Down("StepsArea")
            For i = 1 To UBound(Arr, 1)
                If Arr(i, 1) = "" Then Exit For
            Next i
            
            ' 直接复制，不通过选择
            Set copyRange = ws.Range(ws.Cells(1, 1), ws.Cells(i + 19, 9))
            copyRange.CopyPicture Appearance:=xlScreen, Format:=xlBitmap
            
        Else
            ' 遮住手牌分支
            ws.Range(ws.Cells(14, 1), ws.Cells(16, i)).Interior.ThemeColor = xlThemeColorDark1
            ws.Shapes("CoverBtn").TextFrame.Characters.Text = "遮住手牌&截图"
            
            ' 取消复制状态
            Application.CutCopyMode = False
            
            ' 直接激活单元格，不通过选择
            ws.Cells(16, i).Activate
        End If
    End If
    
    ' 清理对象
    Set ws = Nothing
    Set copyRange = Nothing
End Sub
