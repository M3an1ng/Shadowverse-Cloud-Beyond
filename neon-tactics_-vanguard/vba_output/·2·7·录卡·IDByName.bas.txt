Attribute VB_Name = "·2·7·录卡·IDByName"
' ==================================================
' 函数功能：根据卡牌名称查找对应的卡牌ID（跳过指定ID）
' 参数：nameValue - 卡牌名称
' 返回：Variant - 卡牌ID或错误信息
' 说明：在Name列中查找指定名称，返回对应的ID，但跳过特定的ID值（1000000）
' ==================================================
Function IDByName(nameValue As String) As Variant
    On Error GoTo ErrorHandler
    
    ' ========== 常量定义 ==========
    Const SHEET_NAME As String = "Effect"      ' 效果数据工作表名称
    Const NAME_HEADER As String = "Name"       ' 名称列标题
    Const ID_HEADER As String = "ID"           ' ID列标题
    Const SKIP_ID As String = "1000000"        ' 要跳过的特殊ID值
    
    ' ========== 变量声明 ==========
    Dim ws As Worksheet           ' 工作表对象
    Dim headerRow As Range        ' 标题行范围
    Dim nameCol As Range          ' 名称列范围
    Dim idCol As Range            ' ID列范围
    Dim nameCell As Range         ' 名称查找结果单元格
    Dim firstFound As Range       ' 首次找到的单元格（用于循环检测）
    Dim foundID As String         ' 找到的ID值
    
    ' ========== 工作表初始化 ==========
    Set ws = ThisWorkbook.Sheets(SHEET_NAME)   ' 设置Effect工作表
    
    ' ========== 列标题查找与验证 ==========
    Set headerRow = ws.rows(1)   ' 设置标题行（第一行）
    
    ' 查找Name列位置
    Set nameCol = headerRow.Find(What:=NAME_HEADER, LookAt:=xlWhole, MatchCase:=False)
    ' 查找ID列位置
    Set idCol = headerRow.Find(What:=ID_HEADER, LookAt:=xlWhole, MatchCase:=False)
    
    ' 验证Name列标题是否存在
    If nameCol Is Nothing Then
        MsgBox "错误：未找到Name列标题", vbExclamation
        IDByName = CVErr(xlErrNA)   ' 返回#N/A错误
        Exit Function
    End If
    
    ' 验证ID列标题是否存在
    If idCol Is Nothing Then
        MsgBox "错误：未找到ID列标题", vbExclamation
        IDByName = CVErr(xlErrNA)   ' 返回#N/A错误
        Exit Function
    End If
    
    ' ========== 名称查找与循环处理 ==========
    ' 在Name列中精确查找指定名称
    Set nameCell = ws.Columns(nameCol.Column).Find( _
        What:=nameValue, _
        LookIn:=xlValues, _
        LookAt:=xlWhole, _
        MatchCase:=False)
    
    ' 记录第一个找到的单元格，用于检测循环
    If Not nameCell Is Nothing Then
        Set firstFound = nameCell
    End If
    
    ' ========== 循环查找符合条件的记录 ==========
    ' 遍历所有匹配的名称，直到找到不是SKIP_ID的记录或回到起点
    Do While Not nameCell Is Nothing
        ' 获取当前行对应的ID值
        foundID = CStr(ws.Cells(nameCell.Row, idCol.Column).Value)
        
        ' 检查ID是否为需要跳过的特殊值
        If foundID <> SKIP_ID Then
            IDByName = foundID   ' 返回符合条件的ID
            Exit Function
        End If
        
        ' 继续查找下一个匹配项
        Set nameCell = ws.Columns(nameCol.Column).FindNext(nameCell)
        
        ' 如果回到起点，退出循环（避免无限循环）
        If nameCell.Address = firstFound.Address Then
            Exit Do
        End If
    Loop
    
    ' ========== 未找到符合条件的记录 ==========
    IDByName = -1   ' 返回-1表示未找到
    Exit Function

ErrorHandler:
    ' ========== 错误处理 ==========
    IDByName = CVErr(xlErrValue)   ' 返回#VALUE!错误
    MsgBox "获取ID时出错: " & Err.Description, vbCritical   ' 显示错误信息
End Function
