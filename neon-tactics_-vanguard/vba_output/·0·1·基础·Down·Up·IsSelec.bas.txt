' ==================================================
' 模块功能：主界面单元格区域配置与数据操作
' 包含功能：定义界面区域坐标、读取区域数据、写入区域数据、区域选择检测
' ==================================================

Option Explicit

' ==================================================
' 函数功能：获取所有界面区域的坐标配置

' ==================================================

        "||EnemyEP|Main|6|2" & _
        "||EnemySEP|Main|7|2" & _
        "||AlliedDefense|Main|8|2" & _

        "||AlliedPP|Main|10|2" & _
        "||AlliedMaxPP|Main|11|2" & _
        "||AlliedEP|Main|12|2" & _
        "||AlliedSEP|Main|13|2" & _
        "||EnemyField|Main|2|3|4|7|Horizontal" & _
        "||AlliedField|Main|8|3|10|7|Horizontal" & _
        "||Hand|Main|14|1|15|9|Horizontal" & _
        "||Crest|Main|17|2|19|6|Horizontal" & _

' ==================================================
' 函数功能：从指定区域读取数据
' 参数：Target - 目标区域名称
'        MatchID - 可选，用于匹配特定ID
' 返回：区域数据，可能是值、数组或空值
' ==================================================
Function Down(Target As String, Optional MatchID As String = "") As Variant
    Dim i As Integer
    Dim CellsRC() As String
    Dim Strr() As String
    Dim Data As Variant
    Dim Str As String
    
    CellsRC = SetCellsRC()   ' 获取所有区域配置
    
    ' 处理特殊数据类型区域
    If Target = "Hand" Or Target = "Field" Or Target = "Crest" Or Target = "Deck" Then
        Data = Down("Data")
        
        ' 根据目标类型和ID过滤数据
        For i = 1 To UBound(Data, 1)
            If Data(i, 4) <> Target Then Data(i, 1) = Empty
            If MatchID <> "" And Left(Data(i, 1), 1) <> Left(MatchID, 1) Then Data(i, 1) = Empty
        Next i
        
        Data = NonEmptyArray(Data)   ' 返回非空数据数组
        Down = Data
        Exit Function
        
    ' 处理战斗属性区域（根据先手判断敌我）
    ElseIf Target = "Defense" Or Target = "MaxDefense" Or _
        Target = "PP" Or Target = "MaxPP" Or _
        Target = "EP" Or Target = "SEP" Then
            If Left(Down("FirstMove"), 1) = Left(MatchID, 1) Then
                Target = "Allied" & Target   ' 我方属性
            Else
                Target = "Enemy" & Target    ' 敌方属性
            End If
            
    ' 处理计数区域查询
    ElseIf MatchID <> "" Then
        Data = Down("Count")
        
        ' 查找匹配的计数项
        For i = 1 To UBound(Data, 1)
            If Data(i, 1) = Left(MatchID, 1) & Target Then
                Down = Data(i, 2)   ' 返回计数值
                Exit Function
            End If
        Next i
    End If
    
    ' 遍历区域配置查找目标区域
    For i = 0 To UBound(CellsRC, 1)
        Strr = Split(CellsRC(i), "|")
        
        If Strr(0) = Target Then
            ' 处理矩形区域（多单元格）
            If UBound(Strr) >= 5 Then
                Down = Range(Sheets(Strr(1)).Cells(Val(Strr(2)), Val(Strr(3))), _
                       Sheets(Strr(1)).Cells(Val(Strr(4)), Val(Strr(5))))
            ' 处理单个单元格
            Else
                Down = Sheets(Strr(1)).Cells(Val(Strr(2)), Val(Strr(3)))
            End If
            
            Exit Function
        End If
    Next i
End Function

' ==================================================
' 函数功能：将数据写入指定区域
' 参数：Target - 目标区域名称
'        Arr - 要写入的数据数组
'        MatchID - 可选，用于匹配特定ID
' 返回：无
' ==================================================
Function Up(Target As String, Arr As Variant, Optional MatchID As String = "") As Variant
    Dim i As Integer
    Dim CellsRC() As String
    Dim Strr() As String
    Dim Data As Variant
    
    CellsRC = SetCellsRC()   ' 获取所有区域配置
    
    ' 处理特殊数据类型区域
    If Target = "Field" Or _
        Target = "Defense" Or Target = "MaxDefense" Or _
        Target = "PP" Or Target = "MaxPP" Or _
        Target = "EP" Or Target = "SEP" Then
            ' 根据先手判断敌我区域
            If Left(Down("FirstMove"), 1) = Left(MatchID, 1) Then
                Target = "Allied" & Target   ' 我方区域
            Else
                Target = "Enemy" & Target    ' 敌方区域
            End If
            
    ' 处理计数区域更新
    ElseIf MatchID <> "" Then
        Data = Down("Count")
        
        ' 更新匹配的计数项
        For i = 1 To UBound(Data, 1)
            If Data(i, 1) = Left(MatchID, 1) & Target Then
                Data(i, 2) = Arr
                Call Up("Count", Data)   ' 写回更新后的计数数据
                Exit Function
            End If
        Next i
    End If
    
    ' 遍历区域配置查找目标区域
    For i = 0 To UBound(CellsRC, 1)
        Strr = Split(CellsRC(i), "|")
        
        If Strr(0) = Target Then
            ' 处理矩形区域（多单元格）
            If UBound(Strr) >= 5 Then
                ' 如果传入数组为空，则重新定义数组大小
                If IsEmpty(Arr) Then
                    ReDim Arr(1 To Val(Strr(4)) - Val(Strr(2)) + 1, _
                             1 To Val(Strr(5)) - Val(Strr(3)) + 1)
                End If
                
                ' 写入数据到指定范围
                Range(Sheets(Strr(1)).Cells(Val(Strr(2)), Val(Strr(3))), _
                      Sheets(Strr(1)).Cells(Val(Strr(4)), Val(Strr(5)))) = Arr
            ' 处理单个单元格
            Else
                Sheets(Strr(1)).Cells(Val(Strr(2)), Val(Strr(3))) = Arr
            End If
            
            Exit Function
        End If
    Next i
End Function

' ==================================================
' 函数功能：检查当前选择是否在目标区域内并返回数据
' 参数：Target - 目标区域名称
' 返回：选中单元格的数据或空值
' ==================================================
Function IsSelected(Target As String) As Variant
    Dim i As Integer
    Dim CellsRC() As String
    Dim Strr() As String
    
    CellsRC = SetCellsRC()   ' 获取所有区域配置
    
    ' 遍历区域配置查找目标区域
    For i = 0 To UBound(CellsRC, 1)
        Strr = Split(CellsRC(i), "|")
        
        If Strr(0) = Target Then
            ' 检查工作表是否匹配
            If ActiveSheet.Name <> Strr(1) Then
                Exit For
                
            ' 处理单个单元格区域
            ElseIf UBound(Strr) = 3 Then
                If Selection.Row = Val(Strr(2)) And Selection.Column = Val(Strr(3)) Then
                    IsSelected = Sheets(Strr(1)).Cells(Selection.Row, Selection.Column)
                End If
                
            ' 处理矩形区域
            ElseIf UBound(Strr) >= 5 Then
                ' 检查选择是否在区域范围内
                If Selection.Row >= Val(Strr(2)) And Selection.Row <= Val(Strr(4)) And _
                   Selection.Column >= Val(Strr(3)) And Selection.Column <= Val(Strr(5)) Then
                   
                    ' 处理有方向设置的区域
                    If UBound(Strr) >= 6 Then
                        If Strr(6) = "Vertical" Then
                            ' 垂直方向：返回整列数据
                            IsSelected = Sheets(Strr(1)).Cells(Selection.Row, Val(Strr(3)))
                            
                            ' OrigDeck特殊处理
                            If Target = "OrigDeck" Then
                                IsSelected = Sheets(Strr(1)).Cells(Selection.Row, Val(Strr(3)) + 1)
                            End If
                            
                        ElseIf Strr(6) = "Horizontal" Then
                            ' 水平方向：返回整行数据
                            IsSelected = Sheets(Strr(1)).Cells(Val(Strr(2)), Selection.Column)
                        End If
                    ' 无方向设置：返回当前单元格
                    Else
                        IsSelected = Sheets("Main").Cells(Selection.Row, Selection.Column)
                    End If
                End If
            End If
            
            Exit Function
        End If
    Next i
    
    IsSelected = Empty   ' 无匹配区域返回空值
End Function
