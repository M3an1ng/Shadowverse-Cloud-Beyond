' ========== 通用点击事件处理程序 ==========
Function ToggleButton_Click(Btn As MSForms.ToggleButton)
    ' 遍历所有控件，处理同组ToggleButton的互斥逻辑
    Dim Flag As Boolean
    Flag = False ' 记录同组是否有按键按下
    
    For Each ctrl In Me.Controls
        ' 只处理同组的 ToggleButton
        If ctrl.Tag = Btn.Tag Then
            ' 核心逻辑：当前按钮按下，其他弹起
            If Btn.Value Then ctrl.Value = (ctrl.Name = Btn.Name)
            If ctrl.Value Then Flag = True
        End If
    Next ctrl
    
    ' 如果同组都没有按钮按下，则选中当前按钮
    If Not Flag Then Btn.Value = True
End Function

' 随机选择目标
Private Sub RandomCard_Click()
    ' 判断Tag限制
    If AllCard.Tag = "All" Or SelectCard.Tag = "Select" Then
        RandomCard.Value = False
        Exit Sub
    End If

    ToggleButton_Click RandomCard
    
    ' 询问选择数量并更新按钮显示
    If RandomCard.Caption = Translate("Random") Then
        Dim Str As String
        Str = NumInputBox(Translate("Random"), "Random")
        
        ' 处理不同名选项（@前缀）
        If Left(Str, 1) = "@" Then
            RandomCard.Caption = "Random" & Str
        Else
            RandomCard.Caption = "Random&" & Str
        End If
        
        RandomCard.Caption = SentenceTranslate(RandomCard.Caption)
    End If
    
    ' 恢复默认显示
    If Not RandomCard.Value Then RandomCard.Caption = Translate("Random")
    
    Call EffectPreview
End Sub

' 手动选择目标
Private Sub SelectCard_Click()
    ' 判断Tag限制
    If AllCard.Tag = "All" Or RandomCard.Tag = "Random" Then
        SelectCard.Value = False
        Exit Sub
    End If
    
    ' 检查是否在可选择的时点
    Dim IsSelect As Boolean
    IsSelect = False
    Dim Arr() As String
    If TimePoint = "" Then
        IsSelect = True
    Else
        Arr = Split(TimePoint, "&")
        If Arr(0) = "Fanfare" Then IsSelect = True
        If Arr(0) = "LastWords" Then IsSelect = True
        If Arr(0) = "Play" Then IsSelect = True
        If Arr(0) = "Evolve" Then IsSelect = True
        If Arr(0) = "SuperEvolve" Then IsSelect = True
        If Arr(0) = "Engage" Then IsSelect = True
        If Arr(0) = "Enhance" Then IsSelect = True
    End If
    
    If Not IsSelect Then
        SelectCard.Value = False
        Exit Sub
    End If
    
    ToggleButton_Click SelectCard
    
    ' 询问选择数量并更新按钮显示
    If SelectCard.Caption = Translate("Select") Then
        SelectCard.Caption = SentenceTranslate("Select&" & NumInputBox(Translate("Select"), "Select"))
    End If
    
    ' 恢复默认显示
    If Not SelectCard.Value Then SelectCard.Caption = Translate("Select")
    
    ' 选择模式下的一些限制
    If SelectCard.Value Then
        Deck.Value = False
        Crest.Value = False
        Faith.Value = False
    End If
    
    ' 选择对手时限制生效区域为战场
    If SelectCard.Value And Enemy.Value Then
        Field.Value = True
        Hand.Value = False
        Deck.Value = False
        ActionCards.Value = False
    End If
    
    Call EffectPreview
End Sub

' 效果预览生成函数
Function EffectPreview(Optional InputBtn As String = "")
    Dim i%
    Dim Str As String
    Dim Flag As Boolean
    
    ' 检查特殊目标类型
    If Not Except.Value Then _
        If ThisCard.Value Or Opposing.Value Or PreviousGoal.Value Or PrePrevious.Value Or PreviousFuse.Value Then Flag = True
        
    ' 确保至少选择一个生效方、区域和类型
    If Not Flag Then
        ' 生效方选择验证
        For i = 1 To 2
            If Not (Allied.Value Or Enemy.Value) Then
                If InputBtn = "Allied" Then
                    Enemy.Value = True
                ElseIf InputBtn = "Enemy" Then
                    Allied.Value = True
                Else
                    Allied.Value = True
                End If
            Else
                Exit For
            End If
        Next i
        
        ' 生效区域选择验证
        For i = 1 To 5
            If Not (Field.Value Or Hand.Value Or Deck.Value Or ActionCards.Value) Then
                Select Case InputBtn
                    Case "Field": Hand.Value = True
                    Case "Hand": Deck.Value = True
                    Case "Deck": ActionCards.Value = True
                    Case "ActionCards": Field.Value = True
                    Case Else: Field.Value = True
                End Select
            Else
                Exit For
            End If
        Next i
        
        ' 生效种类选择验证
        For i = 1 To 6
            If Not (Follower.Value Or Spell.Value Or Amulet.Value Or Leader.Value Or Crest.Value Or Faith.Value) Then
                Select Case InputBtn
                    Case "Follower": Spell.Value = True
                    Case "Spell": Amulet.Value = True
                    Case "Amulet": Leader.Value = True
                    Case "Leader": Crest.Value = True
                    Case "Crest": Faith.Value = True
                    Case "Faith": Follower.Value = True
                    Case Else: Follower.Value = True
                End Select
            Else
                Exit For
            End If
        Next i
    End If
    
    ' 生成目标描述字符串
    Goal = Empty
    Dim StrID As String
    Dim StrID2 As String
    StrID = IDByName(CardName.Value)
    StrID2 = IDByName(CardName2.Value)
    
    ' 添加选择方式
    If RandomCard.Value Then Goal = Goal & SentenceTranslate(RandomCard.Caption) & "-"
    If SelectCard.Value Then Goal = Goal & SentenceTranslate(SelectCard.Caption) & "-"
    
    ' 添加生效方
    If Len(Allied.Tag) = 0 Then
        If Allied.Value And Not Enemy.Value Then Goal = Goal & "Allied-"
        If Enemy.Value And Not Allied.Value Then Goal = Goal & "Enemy-"
    End If
    
    ' 添加生效区域
    If (InStr(Field.Tag, "&") > 0 Or Len(Field.Tag) = 0) Then
        If Field.Value Then Goal = Goal & "Field&"
        If Hand.Value Then Goal = Goal & "Hand&"
        If Deck.Value Then Goal = Goal & "Deck&"
        If ActionCards.Value Then Goal = Goal & SentenceTranslate(ActionCards.Caption) & "&"
        If Len(Goal) > 0 Then Goal = Left(Goal, Len(Goal) - 1) & "-"
    End If
    
    ' 添加特殊目标条件
    If Except.Value Then
        If ThisCard.Value Then Goal = Goal & "Other-"
        If Opposing.Value Then Goal = Goal & "Opposing&Except-"
        If PreviousGoal.Value Then Goal = Goal & "It&Except-"
        If PrePrevious.Value Then Goal = Goal & "That&Except-"
        If PreviousFuse.Value Then Goal = Goal & "FuseCards&Except-"
    End If
    
    ' 添加状态条件
    If Evolve.Value Then Goal = Goal & Translate(Evolve.Caption) & "-"
    If Strike.Value Then Goal = Goal & Translate(Strike.Caption) & "-"
    If Damaged.Value Then Goal = Goal & Translate(Damaged.Caption) & "-"
    If Engage.Value Then Goal = Goal & Translate(Engage.Caption) & "-"
    
    ' 添加职业和能力关键词
    If Class.Value Then Goal = Goal & Translate(Class.Caption) & "-"
    If Keyword.Value Then Goal = Goal & "Keyword&" & Translate(Keyword.Caption) & "-"
    
    ' 添加类型
    If Len(Trait.Value) > 0 Then
        i = Val(IDTranslate("Trait", Trait.Value))
        If i > -1 Then Goal = Goal & "Trait:" & i & "-"
    End If
    
    ' 添加数值条件
    Str = Empty
    If Model1.Value Then Str = Str & SentenceTranslate(Model1.Caption) & "-"
    If Model2.Value Then Str = Str & SentenceTranslate(Model2.Caption) & "-"
    If OrLogic.Value Then Str = Replace(Str, "-", "&")
    If Len(Str) > 0 Then Goal = Goal & Left(Str, Len(Str) - 1) & "-"
    
    ' 添加名称条件
    If Len(StrID) = Len("1000000") Then
        If DifferentName.Value Then
            Goal = Goal & "DifferentName&"
        Else
            Goal = Goal & "SameName&"
        End If
        If Len(StrID2) = Len("1000000") Then
            Goal = Goal & StrID & "&" & StrID2 & "-"
        Else
            Goal = Goal & StrID & "-"
        End If
    ElseIf (ThisCard.Value Or Opposing.Value Or PreviousGoal.Value Or PrePrevious.Value Or PreviousFuse.Value Or SelectCard.Value) And (DifferentName.Value Or SameName.Value) Then
        If DifferentName.Value Then
            Goal = Goal & "DifferentName&"
        Else
            Goal = Goal & "SameName&"
        End If
        If ThisCard.Value Then
            Goal = Goal & "ThisCard-"
        ElseIf Opposing.Value Then
            Goal = Goal & "Opposing-"
        ElseIf PreviousGoal.Value Then
            Goal = Goal & "It-"
        ElseIf PrePrevious.Value Then
            Goal = Goal & "That-"
        ElseIf PreviousFuse.Value Then
            Goal = Goal & "FuseCards-"
        ElseIf SelectCard.Value Then
            Goal = Goal & "Select-"
        End If
    End If
    
    ' 添加位置条件
    If LeftMost.Value Then Goal = Goal & SentenceTranslate(LeftMost.Caption) & "-"
    If RightMost.Value Then Goal = Goal & SentenceTranslate(RightMost.Caption) & "-"
    
    ' 添加卡牌种类
    If Follower.Value Then Goal = Goal & "Follower&"
    If Spell.Value Then Goal = Goal & "Spell&"
    If Amulet.Value Then Goal = Goal & "Amulet&"
    If Leader.Value Then Goal = Goal & "Leader&"
    If Crest.Value Then Goal = Goal & "Crest&"
    If Faith.Value Then Goal = Goal & "Faith&"
    
    ' 优化描述字符串
    Goal = Replace(Goal, "Follower&Spell&Amulet", "Card")
    If Hand.Value = False And Deck.Value = False Then Goal = Replace(Goal, "Follower&Amulet", "Card")
    
    ' 修正格式
    If Right(Goal, 1) = "&" Then Goal = Left(Goal, Len(Goal) - 1)
    If Len(Goal) > 0 And Flag And Right(Goal, 1) <> "-" Then Goal = Goal & "-"
    
    ' 处理特殊目标
    If Flag And ThisCard.Value Then Goal = Goal & "ThisCard"
    If Flag And Opposing.Value Then Goal = Goal & "Opposing"
    If Flag And PreviousGoal.Value Then Goal = Goal & "It"
    If Flag And PrePrevious.Value Then Goal = Goal & "That"
    If Flag And PreviousFuse.Value Then Goal = Goal & "FuseCards"
    
    ' 更新显示
    Effect.Caption = SentenceTranslate(Goal)
End Function
