Attribute VB_Name = "·6·2·结算·GetIf"
' ==================================================
' 函数功能：条件判断处理器
' 参数：InputIf - 输入的条件字符串
'        ThisCard - 当前卡牌的匹配ID
'        InputEffect - 输入的效果字符串
' 返回：String - 判断结果（True/False）和修改后的条件字符串
' 说明：处理游戏中的各种条件判断，包括回合条件、解锁条件、次数限制、特性条件等
' ==================================================
Function GetIf(InputIf As String, ThisCard As String, InputEffect As String) As String
    ' ========== 变量声明 ==========
    Dim i%, j%, k%, m%  ' 循环计数器
    Dim Str As String                               ' 临时字符串
    Dim Arr As Variant                              ' 临时数组
    Dim result As Boolean                           ' 判断结果
    Dim Ifrr() As String                            ' 条件分割数组
    Dim EffectSplit() As String                     ' 效果分割数组
    Dim Sign As String                              ' 比较运算符
    Dim CompareX As String, CompareY As String      ' 比较操作数
    
    ' ========== 条件块合并 ==========
    ' 分割条件字符串，合并被{}包围的多部分条件
    Ifrr = Split(InputIf, "|")
    For i = 1 To UBound(Ifrr)
        ' 检查是否以{开头但未闭合（多部分条件）
        If Left(Ifrr(i), 1) = "{" And InStr(Ifrr(i), "}") = 0 Then
            ' 合并后续部分直到找到闭合}
            For j = i + 1 To UBound(Ifrr)
                Ifrr(i) = Ifrr(i) & "|" & Ifrr(j)   ' 合并条件部分
                Ifrr(j) = "\"                        ' 标记已合并的部分
                If InStr(Ifrr(i), "}") > 0 Then Exit For   ' 找到闭合后退出
            Next j
        End If
    Next i
    
    ' ========== 条件判断主循环 ==========
    result = True   ' 默认结果为True
    
    ' 遍历所有条件部分进行判断
    For k = 1 To UBound(Ifrr)
        ' ========== 回合条件判断 ==========
        ' 检查回合相关条件（如Allied-Turn, Enemy-Turn）
        If Right(Ifrr(k), Len("Turn")) = "Turn" Then
            Str = Down("Turn")       ' 获取当前回合数
            
            ' 检查我方回合条件
            If InStr(Ifrr(k), "Allied-") > 0 And Left(Str, 1) <> Left(ThisCard, 1) Then result = False
            ' 检查敌方回合条件
            If InStr(Ifrr(k), "Enemy-") > 0 And Left(Str, 1) = Left(ThisCard, 1) Then result = False
        End If
        
        ' ========== 解锁条件判断 ==========
        ' 检查进化/超级进化解锁条件
        If Left(Ifrr(k), Len("Unlock")) = "Unlock" Then
            Str = Down("Turn")   ' 获取当前回合
            i = Mid(Str, 2, 10)  ' 提取回合数（去掉F/S前缀）
            
            ' 根据先手和进化类型确定解锁回合
            If Left(Str, 1) = "F" Then
                If InStr(Ifrr(k), "SuperEvolve") > 0 Then j = 7 Else j = 5   ' 先手方解锁回合
            Else
                If InStr(Ifrr(k), "SuperEvolve") > 0 Then j = 6 Else j = 4   ' 后手方解锁回合
            End If
            
            ' 检查是否达到解锁回合
            If i < j Then result = False
        End If
        
        ' ========== 每回合次数限制 ==========
        ' 检查每回合可触发次数限制
        If Left(Ifrr(k), Len("TimesPerTurn&")) = "TimesPerTurn&" Then
            EffectSplit = SplitEffect(DownByMatchID(ThisCard, "Effect"))   ' 获取当前效果
            
            ' 查找已有的次数记录
            For i = 0 To UBound(EffectSplit)
                If Left(EffectSplit(i), Len("TimesPerTurn&")) = "TimesPerTurn&" Then
                    m = Val(Mid(EffectSplit(i), 1 + Len("TimesPerTurn&"), 10))   ' 已触发次数
                    j = Val(Mid(Ifrr(k), 1 + Len("TimesPerTurn&"), 10))          ' 允许的最大次数
                    'If m >= j Then result = False Else EffectSplit(i) = "TimesPerTurn&" & m + 1   ' 更新次数
                    If m >= j Then result = False
                    Exit For
                End If
            Next i
            
            ' 重新构建效果字符串
            Str = Empty
            If UBound(EffectSplit) >= 0 Then Str = Join(EffectSplit, "||") & "||" Else Str = Empty
            ' 如果没有找到次数记录，添加新的记录
            'If i > UBound(EffectSplit) Then Str = Str & "TimesPerTurn&1||"
            
            ' 更新效果记录
            Call UpByMatchID(ThisCard, "Effect", Str)
        End If
        
        ' ========== 连击条件判断 ==========
        ' 检查连击次数条件
        If Left(Ifrr(k), Len("Combo&")) = "Combo&" Then
            i = Down("Combo", ThisCard)                                  ' 当前连击数
            j = Val(Mid(Ifrr(k), 1 + Len("Combo&"), 10))                ' 要求的连击数
            If i < j Then result = False
        End If
        
        ' ========== 协作条件判断 ==========
        ' 检查协作次数条件
        If Left(Ifrr(k), Len("Rally&")) = "Rally&" Then
            i = Down("Rally", ThisCard)                                  ' 当前协作数
            j = Val(Mid(Ifrr(k), 1 + Len("Rally&"), 10))                ' 要求的协作数
            If i < j Then result = False
        End If
        
        ' ========== 土之秘术条件判断 ==========
        ' 检查土之印数量条件
        If Left(Ifrr(k), Len("EarthRite&")) = "EarthRite&" Then
            Arr = Down("Field", ThisCard)   ' 获取场地上所有单位
            
            ' 查找场上的土之印
            For i = 1 To UBound(Arr, 1)
                If InStr(Arr(i, 5), "EarthSigil&") > 0 Then Exit For
            Next i
            
            ' 如果找到土之印，检查数量条件
            If i <= UBound(Arr, 1) Then
                EffectSplit = SplitEffect(Arr(i, 5))   ' 分割土之印效果
                
                ' 提取土之印数量
                For j = 0 To UBound(EffectSplit)
                    If Left(EffectSplit(j), Len("EarthSigil&")) = "EarthSigil&" Then
                        i = Val(Mid(EffectSplit(j), 1 + Len("EarthSigil&"), 10))   ' 当前土之印数量
                        j = Val(Mid(Ifrr(k), 1 + Len("EarthRite&"), 10))           ' 要求的土之印数量
                        If i < j Then result = False
                        Exit For
                    End If
                Next j
            Else
                result = False   ' 没有找到土之印
            End If
        End If
        
        ' ========== 觉醒条件判断 ==========
        ' 检查最大PP是否达到7（觉醒条件）
        If Left(Ifrr(k), Len("Overflow")) = "Overflow" Then
            i = Down("MaxPP", ThisCard)   ' 获取最大PP
            If i < 7 Then result = False
        End If
        
        ' ========== 唤灵条件判断 ==========
        ' 检查墓场数量条件
        If Left(Ifrr(k), Len("Necromancy")) = "Necromancy" Then
            i = Down("Shadows", ThisCard)                                ' 当前墓场数量
            j = Val(Mid(Ifrr(k), 1 + Len("Necromancy&"), 10))           ' 要求的墓场数量
            If i < j Then result = False
        End If
        
        ' ========== 血祭条件判断 ==========
        ' 检查墓场数量条件
        If Left(Ifrr(k), Len("Sacrifice")) = "Sacrifice" Then
            i = Down("BloodPool", ThisCard)                                ' 当前墓场数量
            j = Val(Mid(Ifrr(k), 1 + Len("Sacrifice&"), 10))           ' 要求的墓场数量
            If i < j Then result = False
        End If
        
        ' ========== 查重条件判断 ==========
        ' 检查是否存在重复名称的单位
        If Left(Ifrr(k), Len("NoDuplicates-")) = "NoDuplicates-" Then
            Str = Mid(Ifrr(k), 1 + Len("NoDuplicates-"), 100)           ' 提取检查范围
            
            ' 获取随机选择的结果（用于比较）
            Arr = GetGoal("Random@100-" & Str, ThisCard)
            i = UBound(Arr, 1)                                          ' 随机选择的数量
            
            ' 获取实际选择的结果
            Arr = GetGoal(Str, ThisCard)
            j = UBound(Arr, 1)                                          ' 实际选择的数量
            
            ' 如果实际选择数量大于随机选择数量，说明有重复
            If i < j Then result = False
        End If
        
        ' ========== 奥义条件判断 ==========
        ' 检查最大PP是否达到7（觉醒条件）
        If Left(Ifrr(k), Len("Skybound")) = "Skybound" Then
            i = Val(Mid(Down("Turn", ThisCard), 2, 10)) ' 获取当前回合数
            
            Str = DownByMatchID(ThisCard, "Effect")
            Arr = SplitEffect(Str)
            For j = 0 To UBound(Arr)
                If Left(Arr(j), Len("Skybound&")) = "Skybound&" Then
                    j = Val(Mid(Arr(j), 1 + Len("Skybound&"), 10))
                    Exit For
                End If
            Next j
            
            If i < 10 - j Then result = False
        End If
        
        ' ========== 解放奥义条件判断 ==========
        ' 检查最大PP是否达到7（觉醒条件）
        If Left(Ifrr(k), Len("SuperSkybound")) = "SuperSkybound" Then
            i = Val(Mid(Down("Turn", ThisCard), 2, 10)) ' 获取当前回合数
            
            Str = DownByMatchID(ThisCard, "Effect")
            Arr = SplitEffect(Str)
            For j = 0 To UBound(Arr)
                If Left(Arr(j), Len("Skybound&")) = "Skybound&" Then
                    j = Val(Mid(Arr(j), 1 + Len("Skybound&"), 10))
                    Exit For
                End If
            Next j
            
            If i < 15 - j Then result = False
        End If
        
        ' ========== 数值比较条件判断 ==========
        ' 提取比较运算符
        Sign = Empty
        If InStr(Ifrr(k), "<>") > 0 Then
            Sign = "<>"
        ElseIf InStr(Ifrr(k), "<") > 0 Then
            Sign = "<"
        ElseIf InStr(Ifrr(k), "=") > 0 Then
            Sign = "="
        ElseIf InStr(Ifrr(k), ">") > 0 Then
            Sign = ">"
        End If
        
        ' 如果找到比较运算符，进行数值比较
        If Sign <> "" Then
            i = InStr(Ifrr(k), "}" & Sign)                              ' 查找运算符位置
            Str = Mid(Ifrr(k), i + Len("}" & Sign), 100)                ' 提取比较值Y
            
            ' 解析比较值Y（可能是数值或表达式）
            If Val(Str) > 0 Or Str = "0" Then
                CompareY = Val(Str)                                     ' 直接数值
                j = CompareY
            Else
                CompareY = "{" & GetX(Mid(Str, 2, Len(Str) - 2), ThisCard, InputEffect) & "}"   ' 表达式求值
                j = Val(Mid(CompareY, 2, 10))
            End If
            
            ' 解析比较值X（表达式求值）
            Str = Left(Ifrr(k), i)
            CompareX = "{" & GetX(Mid(Str, 2, Len(Str) - 2), ThisCard, InputEffect) & "}"
            
            ' 进行数值比较
            i = Val(Mid(CompareX, 2, 10))
            If Sign = "<>" And i = j Then result = False                ' 不等于
            If Sign = "<" And i >= j Then result = False                ' 小于
            If Sign = "=" And i <> j Then result = False                ' 等于
            If Sign = ">" And i <= j Then result = False                ' 大于
            
            ' 更新条件字符串（替换为实际比较表达式）
            Ifrr(k) = CompareX & Sign & CompareY
        End If
        
        ' 如果任一条件失败，提前退出循环
        If Not result Then Exit For
    Next k
    
    ' ========== 结果构建 ==========
    Ifrr(0) = result                    ' 将结果放入数组第一个元素
    GetIf = Join(Ifrr, "|")             ' 合并所有条件部分
    GetIf = Replace(GetIf, "|\", "")    ' 清理已合并的标记
End Function
