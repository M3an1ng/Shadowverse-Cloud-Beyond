Attribute VB_Name = "·7·7·对局·MatchGetPP"
' ==================================================
' 过程功能：处理获取PP（能量点）操作
' 说明：后手玩家在特定回合获取额外能量点
' 包含使用条件检查、状态更新和界面刷新
' ==================================================
Sub MatchGetPP()
    Dim Str As String

    ' ========== 优化设置 ==========
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    Application.DisplayAlerts = False
    
    ' ========== 使用权限检查 ==========
    ' 检查按钮状态，无法使用直接退出
    Str = Down("GetPP")
    If Str <> "能量值+1" Then GoTo CleanUp
    
    ' ========== 回合条件检查 ==========
    ' 不是后手回合直接退出
    If Left(Down("Turn"), 1) <> "S" Then GoTo CleanUp
    
    ' ========== 玩家身份检查 ==========
    ' 不是后手方直接退出
    If Left(Down("FirstMove"), 1) <> "S" Then GoTo CleanUp
    
    ' ========== PP获取处理 ==========
    ' 增加1点己方PP
    Call Up("AlliedPP", Val(Down("AlliedPP")) + 1)
    
    ' ========== 界面状态更新 ==========
    ' 更新按钮文本为已使用状态
    Call Up("GetPP", "已使用")
    Sheets("Main").Shapes("PPBtn").TextFrame.Characters.Text = Down("GetPP")
    
    ' ========== 步骤记录 ==========
    ' 记录获取PP的操作步骤
    Call UpSteps("GetPP", Left(Down("Turn"), 1) & "1000")
    
    ' ========== 界面刷新 ==========
    ' 刷新步骤显示区域
    Call UpStepsArea
    
CleanUp:
    ' ========== 恢复应用程序状态 ==========
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
End Sub
