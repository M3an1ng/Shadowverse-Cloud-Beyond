' ========== 生效条件和生效方设置 ==========
' 设置效果生效条件
Private Sub Effect_Click()
    Dim Arr
    Dim response As VbMsgBoxResult
    Arr = Split(CardEffect, "||")
    
    ' 检查是否为改变效果类型
    If Arr(UBound(Arr)) = "Change|" Then
        ' 进化时改为超进化时的确认
        If TimePoint = "Evolve" Then
            response = MsgBox("是否要改为超进化时？", vbYesNo, Translate("SuperEvolve"))
            If response = vbYes Then
                CardEffect = CardEffect & "SuperEvolve||"
                Effect.Caption = SentenceTranslate(CardEffect)
                Exit Sub
            End If
        ' 入场曲/法术发动改为爆能强化的确认
        ElseIf TimePoint = "Fanfare" Or TimePoint = "Play" Then
            response = MsgBox("是否要改为爆能强化？", vbYesNo, Translate("Enhance"))
            If response = vbYes Then
                CardEffect = CardEffect & "Enhance&" & NumInputBox(Translate("Enhance"), "Engage") & "||"
                Effect.Caption = SentenceTranslate(CardEffect)
                Exit Sub
            End If
        End If
    End If
    
    ' 打开条件设置窗体
    EnterCard·2·If.Show
    If IfEffect = "If|" Then IfEffect = Empty
    Effect.Caption = SentenceTranslate(CardEffect & IfEffect)
End Sub

Private Sub UntilTurn_Click()
    If UntilTurn.Caption = Translate("UntilTurnButton") Then
        Dim options() As String
        options = Split(SentenceTranslate("UntilTurnButton|Allied|Enemy|Allied&Enemy"), "|")
        Dim i%
        i = Val(ShowChoice(options, 70, 2))

        If i = 3 Then
            UntilTurn.Caption = SentenceTranslate("Turn-")
        Else
            UntilTurn.Caption = SentenceTranslate(Translate(options(i)) & "-Turn-")
        End If
        
        options = Split(SentenceTranslate("UntilTurnButton|Start|End"), "|")
        i = Val(ShowChoice(options, 50, 2))

        UntilTurn.Caption = UntilTurn.Caption & options(i) & "-" & Translate("Until")
    End If
    
    If Not UntilTurn.Value Then
        UntilTurn.Caption = Translate("UntilTurnButton")
        Whenever.Value = True
    Else
        Whenever.Value = False
    End If
End Sub

Private Sub NamingCount_Click()
    Dim Arr As Variant
    Dim i%, j%
    Dim Str As String
    Str = "NamingCount"
    Arr = Down("Count")
    For i = 1 To UBound(Arr, 1)
        If Arr(i, 1) = "" Then Exit For
        If Mid(Arr(i, 1), 2, 100) <> "Class" Then
            Str = Str & "|" & Mid(Arr(i, 1), 2, 100)
        End If
        i = i + 1
    Next i
    Dim options() As String
    options = Split(SentenceTranslate(Str), "|")
    i = Val(ShowChoice(options, 50, 4))
    If i = 0 Then Exit Sub
    Ability = Ability & "NamingCount|" & Translate(options(i)) & "|" & NumInputBox(options(i), Empty)
    Call UnloadForm
End Sub

' ========== 窗体初始化和退出处理 ==========

' 窗体激活时初始化变量
Private Sub UserForm_Activate()
    IfEffect = Empty
    Ability = Empty
    Allied.Value = True
    Whenever.Value = True
    Effect.Caption = SentenceTranslate(CardEffect)
End Sub

' 统一退出处理函数
Function UnloadForm()
    ' 记录条件效果
    If Len(IfEffect) > 0 Then CardEffect = CardEffect & IfEffect & "||"
    
    ' 记录能力效果
    CardEffect = CardEffect & Ability
    
    ' 处理特殊效果类型
    If Change.Value Then
        CardEffect = CardEffect & "||Change|"
        Effect.Caption = SentenceTranslate(CardEffect)
        ' 重新初始化并打开效果设置
        IfEffect = Empty
        Ability = Empty
        Change.Value = False
        SameTimePoint.Value = False
        Repeat = False
        Call Effect_Click
    ElseIf SameTimePoint.Value Then
        CardEffect = CardEffect & "||"
        Effect.Caption = SentenceTranslate(CardEffect)
        ' 重新初始化
        IfEffect = Empty
        Ability = Empty
        Change.Value = False
        SameTimePoint.Value = False
        Repeat = False
    Else
        ' 正常退出窗体
        Unload EnterCard·3·Ability
    End If
End Function
