Attribute VB_Name = "·9·3·文件·ShowEffect"
' ==================================================
' 过程功能：打开查卡窗口
' 说明：根据当前选中的卡牌位置自动填入卡牌名称
' 优先尝试显示图片，失败则显示效果文本窗体
' ==================================================
Sub ShowEffect()
    ' ========== 优化设置 ==========
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    Application.DisplayAlerts = False
    
    ' ========== 变量声明 ==========
    Dim Flag As Boolean
    Dim Arr As Variant
    Dim CardName As String
    
    ' ========== 卡牌位置检测 ==========
    ' 按优先级顺序检测选中的卡牌位置
    ' 己方战场
    Arr = IsSelected("AlliedField")
    ' 敌方战场
    If IsEmpty(Arr) Then Arr = IsSelected("EnemyField")
    ' 手牌
    If IsEmpty(Arr) Then Arr = IsSelected("Hand")
    ' 纹章区
    If IsEmpty(Arr) Then Arr = IsSelected("Crest")
    ' 已使用区域
    If IsEmpty(Arr) Then Arr = IsSelected("PlayArea")
    ' 已破坏区域
    If IsEmpty(Arr) Then Arr = IsSelected("DestroyArea")
    ' 卡牌集合
    If IsEmpty(Arr) Then Arr = IsSelected("CardSet·Cards")
    ' 原始牌组
    If IsEmpty(Arr) Then Arr = IsSelected("OrigDeck")
    
    ' ========== 卡牌名称获取 ==========
    ' 如果检测到选中的卡牌，获取卡牌名称；否则直接退出
    If Not IsEmpty(Arr) Then
        CardName = CStr(Arr)
    Else
        GoTo CleanUp
    End If
    
    ' ========== 图片显示尝试 ==========
    ' 优先尝试从路径显示卡牌图片
    Flag = ShowImageFromPath(Down("Path"), CardName)
    
    ' ========== 效果文本显示 ==========
    ' 如果图片显示失败，则显示效果文本窗体
    If Not Flag Then
        EffectForm.Caption = CardName
        EffectForm.Show
    End If
    
CleanUp:
    ' ========== 恢复应用程序状态 ==========
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
End Sub
