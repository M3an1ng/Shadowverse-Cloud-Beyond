Attribute VB_Name = "·4·2·辅助·NonEmptyArray"
' ==================================================
' 函数功能：过滤非空数组，移除空行或保留数组结构
' 参数：inputArray - 输入数组
'        IsPreserve - 可选，是否保留原数组结构，默认为False
' 返回：Variant - 处理后的数组
' 说明：根据第一列的值过滤数组，移除空值、零值和空字符串所在的行
' ==================================================
Function NonEmptyArray(inputArray As Variant, Optional IsPreserve As Boolean = False) As Variant
    ' ========== 输入验证 ==========
    ' 检查输入是否为有效数组
    If Not IsArray(inputArray) Then GoTo Unload
    
    ' ========== 变量声明 ==========
    Dim resultArray() As Variant   ' 结果数组
    Dim numRows As Long, numCols As Long   ' 输入数组的行列数
    Dim keepCount As Long          ' 需要保留的行数
    Dim i As Long, j As Long, k As Long   ' 循环计数器
    
    ' ========== 数组维度计算 ==========
    ' 计算输入数组的行数
    numRows = UBound(inputArray, 1) - LBound(inputArray, 1) + 1
    If numRows < 0 Then GoTo Unload   ' 无效行数处理
    
    ' 计算输入数组的列数
    numCols = UBound(inputArray, 2) - LBound(inputArray, 2) + 1
    If numCols < 0 Then GoTo Unload   ' 无效列数处理
    
    ' ========== 结果数组初始化 ==========
    If IsPreserve Then
        ' 保留模式：创建与原数组相同大小的结果数组
        ReDim resultArray(1 To numRows, 1 To numCols)
    Else
        ' 紧凑模式：计算需要保留的非空行数
        keepCount = 0
        
        ' 遍历所有行，统计第一列非空的行数
        For i = LBound(inputArray, 1) To UBound(inputArray, 1)
            ' 检查第一列是否包含有效数据
            If Not IsEmpty(inputArray(i, LBound(inputArray, 2))) And _
               Not IsNull(inputArray(i, LBound(inputArray, 2))) And _
               inputArray(i, LBound(inputArray, 2)) <> "" And _
               inputArray(i, LBound(inputArray, 2)) <> 0 Then
                keepCount = keepCount + 1   ' 增加有效行计数
            End If
        Next i
        
        ' ========== 无有效数据处理 ==========
        If keepCount = 0 Then
            ' 没有有效数据时返回1行空数组
            ReDim resultArray(1 To 1, 1 To numCols)
            NonEmptyArray = resultArray
            Exit Function
        End If
        
        ' 根据有效行数创建紧凑的结果数组
        ReDim resultArray(1 To keepCount, 1 To numCols)
    End If
    
    ' ========== 数据填充 ==========
    k = 1   ' 结果数组的行索引
    
    ' 遍历输入数组的所有行
    For i = LBound(inputArray, 1) To UBound(inputArray, 1)
        ' 检查当前行第一列是否包含有效数据
        If Not IsEmpty(inputArray(i, LBound(inputArray, 2))) And _
           Not IsNull(inputArray(i, LBound(inputArray, 2))) And _
           inputArray(i, LBound(inputArray, 2)) <> "" And _
           inputArray(i, LBound(inputArray, 2)) <> 0 Then
            
            ' 复制整行数据到结果数组
            For j = LBound(inputArray, 2) To UBound(inputArray, 2)
                resultArray(k, j) = inputArray(i, j)
            Next j
            
            k = k + 1   ' 移动到结果数组的下一行
        End If
    Next i
    
    NonEmptyArray = resultArray   ' 返回处理后的数组
    Exit Function

Unload:
    ' ========== 错误处理 ==========
    ' 输入无效时返回原始数组
    NonEmptyArray = inputArray
End Function
