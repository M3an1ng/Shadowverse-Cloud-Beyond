Attribute VB_Name = "·5·1·步骤·UpSteps"
' ==================================================
' 函数功能：更新步骤记录
' 参数：InputAction - 输入的动作描述
'        MatchID - 匹配ID值
' 说明：将当前回合的动作记录添加到Steps区域，用于游戏步骤追踪
' ==================================================
Function UpSteps(InputAction As String, MatchID As String, Optional InputValue As String = "")
    ' ========== 变量声明 ==========
    Dim i As Integer        ' 循环计数器
    Dim Arr As Variant      ' 步骤数据数组
    
    ' ========== 步骤数据获取 ==========
    ' 获取Steps区域的所有步骤记录
    Arr = Down("Steps")
    
    ' ========== 空位查找与记录添加 ==========
    ' 遍历步骤数组查找第一个空位置
    For i = 1 To UBound(Arr)
        If Arr(i, 1) = "" Then
            ' ========== 步骤信息填充 ==========
            Arr(i, 1) = Down("Turn")       ' 记录当前回合数
            Arr(i, 2) = InputAction        ' 记录动作描述
            Arr(i, 3) = MatchID            ' 记录匹配ID
            Arr(i, 4) = InputValue         ' 记录动作值
            Exit For                       ' 找到空位后退出循环
        End If
    Next i
    
    ' ========== 数据更新 ==========
    ' 将更新后的步骤数据写回Steps区域
    Call Up("Steps", Arr)
End Function
