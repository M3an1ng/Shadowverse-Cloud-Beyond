Attribute VB_Name = "·2·3·录卡·CreateFollower"
' ==================================================
' 函数功能：创建随从卡牌
' 参数：CardName - 卡牌名称
' 返回：新创建的随从卡牌ID
' 说明：基于默认模板创建1费青铜随从卡牌
' ==================================================
Function CreateFollower(CardName As String)
    ' ========== 变量声明 ==========
    Dim newID As String     ' 新卡牌ID
    Dim CardSet As String   ' 卡包名称
    Dim Class As String     ' 职业类型
    
    ' ========== 获取默认配置 ==========
    CardSet = DownByID("1000000", "CardSet")   ' 从默认卡牌获取卡包
    Class = DownByID("1000000", "Class")       ' 从默认卡牌获取职业
    
    ' ========== 生成卡牌ID ==========
    newID = GetCardSetID(CardSet) & IDTranslate("Class", Class)   ' 卡包+职业
    newID = newID & 0                                               ' 可构筑标志（0不可构筑）
    newID = newID & IDTranslate("Rarity", "Bronze")                ' 稀有度（青铜）
    newID = GetCardID(newID)                                       ' 获取完整卡牌ID
    
    ' ========== 卡牌属性录入 ==========
    Call UpByID(newID, "CardSet", CardSet)     ' 设置卡包
    Call UpByID(newID, "Class", Class)         ' 设置职业
    Call UpByID(newID, "Rarity", "Bronze")     ' 设置稀有度
    Call UpByID(newID, "Name", CardName)       ' 设置卡牌名称
    Call UpByID(newID, "Type", "Follower")     ' 设置卡牌类型为随从
    Call UpByID(newID, "Cost", "1")            ' 设置费用为1
    Call UpByID(newID, "Attack", "1")          ' 设置攻击力为1
    Call UpByID(newID, "Defense", "1")         ' 设置生命值为1
    
    CreateFollower = newID   ' 返回新卡牌ID
End Function

' ==================================================
' 函数功能：创建纹章卡牌

    Class = DownByID("1000000", "Class")       ' 从默认卡牌获取职业
    Rarity = DownByID("1000000", "Rarity")     ' 从默认卡牌获取稀有度
    
    ' ========== 生成卡牌ID ==========
    newID = GetCardSetID(CardSet) & IDTranslate("Class", Class)   ' 卡包+职业
    newID = newID & 0                                               ' 可构筑标志（0不可构筑）
    newID = newID & IDTranslate("Rarity", Rarity)                  ' 稀有度
    newID = GetCardID(newID)                                       ' 获取完整卡牌ID
    
    ' ========== 卡牌属性录入 ==========
    Call UpByID(newID, "CardSet", CardSet)     ' 设置卡包

    Call UpByID(newID, "Rarity", Rarity)       ' 设置稀有度

    Call UpByID(newID, "Type", "Crest")        ' 设置卡牌类型为纹章
    Call UpByID(newID, "Cost", "0")            ' 设置费用为0
    
    CreateCrest = newID   ' 返回新卡牌ID
End Function
