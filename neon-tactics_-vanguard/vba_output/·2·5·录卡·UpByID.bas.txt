Attribute VB_Name = "·2·5·录卡·UpByID"
' ==================================================
' 函数功能：根据ID更新或创建效果记录
' 参数：idValue - 卡牌ID
'        HeaderName - 属性列标题名称
'        NewValue - 新的属性值
' 返回：Boolean - 操作是否成功
' 说明：支持更新现有记录或创建新记录，自动处理列定位
' ==================================================
Function UpByID(idValue As String, HeaderName As String, NewValue As Variant) As Boolean
    On Error GoTo ErrorHandler
    
    ' ========== 常量定义 ==========
    Const SHEET_NAME As String = "Effect"      ' 效果数据工作表名称
    Const ID_HEADER As String = "ID"           ' ID列标题名称
    
    ' ========== 变量声明 ==========
    Dim ws As Worksheet           ' 工作表对象
    Dim headerRow As Range        ' 标题行范围
    Dim idCol As Range            ' ID列范围
    Dim headerCol As Range        ' 目标属性列范围
    Dim idCell As Range           ' ID查找结果单元格
    Dim targetRow As Long         ' 目标行号
    Dim lastRow As Long           ' 最后数据行号
    Dim newRow As Long            ' 新行号
    
    ' ========== 工作表初始化 ==========
    Set ws = ThisWorkbook.Sheets(SHEET_NAME)   ' 设置Effect工作表
    
    ' ========== 列标题定位 ==========
    Set headerRow = ws.rows(1)   ' 设置标题行（第一行）
    
    ' 查找ID列位置
    Set idCol = headerRow.Find(What:=ID_HEADER, LookAt:=xlWhole, MatchCase:=False)
    ' 查找目标属性列位置
    Set headerCol = headerRow.Find(What:=HeaderName, LookAt:=xlWhole, MatchCase:=False)
    
    ' ========== 列标题验证 ==========
    If idCol Is Nothing Then
        MsgBox "错误：未找到ID列标题", vbExclamation
        UpByID = False
        Exit Function
    End If
    
    If headerCol Is Nothing Then
        MsgBox "错误：未找到表头'" & HeaderName & "'", vbExclamation
        UpByID = False
        Exit Function
    End If
    
    ' ========== 现有记录查找 ==========
    ' 在ID列中精确查找指定ID
    Set idCell = ws.Columns(idCol.Column).Find( _
        What:=idValue, _
        LookIn:=xlValues, _
        LookAt:=xlWhole, _
        MatchCase:=False)
    
    ' ========== 记录更新逻辑 ==========
    If Not idCell Is Nothing Then
        ' 找到现有记录：更新对应属性值
        targetRow = idCell.Row   ' 获取目标行号
        ws.Cells(targetRow, headerCol.Column).Value = NewValue   ' 更新属性值
        UpByID = True
        Exit Function
    End If
    
    ' ========== 新记录创建逻辑 ==========
    ' 获取ID列最后一行位置
    lastRow = ws.Cells(ws.rows.count, idCol.Column).End(xlUp).Row
    If lastRow = 1 Then lastRow = 1   ' 如果只有标题行，从第2行开始
    
    ' 计算新行位置
    newRow = lastRow + 1
    
    ' 写入新记录的ID
    ws.Cells(newRow, idCol.Column).Value = idValue
    
    ' 写入新记录的目标属性值
    ws.Cells(newRow, headerCol.Column).Value = NewValue
    
    ' 可选功能：自动填充其他表头（注释状态）
    ' AutoFillHeaders ws, newRow
    
    UpByID = True   ' 返回操作成功
    Exit Function

ErrorHandler:
    ' ========== 错误处理 ==========
    UpByID = False   ' 返回操作失败
    MsgBox "更新数据时出错: " & Err.Description, vbCritical   ' 显示错误信息
End Function
