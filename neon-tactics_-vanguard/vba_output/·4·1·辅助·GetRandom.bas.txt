Attribute VB_Name = "·4·1·辅助·GetRandom"
' ==================================================
' 函数功能：生成不重复的随机整数序列
' 参数：n - 序列的最大值，序列从1开始到n
' 返回：Variant - 包含1到n的随机不重复序列的数组
' 算法：Fisher-Yates洗牌算法（Knuth洗牌算法）
' ==================================================
Function GetRandom(n As Long) As Variant
    ' ========== 输入参数验证 ==========
    If n < 1 Then
        GetRandom = Array()   ' 无效输入返回空数组
        Exit Function
    End If
    
    ' ========== 变量声明 ==========
    Dim i As Long, j As Long   ' 循环计数器
    Dim temp As Long           ' 临时交换变量
    Dim Arr() As Long          ' 结果数组
    
    ' ========== 初始有序数组创建 ==========
    ' 创建从1到n的有序数组 [1, 2, 3, ..., n]
    ReDim Arr(1 To n)
    For i = 1 To n
        Arr(i) = i   ' 填充顺序数值
    Next i
    
    ' ========== Fisher-Yates洗牌算法 ==========
    Randomize   ' 初始化随机数生成器种子
    
    ' 从后向前遍历数组
    For i = n To 2 Step -1
        ' 生成随机索引j，范围在1到i之间
        j = Int(i * Rnd) + 1
        
        ' 交换当前元素Arr(i)和随机元素Arr(j)
        temp = Arr(i)
        Arr(i) = Arr(j)
        Arr(j) = temp
    Next i
    
    GetRandom = Arr   ' 返回随机序列数组
End Function
