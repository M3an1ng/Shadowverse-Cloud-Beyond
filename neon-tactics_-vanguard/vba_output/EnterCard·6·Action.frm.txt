Private Sub Keyword_Click()
    Dim options() As String
    options = Split(SentenceTranslate( _
                "Keyword&|Fanfare|LastWords|Evolve|SuperEvolve|Strike|Clash|" & _
                "Fuse|Enhance|Accelerate|Crystallize|" & _
                "EarthRite|Necromancy|Sacrifice|Mode|Reanimate"), "|")
    Dim i%
    i = Val(ShowChoice(options, 60, 4))
    
    '标签Goal窗体
    If i <= 6 Then
        EnterCard·4·Goal.Tag = "All||Field|Follower"
        If options(i) = Translate("Fanfare") Then EnterCard·4·Goal.Tag = "All||Field|Follower&Amulet"
        If options(i) = Translate("LastWords") Then EnterCard·4·Goal.Tag = "All||Field|Follower&Amulet"
        If options(i) = Translate("Engage") Then EnterCard·4·Goal.Tag = "All||Field|Amulet"
    ElseIf i <= 10 Then
        EnterCard·4·Goal.Tag = "All||Hand|Follower&Spell&Amulet"
    Else
        EnterCard·4·Goal.Tag = "All||Field&Hand&Deck|Follower&Spell&Amulet&Crest&Faith"
        If options(i) = Translate("Mode") Then EnterCard·4·Goal.Tag = "All||Field&Hand&Deck|Follower&Spell&Amulet"
    End If
    
    Action = "Keyword&" & Translate(options(i))
    
    Unload EnterCard·6·Action
End Sub


Private Sub UserForm_Activate()

    If CardEffect <> "" Then EnterCard·6·Action.Caption = SentenceTranslate(CardEffect)
End Sub
