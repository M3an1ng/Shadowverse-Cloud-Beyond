Attribute VB_Name = "·6·4·结算·GetGoal"
' ==================================================
' 函数功能：获取符合条件的卡牌目标
' 参数：
'   InputGoal - 输入的目标条件字符串
'   ThisCard - 当前卡牌ID
'   InputEffect - 输入的效果字符串（可选）
' 返回：String() - 符合条件的卡牌ID数组
' 说明：根据多种条件筛选卡牌目标，包括区域、类型、状态、属性等
' ==================================================
Function GetGoal(ByVal InputGoal As String, ThisCard As String, Optional InputEffect As String = "", Optional OrigGoal As String = "None") As String()
    ' ========== 变量声明 ==========
    Dim Goalrr() As String           ' 结果数组
    ReDim Goalrr(0 To 0)             ' 初始化为一个空元素
    GetGoal = Goalrr                 ' 默认返回空数组
    'PreviousGoal = Empty            ' 清空上一次目标记录
    Dim PreviousFuse As String       ' 上一次融合目标记录
    Dim PrePrevious As String       ' 上一次融合目标记录
    
    Dim i%, j%, k%                   ' 循环计数器
    Dim Str As String                ' 临时字符串
    Dim Str2 As String               ' 临时字符串2
    Dim Flag As Boolean              ' 标志位
    Dim Effect As String             ' 效果字符串
    Dim Irr() As String              ' 临时字符串数组
    
    ' ========== 已处理目标直接返回 ==========
    ' 检查是否已经是处理好的Goal格式
    i = InStr(InputGoal, "Goal&")
    If i > 0 Then
        PreviousGoal = Mid(InputGoal, i + Len("Goal&"), 1000)
        Goalrr = Split("&" & PreviousGoal, "&")
        GetGoal = Goalrr
        Exit Function
    End If
    
    ' ========== 处理过程变量 ==========
    Dim EraseFlag As Boolean         ' 删除标志
    Dim AllGoal As Variant           ' 所有目标数据
    Dim TempGoal As Variant          ' 临时目标数据

    ' ========== 设置交战对手 ==========
    Dim Opposing As String
    If Left(ThisCard, 1) = Left(StrikeMatchID, 1) Then
        If ClashMatchID = "" Then Opposing = "None" Else Opposing = ClashMatchID
    Else
        Opposing = StrikeMatchID
    End If

    ' ========== 设置上一个限定目标 ==========
    i = InStr(InputEffect, InputGoal)
    If i = 0 Then i = InStr(InputEffect, OrigGoal)
    If i = 0 Then i = Len(InputEffect) + 1
    Str = Left(InputEffect, i - 1)
    For j = 1 To i - 5
        If Mid(InputEffect, i - 5 - j + 1, 5) = "Goal&" Then
            Str = Mid(Str, i - 5 - j + 1 + 5, 100)
            Exit For
        End If
    Next j
    If j <= i - 5 Then
        i = InStr(Str, "|")
        If i > 0 Then PreviousGoal = Left(Str, i - 1)
    Else
        PreviousGoal = Empty
    End If

    ' ========== 设置上上个限定目标 ==========
    i = InStr(InputEffect, InputGoal)
    If i = 0 Then i = InStr(InputEffect, OrigGoal)
    If i > 0 Then
        Str = Left(InputEffect, i - 1)
        k = 0
        For j = 1 To i - 5
            If Mid(InputEffect, i - 5 - j + 1, 5) = "Goal&" Then
                Str = Mid(Str, i - 5 - j + 1 + 5, 100)
                k = k + 1
            End If
            If k = 2 Then Exit For
        Next j
        If j <= i - 5 Then
            i = InStr(Str, "|")
            If i > 0 Then PrePrevious = Left(Str, i - 1)
        Else
            PrePrevious = Empty
        End If
    End If
    
    ' ========== 设置上一次融合目标 ==========
    If InStr(InputGoal, "FuseCards") > 0 Then
        TempGoal = NonEmptyArray(Down("Steps"))
        PreviousFuse = Empty
        For i = 1 To UBound(TempGoal, 1)
            j = UBound(TempGoal, 1) - i + 1
            If TempGoal(j, 2) = "Keyword&Fuse" Then Exit For
        Next i
        For i = j To UBound(TempGoal, 1)
            If TempGoal(i, 2) = "Fused" Then PreviousFuse = PreviousFuse & TempGoal(i, 3) & "&"
        Next i
        If PreviousFuse <> "" Then PreviousFuse = Left(PreviousFuse, Len(PreviousFuse) - 1)
    End If
    
    ' ========== 处理XYZ变量 ==========
    ' 将条件中的X、Y、Z替换为实际数值
    For j = 1 To Len(InputGoal)
        Str = Mid(InputGoal, j, 1)
        If InStr("XYZ", Str) > 0 Then
            i = InStr(InputEffect, Str & "=")
            Str = Mid(InputEffect, i + 2, 100)
            i = InStr(Str, "||")
            If i > 0 Then Str = Left(Str, i - 1)
            
            k = Val(GetX(Str, ThisCard, "Goal&" & PreviousGoal & "||" & Str))
            InputGoal = Replace(InputGoal, Mid(InputGoal, j, 1), k)
        End If
    Next j
    
    Dim Arr() As String              ' 输入条件分割数组
    Arr = Split(InputGoal, "-")
    AllGoal = Down("Data")           ' 获取所有卡牌数据

'========================================================================================================================================================================================================
    ' ========== 具体目标筛选 ==========
    ' 处理特殊目标类型：ThisCard、Opposing、It、FuseCards
    If Arr(UBound(Arr)) = "ThisCard" Then
        ' 遍历找到当前卡牌
        For i = 1 To UBound(AllGoal)
            If AllGoal(i, 1) = ThisCard Then Exit For
        Next i
        ' 如果没找到直接退出函数
        If i > UBound(AllGoal) Then Exit Function
        ' 提取当前卡牌数据
        ReDim TempGoal(1 To 1, 1 To UBound(AllGoal, 2))
        For j = 1 To UBound(TempGoal, 2)
            TempGoal(1, j) = AllGoal(i, j)
        Next j
        AllGoal = TempGoal
        
    ElseIf Arr(UBound(Arr)) = "Opposing" Then
        ' 遍历找到交战对手
        For i = 1 To UBound(AllGoal)
            If AllGoal(i, 1) = Opposing Then Exit For
        Next i
        ' 如果没找到直接退出函数
        If i > UBound(AllGoal) Then Exit Function
        ' 提取对手数据
        ReDim TempGoal(1 To 1, 1 To UBound(AllGoal, 2))
        For j = 1 To UBound(TempGoal, 2)
            TempGoal(1, j) = AllGoal(i, j)
        Next j
        AllGoal = TempGoal
        
    ElseIf Arr(UBound(Arr)) = "It" Then
        ' 如果没找到上一个目标直接退出函数
        If PreviousGoal = "" Then Exit Function
        
        Goalrr = Split(PreviousGoal, "&")
        ReDim AllGoal(1 To UBound(Goalrr) + 1, 1 To UBound(AllGoal, 2))
        
        ' 遍历找It目标
        For i = 0 To UBound(Goalrr)
            TempGoal = Down("Data")
            For j = 1 To UBound(TempGoal, 1)
                If TempGoal(j, 1) = Goalrr(i) Then Exit For
            Next j
            If j <= UBound(TempGoal, 1) Then
                For k = 1 To UBound(TempGoal, 2)
                    AllGoal(i + 1, k) = TempGoal(j, k)
                Next k
            End If
        Next i
        
    ElseIf Arr(UBound(Arr)) = "That" Then
        ' 如果没找到上一个目标直接退出函数
        If PrePrevious = "" Then Exit Function
        
        Goalrr = Split(PrePrevious, "&")
        ReDim AllGoal(1 To UBound(Goalrr) + 1, 1 To UBound(AllGoal, 2))
        
        ' 遍历找It目标
        For i = 0 To UBound(Goalrr)
            TempGoal = Down("Data")
            For j = 1 To UBound(TempGoal, 1)
                If TempGoal(j, 1) = Goalrr(i) Then Exit For
            Next j
            If j <= UBound(TempGoal, 1) Then
                For k = 1 To UBound(TempGoal, 2)
                    AllGoal(i + 1, k) = TempGoal(j, k)
                Next k
            End If
        Next i
        
    ElseIf Arr(UBound(Arr)) = "FuseCards" Then
        ' 如果没找到融合目标直接退出函数
        If PreviousFuse = "" Then Exit Function
        
        Goalrr = Split(PreviousFuse, "&")
        ReDim AllGoal(1 To UBound(Goalrr) + 1, 1 To UBound(AllGoal, 2))
        
        ' 遍历找融合目标
        For i = 0 To UBound(Goalrr)
            TempGoal = Down("Data")
            For j = 1 To UBound(TempGoal, 1)
                If TempGoal(j, 1) = Goalrr(i) Then Exit For
            Next j
            If j <= UBound(TempGoal, 1) Then
                For k = 1 To UBound(TempGoal, 2)
                    AllGoal(i + 1, k) = TempGoal(j, k)
                Next k
            End If
        Next i
    End If

'========================================================================================================================================================================================================
    ' ========== 生效方筛选 ==========
    ' 根据Allied（友方）或Enemy（敌方）条件筛选
    If InStr(InputGoal, "Allied-") > 0 Then
        For i = 1 To UBound(AllGoal, 1)
            If Left(AllGoal(i, 1), 1) <> Left(ThisCard, 1) Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    ElseIf InStr(InputGoal, "Enemy-") > 0 Then
        For i = 1 To UBound(AllGoal, 1)
            If Left(AllGoal(i, 1), 1) = Left(ThisCard, 1) Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    End If

    ' ========== 生效区域筛选 ==========
    ' 根据区域条件筛选：场地、手牌、牌组等
    If InStr(InputGoal, "AnyArea-") > 0 Then
    
    ElseIf InStr(InputGoal, "Field") > 0 Or InStr(InputGoal, "Hand") > 0 Or InStr(InputGoal, "Deck") > 0 Or _
        InStr(InputGoal, "ThisTurn") > 0 Or InStr(InputGoal, "ThisMatch") > 0 Then
        
        '处理本回合
        Str = ""
        j = InStr(InputGoal, "ThisTurn")
        If j > 0 Then
            Str = Mid(InputGoal, j + Len("ThisTurn&"), 100)
            j = InStr(Str, "-")
            If j > 0 Then Str = Left(Str, j - 1)
            j = InStr(Str, "&")
            If j > 0 Then Str = Left(Str, j - 1)
        End If
        '处理本次对战
        j = InStr(InputGoal, "ThisMatch")
        If j > 0 Then
            Str = Mid(InputGoal, j + Len("ThisMatch&"), 100)
            j = InStr(Str, "-")
            If j > 0 Then Str = Left(Str, j - 1)
            j = InStr(Str, "&")
            If j > 0 Then Str = Left(Str, j - 1)
        End If
        Str2 = Down("Turn")
        
        For i = 1 To UBound(AllGoal)
            EraseFlag = True
            If InStr(InputGoal, "ThisTurn") > 0 Then
                If SearchSteps(CStr(AllGoal(i, 1)), Str, Str2) Then EraseFlag = False
            ElseIf InStr(InputGoal, "ThisMatch") > 0 Then
                If SearchSteps(CStr(AllGoal(i, 1)), Str) Then EraseFlag = False
            ElseIf InStr(InputGoal, AllGoal(i, 4)) > 0 Then
                EraseFlag = False
            ElseIf InStr(InputGoal, "Field") > 0 And (AllGoal(i, 4) = "Crest" Or AllGoal(i, 4) = "Faith" Or AllGoal(i, 4) = "Leader") Then
                EraseFlag = False
            End If
            
            If EraseFlag Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    End If

    ' ========== 生效种类筛选 ==========
    ' 处理卡牌类型条件：从者、法术、护符等
    If InStr(InputGoal, "Hand-") > 0 Or InStr(InputGoal, "Deck-") > 0 Or InStr(InputGoal, "AnyArea-") > 0 Then
        InputGoal = Replace(InputGoal, "-Card", "-Follower&Spell&Amulet")
        If Left(InputGoal, Len("Card")) = "Card" Then InputGoal = "Follower&Spell&Amulet" & Mid(InputGoal, 1 + Len("Card"), 100)
    Else
        InputGoal = Replace(InputGoal, "-Card", "-Follower&Amulet")
        If Left(InputGoal, Len("Card")) = "Card" Then InputGoal = "Follower&Amulet" & Mid(InputGoal, 1 + Len("Card"), 100)
    End If
    
    ' 根据具体类型筛选
    If InStr(InputGoal, "Follower") > 0 Or InStr(InputGoal, "Spell") > 0 Or InStr(InputGoal, "Amulet") > 0 Or _
        InStr(InputGoal, "Leader") > 0 Or InStr(InputGoal, "Crest") > 0 Or InStr(InputGoal, "Faith") > 0 Then
        
        ' 遍历检查类型匹配
        For i = 1 To UBound(AllGoal)
            Str = DownByID(CStr(AllGoal(i, 2)), "Type")
            If InStr(InputGoal, Str) = 0 Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    End If

'========================================================================================================================================================================================================
    ' ========== 排除特定目标 ==========
    ' 处理排除条件：Other、Opposing、It等
    Str = ""
    If InStr(InputGoal, "Other-") > 0 Then
        Str = ThisCard
    ElseIf InStr(InputGoal, "Opposing&Except-") > 0 Then
        If Opposing <> "" Then Str = Opposing
    ElseIf InStr(InputGoal, "It&Except-") > 0 Then
        Str = PreviousGoal
    ElseIf InStr(InputGoal, "That&Except-") > 0 Then
        Str = PrePrevious
    ElseIf InStr(InputGoal, "FuseCards&Except-") > 0 Then
        Str = PreviousFuse
    End If
    If Str <> "" Then
        For i = 1 To UBound(AllGoal)
            If InStr(Str, AllGoal(i, 1)) > 0 Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    End If

'========================================================================================================================================================================================================
    ' ========== 进化状态筛选 ==========
    ' 根据进化状态筛选
    If InStr(InputGoal, "UnEvolved-") > 0 Then
        For i = 1 To UBound(AllGoal)
            ' 检查是否已进化
            If InStr(AllGoal(i, 5), "Evolved||") > 0 Or InStr(AllGoal(i, 5), "SuperEvolved||") > 0 Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    ElseIf InStr(InputGoal, "SuperEvolved-") > 0 Then
        For i = 1 To UBound(AllGoal)
            ' 检查是否未超进化
            If InStr(AllGoal(i, 5), "SuperEvolved||") = 0 Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    ElseIf InStr(InputGoal, "Evolved-") > 0 Then
        For i = 1 To UBound(AllGoal)
            ' 检查是否未进化
            If InStr(AllGoal(i, 5), "Evolved||") = 0 And InStr(AllGoal(i, 5), "SuperEvolved||") = 0 Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    End If

    ' ========== 攻击状态筛选 ==========
    ' 根据是否已攻击筛选
    If InStr(InputGoal, "UnStriked-") > 0 Then
        For i = 1 To UBound(AllGoal)
            ' 检查是否已攻击
            If InStr(AllGoal(i, 5), "Striked&") > 0 Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    ElseIf InStr(InputGoal, "Striked-") > 0 Then
        For i = 1 To UBound(AllGoal)
            ' 检查是否未攻击
            If InStr(AllGoal(i, 5), "Striked&") = 0 Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    End If

    ' ========== 受伤状态筛选 ==========
    ' 根据受伤状态筛选
    If InStr(InputGoal, "UnDamaged-") > 0 Then
        For i = 1 To UBound(AllGoal)
            ' 检查受伤数值
            j = InStr(AllGoal(i, 5), "Damaged&")
            If j > 0 Then j = Val(Mid(AllGoal(i, 5), j + Len("Damaged&"), 10))
            If j > 0 Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    ElseIf InStr(InputGoal, "Damaged-") > 0 Then
        For i = 1 To UBound(AllGoal)
            ' 检查未受伤
            j = InStr(AllGoal(i, 5), "Damaged&")
            If j > 0 Then j = Val(Mid(AllGoal(i, 5), j + Len("Damaged&"), 10))
            If j = 0 Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    End If
    
    ' ========== 启动状态筛选 ==========
    ' 根据是否已启动筛选
    If InStr(InputGoal, "UnEngaged-") > 0 Then
        For i = 1 To UBound(AllGoal)
            ' 检查是否已启动
            If InStr(AllGoal(i, 5), "Engaged||") > 0 Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    ElseIf InStr(InputGoal, "Engaged-") > 0 Then
        For i = 1 To UBound(AllGoal)
            ' 检查是否未启动
            If InStr(AllGoal(i, 5), "Engaged||") = 0 Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    End If

'========================================================================================================================================================================================================
    ' ========== 职业筛选 ==========
    ' 根据卡牌职业筛选
    Str = ""
    If InStr(InputGoal, "Forestcraft-") > 0 Then
        Str = "Forestcraft"
    ElseIf InStr(InputGoal, "Swordcraft-") > 0 Then
        Str = "Swordcraft"
    ElseIf InStr(InputGoal, "Runecraft-") > 0 Then
        Str = "Runecraft"
    ElseIf InStr(InputGoal, "Dragoncraft-") > 0 Then
        Str = "Dragoncraft"
    ElseIf InStr(InputGoal, "Abysscraft-") > 0 Then
        Str = "Abysscraft"
    ElseIf InStr(InputGoal, "Havencraft-") > 0 Then
        Str = "Havencraft"
    ElseIf InStr(InputGoal, "Portalcraft-") > 0 Then
        Str = "Portalcraft"
    ElseIf InStr(InputGoal, "Neutral-") > 0 Then
        Str = "Neutral"
    End If
    If Str <> "" Then
        For i = 1 To UBound(AllGoal)
            If DownByID(CStr(AllGoal(i, 2)), "Class") <> Str Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    End If

'========================================================================================================================================================================================================
    ' ========== 能力关键词筛选 ==========
    ' 根据能力关键词筛选
    i = InStr(InputGoal, "Keyword&")
    If i > 0 Then
        Str = Mid(InputGoal, i + Len("Keyword&"), 100)
        i = InStr(Str, "-")
        Str = Left(Str, i - 1)
        
        ' 处理特定关键词
        If InStr("Fanfare|LastWords|Evolve|SuperEvolve|Strike|Clash|Engage|Spellboost|Fuse|Enhance|Accelerate|Crystallize|Rush|Storm|Ambush|Bane|Ward|Drain|Intimidate|Aura|Barrier", Str) > 0 Then
            For i = 1 To UBound(AllGoal)
                EraseFlag = True
                ' 检查基础能力
                For j = 1 To 9
                    Effect = DownByID(CStr(AllGoal(i, 2)), "Effect" & j)
                    If Effect = "" Then Exit For
                    If Left(Effect, Len(Str)) = Str Then EraseFlag = False
                Next j
                
                ' 检查获得的能力
                If InStr(AllGoal(i, 5), "{" & Str) > 0 Then EraseFlag = False
                
                ' 检查已失去的能力
                If InStr(AllGoal(i, 5), "Remove|" & Str) > 0 Then EraseFlag = True
                
                If EraseFlag Then AllGoal(i, 1) = Empty
            Next i
        ElseIf Str = "Countdown" Then
            ' 处理倒数关键词
            For i = 1 To UBound(AllGoal)
                If DownByID(CStr(AllGoal(i, 2)), "Countdown") = "" Then AllGoal(i, 1) = Empty
            Next i
        Else
            ' 处理其他关键词
            For i = 1 To UBound(AllGoal)
                EraseFlag = True
                For j = 1 To 9
                    Effect = DownByID(CStr(AllGoal(i, 2)), "Effect" & j)
                    If Effect = "" Then Exit For
                    If InStr(Effect, Str) > 0 Then EraseFlag = False
                Next j

                If EraseFlag Then AllGoal(i, 1) = Empty
            Next i
        End If
        AllGoal = NonEmptyArray(AllGoal)
    End If

'========================================================================================================================================================================================================
    ' ========== 类型筛选 ==========
    ' 根据卡牌类型特征筛选
    i = InStr(InputGoal, "Trait:")
    If i > 0 Then
        Str = Val(Mid(InputGoal, i + Len("Trait:"), 100))
        For i = 1 To UBound(AllGoal)
            EraseFlag = True
            If DownByID(CStr(AllGoal(i, 2)), "Trait") = IDTranslate("Trait", Str) Then EraseFlag = False
            ' 检查获得的效果类型
            If InStr(AllGoal(i, 5), "Trait:" & Str) Then EraseFlag = False
            If EraseFlag Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    End If

'========================================================================================================================================================================================================
    ' ========== 名称筛选 ==========
    ' 根据同名或不同名条件筛选
    i = InStr(InputGoal, "SameName&")
    If i = 0 Then i = InStr(InputGoal, "DifferentName&")
    If i > 0 Then
        Str = Mid(InputGoal, i, 100)
        i = InStr(Str, "&")
        Effect = Left(Str, i - 1)
        Str = Mid(Str, i + 1, 100)
        i = InStr(Str, "-")
        Str = Left(Str, i - 1)
        
        ' 提取比较对象的ID
        If Str = "ThisCard" Then
            Str = DownByMatchID(ThisCard, "ID")
        ElseIf Str = "Opposing" Then
            Str = DownByMatchID(Opposing, "ID")
        ElseIf Str = "It" Then
            If PreviousGoal = "" Then
                Str = "1000000"
            Else
                Goalrr = Split(PreviousGoal, "&")
                ' 遍历获取It目标的ID
                For j = 0 To UBound(Goalrr)
                    If Goalrr(j) <> "" Then Goalrr(j) = DownByMatchID(Goalrr(j), "ID")
                Next j
                Str = Join(Goalrr, "&")
            End If
        ElseIf Str = "That" Then
            If PrePrevious = "" Then
                Str = "1000000"
            Else
                Goalrr = Split(PrePrevious, "&")
                ' 遍历获取It目标的ID
                For j = 0 To UBound(Goalrr)
                    If Goalrr(j) <> "" Then Goalrr(j) = DownByMatchID(Goalrr(j), "ID")
                Next j
                Str = Join(Goalrr, "&")
            End If
        ElseIf Str = "FuseCards" Then
            If PreviousFuse = "" Then
                Str = "1000000"
            Else
                Goalrr = Split(PreviousFuse, "&")
                ' 遍历获取融合目标的ID
                For j = 0 To UBound(Goalrr)
                    If Goalrr(j) <> "" Then Goalrr(j) = DownByMatchID(Goalrr(j), "ID")
                Next j
                Str = Join(Goalrr, "&")
            End If
        ElseIf Str = "Select" Then
            Str = ""
        Else
            Str = Str
        End If
        
        ' 根据同名或不同名进行筛选
        If Str <> "" Then
            If Effect = "SameName" Then
                For i = 1 To UBound(AllGoal)
                    If InStr(Str, AllGoal(i, 2)) = 0 Then AllGoal(i, 1) = Empty
                Next i
            ElseIf Effect = "DifferentName" Then
                For i = 1 To UBound(AllGoal)
                    If InStr(Str, AllGoal(i, 2)) > 0 Then AllGoal(i, 1) = Empty
                Next i
            End If
            AllGoal = NonEmptyArray(AllGoal)
        End If
    End If

'========================================================================================================================================================================================================
    ' ========== 属性比较筛选 ==========
    ' 根据费用、攻击力、生命值、倒计数等属性筛选
    Str = Empty
    Str2 = Empty
    For i = 0 To UBound(Arr)
        If Left(Arr(i), Len("Orig")) = "Orig" Or _
        Left(Arr(i), Len("Cost")) = "Cost" Or _
        Left(Arr(i), Len("Attack")) = "Attack" Or _
        Left(Arr(i), Len("Defense")) = "Defense" Or _
        Left(Arr(i), Len("Countdown")) = "Countdown" Then
            If Str = "" Then Str = Arr(i) Else Str2 = Arr(i)
        End If
    Next i
    
    ' 处理属性比较条件
    i = InStr(Str, "&")
    If i > 0 Then
        Str2 = Mid(Str, 1 + i, 100)
        Str = Left(Str, i - 1)
        AllGoal = CompareModel(Str, AllGoal, Str2)
    ElseIf InStr(Str, "Highest") = 0 And InStr(Str, "Lowest") = 0 And Str <> "" Then
        AllGoal = CompareModel(Str, AllGoal)
        AllGoal = CompareModel(Str2, AllGoal)
    ElseIf Str <> "" Then
        AllGoal = CompareModel(Str2, AllGoal)
        AllGoal = CompareModel(Str, AllGoal)
    End If

'========================================================================================================================================================================================================
    ' ========== 位置筛选 ==========
    ' 根据在场地中的位置筛选
    i = InStr(InputGoal, "LeftMost&")
    j = 0
    If i > 0 Then j = Val(Mid(InputGoal, i + Len("LeftMost&"), 10))
    If j > 0 Then
        For i = j + 1 To UBound(AllGoal, 1)
            AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    End If
    
    ' 最后位置筛选
    i = InStr(InputGoal, "RightMost&")
    j = 0
    If i > 0 Then j = Val(Mid(InputGoal, i + Len("RightMost&"), 10))
    If j > 0 Then
        For i = 1 To UBound(AllGoal, 1) - j
            AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
    End If

'========================================================================================================================================================================================================
    ' ========== 防报错处理 ==========
    ' 如果没有找到任何目标，返回空结果
    ReDim Goalrr(0 To 0)
    If IsEmpty(AllGoal) Or AllGoal(1, 1) = "" Then
        PreviousGoal = ""
        GetGoal = Goalrr
        Exit Function
    End If

    ' ========== 选择操作前置处理 ==========
    ' 为玩家选择操作做准备
    If InStr(InputGoal, "Select&") > 0 Then
        ' 去掉灵气和土之印等不可选目标
        Str = Down("Turn")
        For i = 1 To UBound(AllGoal, 1)
            If Left(Str, 1) = Left(AllGoal(i, 1), 1) Then
                
            ElseIf AllGoal(i, 4) = "Field" Or AllGoal(i, 4) = "Leader" Then
                If (InStr(AllGoal(i, 5), "Aura") > 0 Or InStr(AllGoal(i, 5), "Ambush") > 0 Or InStr(AllGoal(i, 5), "EarthSigil&") > 0) Then
                    Arr = SplitEffect(CStr(AllGoal(i, 5)))
                    Flag = False
                    For j = 0 To UBound(Arr)
                        If Arr(j) = "Aura" Or Arr(j) = "Ambush" Or InStr(Arr(j), "{Aura}") > 0 Or InStr(AllGoal(i, 5), "{Ambush}") > 0 Or InStr(Arr(j), "EarthSigil&") > 0 Then
                            Flag = True
                        ElseIf Arr(j) = "Remove|Aura" Or Arr(j) = "Remove|Ambush" Then
                            Flag = False
                            Exit For
                        End If
                    Next j
                    If Flag Then AllGoal(i, 1) = Empty
                ElseIf InStr(AllGoal(i, 5), "CantSelectOther") > 0 Then
                    ' 处理不能选择其他的效果
                    For j = 1 To UBound(AllGoal, 1)
                        If InStr(AllGoal(j, 5), "CantSelectOther") = 0 Then AllGoal(j, 1) = Empty
                    Next j
                    Exit For
                End If
            End If
        Next i
        AllGoal = NonEmptyArray(AllGoal)

        ' 调换自己与对手的顺序（对手方优先）
        ReDim TempGoal(1 To UBound(AllGoal, 1), 1 To UBound(AllGoal, 2))
        Str = Down("FirstMove")
        If Str = "FirstMove" Then Str = "S" Else Str = "F"
        k = 1
        For i = 1 To UBound(AllGoal, 1)
            If Left(AllGoal(i, 1), 1) = Str Then
                For j = 1 To UBound(AllGoal, 2)
                    TempGoal(k, j) = AllGoal(i, j)
                Next j
                k = k + 1
            End If
        Next i
        For i = 1 To UBound(AllGoal, 1)
            If Left(AllGoal(i, 1), 1) <> Str Then
                For j = 1 To UBound(AllGoal, 2)
                    TempGoal(k, j) = AllGoal(i, j)
                Next j
                k = k + 1
            End If
        Next i
        AllGoal = TempGoal
        
        ' 敌我方边界插入空档（界面美化）
        ReDim TempGoal(1 To 10 + UBound(AllGoal, 1), 1 To UBound(AllGoal, 2))
        For j = 1 To UBound(AllGoal, 2)
            TempGoal(1, j) = AllGoal(1, j)
        Next j
        k = 2
        For i = 2 To UBound(AllGoal, 1)
            ' 在敌我方边界插入空档
            If Left(AllGoal(i - 1, 1), 1) <> Left(AllGoal(i, 1), 1) Then
                For j = 1 To 6
                    If (k - 1) Mod 6 = 0 Then
                        Exit For
                    Else
                        TempGoal(k, 1) = 9999
                        k = k + 1
                    End If
                Next j
            End If
            For j = 1 To UBound(AllGoal, 2)
                TempGoal(k, j) = AllGoal(i, j)
            Next j
            k = k + 1
        Next i
        AllGoal = NonEmptyArray(TempGoal)
    End If

'========================================================================================================================================================================================================
    ' ========== 最终目标选择处理 ==========
    ' 处理选择、随机或全部返回
    If InStr(InputGoal, "Select&") > 0 Then
        ' 获取需选择数量
        i = InStr(InputGoal, "Select&")
        Str = Mid(InputGoal, i + Len("Select&"), 10)
        j = Val(Str)
        
        ' 需选择数大于选项数
        If AllGoal(1, 1) = "" Then
            PreviousGoal = "-1"
            Exit Function
        End If
        If j > UBound(AllGoal, 1) Then
            If DownByID(DownByMatchID(ThisCard, "ID"), "Type") = "Spell" Then
                PreviousGoal = ""
                Exit Function
            Else
                j = UBound(AllGoal, 1)
            End If
        End If
        
        ' 生成选项供玩家选择
        Dim options() As String
        Str = Translate("Select") & "&" & j
        For i = 1 To UBound(AllGoal, 1)
            Str = Str & "|" & DownByID(CStr(AllGoal(i, 2)), "Name")
        Next i
        options = Split(Str, "|")
        If j = 0 Then j = -1
        Str = ShowChoice(options, 150, 6, j)
        If Str = "" Then
            PreviousGoal = ""
            Exit Function
        End If
        Irr = Split(Str, "&")
        
        ' 直接退出处理
        For i = 0 To UBound(Irr)
            If options(Irr(i)) = "" Then Exit Function
        Next i
        
        ' 处理名称相关选择
        If InStr(InputGoal, "DifferentName&Select") > 0 Then
            Str = Empty
            For i = 0 To UBound(Irr)
                Str = Str & AllGoal(Irr(i), 2) & "&"
            Next i
            For i = 1 To UBound(AllGoal)
                If InStr(Str, AllGoal(i, 2)) > 0 Then AllGoal(i, 1) = Empty
            Next i
        ElseIf InStr(InputGoal, "SameName&Select") > 0 Then
            Str = Empty
            For i = 0 To UBound(Irr)
                Str = Str & AllGoal(Irr(i), 2) & "&"
            Next i
            For i = 1 To UBound(AllGoal)
                If InStr(Str, AllGoal(i, 2)) = 0 Then AllGoal(i, 1) = Empty
            Next i
        Else
            ' 普通选择处理
            For i = 1 To UBound(AllGoal, 1)
                If InStr(Str, i) = 0 Then AllGoal(i, 1) = Empty
            Next i
        End If
        
        ' 清理空元素
        For i = 1 To UBound(AllGoal, 1)
            If AllGoal(i, 1) = "9999" Then AllGoal(i, 1) = Empty
        Next i
        AllGoal = NonEmptyArray(AllGoal)
        
        ' 构建结果数组
        ReDim Goalrr(1 To UBound(AllGoal, 1))
        For i = 1 To UBound(Goalrr)
            Goalrr(i) = AllGoal(i, 1)
        Next i
        
    ElseIf InStr(InputGoal, "Random") > 0 And QueueAction = "" Then
        ' 随机选择处理
        i = InStr(InputGoal, "Random")
        Str = Mid(InputGoal, i + Len("Random&"), 10)
        j = Val(Str)
        If j > UBound(AllGoal, 1) Then j = UBound(AllGoal, 1)
        
        ' 产生随机数组
        Dim Randomrr As Variant
        Randomrr = GetRandom(UBound(AllGoal, 1))
        
        ' 随机选择处理
        If j >= 1 Then
            ' 处理不同名随机
            If InStr(InputGoal, "Random@") > 0 Then
                ReDim TempGoal(1 To j, 1 To 2)
                TempGoal(1, 1) = AllGoal(Randomrr(1), 1)
                TempGoal(1, 2) = AllGoal(Randomrr(1), 2)
                k = 2
                For i = 2 To UBound(Randomrr)
                    If k > UBound(TempGoal, 1) Then Exit For
                    For j = 1 To k - 1
                        If TempGoal(j, 2) = AllGoal(Randomrr(i), 2) Then Exit For
                    Next j
                    If j > k - 1 Then
                        TempGoal(k, 1) = AllGoal(Randomrr(i), 1)
                        TempGoal(k, 2) = AllGoal(Randomrr(i), 2)
                        k = k + 1
                    End If
                Next i
                ReDim Goalrr(1 To k - 1)
                For i = 1 To UBound(Goalrr)
                    Goalrr(i) = TempGoal(i, 1)
                Next i
            Else
                ' 普通随机选择
                ReDim Goalrr(1 To j)
                For i = 1 To UBound(Goalrr)
                    Goalrr(i) = AllGoal(Randomrr(i), 1)
                Next i
            End If
        End If
        
    ElseIf InStr(InputGoal, "Random") > 0 And QueueAction <> "" Then
        PreviousGoal = EraseQueueAction("Random")
        If PreviousGoal = "" Then
            MsgBox ("出现bug，请将excel发给sgg，交换控制方继续对局")
            Application.DisplayAlerts = False
            ThisWorkbook.Close False
            Application.DisplayAlerts = True
        ElseIf PreviousGoal = "0" Then
        
        Else
            TempGoal = Split(PreviousGoal, "&")
            ReDim Goalrr(1 To UBound(TempGoal, 1) + 1)
            For i = 1 To UBound(Goalrr)
                Goalrr(i) = TempGoal(i - 1)
            Next i
        End If
    Else
        ' 返回所有目标
        ReDim Goalrr(1 To UBound(AllGoal, 1))
        For i = 1 To UBound(Goalrr)
            Goalrr(i) = AllGoal(i, 1)
        Next i
    End If
    
    ' ========== 存储并返回结果 ==========
    GetGoal = Goalrr
    PreviousGoal = Join(Goalrr, "&")
    If InStr(InputGoal, "Random") > 0 Then
        If PreviousGoal = "" Then Call UpSteps("Random", "0") Else Call UpSteps("Random", PreviousGoal)
    End If
End Function

' ==================================================
' 函数功能：比较卡牌属性模型
' 参数：
'   InputStr - 输入比较字符串
'   AllGoal - 所有目标数据
'   OrStr - 或条件字符串（可选）
' 返回：Variant - 筛选后的目标数据
' 说明：根据属性比较条件筛选卡牌，支持最高/最低值比较
' ==================================================
Function CompareModel(InputStr As String, AllGoal As Variant, Optional OrStr As String = "") As Variant
    ' ========== 变量声明 ==========
    Dim i%, j%                           ' 循环计数器
    Dim Str As String                    ' 临时字符串
    Dim Strr() As String                 ' 属性类型数组
    Dim EraseFlag As Boolean             ' 删除标志
    
    Dim CompareX As String, CompareY As String, CompareLogic As String    ' 比较参数
    Dim ResultX As Integer, ResultY As Integer                           ' 比较结果
    Dim CompareX2 As String, CompareY2 As String, CompareLogic2 As String ' 或条件比较参数
    Dim ResultX2 As Integer, ResultY2 As Integer                         ' 或条件比较结果
    
    Dim Irr() As Integer                 ' 临时整数数组

    CompareModel = AllGoal
    If InputStr = "" Then Exit Function

    ' ========== 解析比较条件 ==========
    ' 查找比较属性类型
    Strr = Split("OrigCost|Cost|OrigAttack|Attack|OrigDefense|Defense|OrigCountdown|Countdown", "|")
    For i = 0 To UBound(Strr)
        If Left(InputStr, Len(Strr(i))) = Strr(i) Then
            CompareX = Strr(i)
            Str = Mid(InputStr, 1 + Len(Strr(i)), 100)
            Exit For
        End If
    Next i
    
    ' ========== 最高/最低值处理 ==========
    ' 处理$Highest和$Lowest特殊条件
    If Left(Str, 1) = "$" Then
        ReDim Irr(1 To UBound(AllGoal, 1))
        For i = 1 To UBound(AllGoal, 1)
            Irr(i) = GetModel(CompareX, CStr(AllGoal(i, 2)), CStr(AllGoal(i, 5)), CStr(AllGoal(i, 1)))
            ' 倒数特殊处理
            If (CompareX = "Countdown" Or CompareX = "OrigCountdown") And Irr(i) = 0 Then
                If Str = "$Highest" Then Irr(i) = 999 Else Irr(i) = -1
            End If
        Next i
        If Str = "$Highest" Then ResultY = Application.WorksheetFunction.Max(Irr) Else ResultY = Application.WorksheetFunction.Min(Irr)
        Str = "=" & ResultY
    End If
    
    ' ========== 提取比较逻辑和数值 ==========
    CompareY = Mid(Str, 2, 10)
    CompareLogic = Left(Str, 1)
    If Left(Str, 2) = "<>" Then
        CompareY = Mid(Str, 3, 10)
        CompareLogic = Left(Str, 2)
    End If

    ' ========== 或条件处理 ==========
    ' 解析或条件比较参数
    If OrStr <> "" Then
        For i = 0 To UBound(Strr)
            If Left(OrStr, Len(Strr(i))) = Strr(i) Then
                CompareX2 = Strr(i)
                Str = Mid(OrStr, 1 + Len(Strr(i)), 100)
                Exit For
            End If
        Next i
        
        ' 或条件的最高/最低值处理
        If Left(Str, 1) = "$" Then
            ReDim Irr(1 To UBound(AllGoal, 1))
            For i = 1 To UBound(AllGoal, 1)
                Irr(i) = GetModel(CompareX2, CStr(AllGoal(i, 2)), CStr(AllGoal(i, 5)), CStr(AllGoal(i, 1)))
                ' 倒数特殊处理
                If (CompareX2 = "Countdown" Or CompareX2 = "OrigCountdown") And Irr(i) = 0 Then
                    If Str = "$Highest" Then Irr(i) = 999 Else Irr(i) = -1
                End If
            Next i
            If Str = "$Highest" Then ResultY2 = Application.WorksheetFunction.Max(Irr) Else ResultY2 = Application.WorksheetFunction.Min(Irr)
            Str = "=" & ResultY2
        End If
        
        ' 提取或条件比较逻辑和数值
        CompareY2 = Mid(Str, 2, 10)
        CompareLogic2 = Left(Str, 1)
        If Left(Str, 2) = "<>" Then
            CompareY2 = Mid(Str, 3, 10)
            CompareLogic2 = Left(Str, 2)
        End If
    End If

    ' ========== 属性比较筛选 ==========
    ' 遍历所有目标进行属性比较
    For i = 1 To UBound(AllGoal, 1)
        EraseFlag = True
        
        ' 获取比较数值
        ResultX = GetModel(CompareX, CStr(AllGoal(i, 2)), CStr(AllGoal(i, 5)), CStr(AllGoal(i, 1)))
        If CompareY = "0" Or Val(CompareY) > 0 Then
            ResultY = Val(CompareY)
        Else
            ResultY = GetModel(CompareY, CStr(AllGoal(i, 2)), CStr(AllGoal(i, 5)), CStr(AllGoal(i, 1)))
        End If
        
        ' 倒数特殊处理：没有倒数的卡牌直接排除
        If (CompareX = "Countdown" Or CompareX = "OrigCountdown") And ResultX = 0 Then AllGoal(i, 1) = 0
        
        ' ========== 主要条件比较 ==========
        ' 根据比较逻辑进行数值比较
        If CompareLogic = "<>" Then
            If ResultX <> ResultY Then EraseFlag = False
        ElseIf CompareLogic = "<" Then
            If ResultX < ResultY Then EraseFlag = False
        ElseIf CompareLogic = "=" Then
            If ResultX = ResultY Then EraseFlag = False
        ElseIf CompareLogic = ">" Then
            If ResultX > ResultY Then EraseFlag = False
        End If
        
        ' ========== 或条件比较 ==========
        ' 处理或条件比较
        If OrStr <> "" Then
            ResultX2 = GetModel(CompareX2, CStr(AllGoal(i, 2)), CStr(AllGoal(i, 5)), CStr(AllGoal(i, 1)))
            If CompareY2 = "0" Or Val(CompareY2) > 0 Then
                ResultY2 = Val(CompareY2)
            Else
                ResultY2 = GetModel(CompareY2, CStr(AllGoal(i, 2)), CStr(AllGoal(i, 5)), CStr(AllGoal(i, 1)))
            End If
            
            ' 倒数特殊处理
            If (CompareX2 = "Countdown" Or CompareX2 = "OrigCountdown") And ResultX2 = 0 Then AllGoal(i, 1) = 0
            
            ' 或条件比较逻辑
            If CompareLogic2 = "<>" Then
                If ResultX2 <> ResultY2 Then EraseFlag = False
            ElseIf CompareLogic2 = "<" Then
                If ResultX2 < ResultY2 Then EraseFlag = False
            ElseIf CompareLogic2 = "=" Then
                If ResultX2 = ResultY2 Then EraseFlag = False
            ElseIf CompareLogic2 = ">" Then
                If ResultX2 > ResultY2 Then EraseFlag = False
            End If
        End If
        
        ' ========== 标记删除 ==========
        ' 根据比较结果标记需要删除的目标
        If EraseFlag Then AllGoal(i, 1) = Empty
    Next i
    
    ' ========== 清理并返回结果 ==========
    AllGoal = NonEmptyArray(AllGoal)
    CompareModel = AllGoal
End Function
