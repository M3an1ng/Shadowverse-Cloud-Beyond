Attribute VB_Name = "·2·1·录卡·GetCardSetID"
' ==================================================
' 过程功能：打开录卡窗口
' ==================================================
Sub ShowEnterCard()
    ' ========== 优化设置 ==========
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    Application.DisplayAlerts = False
    
    EnterCard.Show   ' 显示录卡窗体
    
CleanUp:
    ' ========== 恢复应用程序状态 ==========
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
End Sub

' ==================================================
' 函数功能：卡包ID生成器
' 参数：CardSet - 卡包名称
' 返回：卡包ID字符串（两位数字）或错误信息
' ID规则：AABCDXX（整体卡牌ID结构，本函数生成AA部分）
'   AA代表卡包，从10开始，最多90个卡包（10-99）
'   B代表职业（0中立，1精灵，2皇家，3巫师，4龙族，5梦魇，6主教，7超越者）
'   C代表是否可构筑（0不可构筑，1可构筑）
'   D代表稀有度（1虹，2金，3银，4铜）
'   XX代表序号，从01开始，同条件下最多99张卡片
' ==================================================
Function GetCardSetID(CardSet As String) As String
    On Error GoTo ErrorHandler
    
    ' ========== 常量定义 ==========
    Const SHEET_NAME As String = "Effect"   ' 数据源工作表名称
    Const START_ID As Integer = 10           ' 起始卡包ID
    Const END_ID As Integer = 99             ' 结束卡包ID（两位数范围）
    
    ' ========== 变量声明 ==========
    Dim ws As Worksheet              ' 工作表对象
    Dim lastRow As Long              ' 最后数据行号
    Dim i As Long                    ' 循环计数器
    Dim idValue As String            ' 临时存储ID值
    Dim foundID As String            ' 找到的卡包ID
    Dim existingIDs As Object        ' 已使用ID字典
    Dim newID As Integer             ' 新ID编号
    
    ' ========== 工作表初始化 ==========
    Set ws = ThisWorkbook.Sheets(SHEET_NAME)   ' 设置Effect工作表
    
    ' 获取B列最后一行数据位置
    lastRow = ws.Cells(ws.rows.count, "B").End(xlUp).Row
    
    ' ========== 已使用ID收集 ==========
    Set existingIDs = CreateObject("Scripting.Dictionary")   ' 创建字典存储已用ID
    
    ' 遍历Effect工作表查找卡包并收集所有已用ID
    For i = 3 To lastRow   ' 从第3行开始（假设第1-2行是标题）
        ' 检查当前行是否匹配目标卡包
        If LCase(Trim(ws.Cells(i, 2).Value)) = LCase(Trim(CardSet)) Then
            idValue = CStr(ws.Cells(i, 1).Value)   ' 获取卡牌ID
            
            ' 提取ID前两位作为卡包ID
            If Len(idValue) >= 2 Then
                foundID = Left(idValue, 2)
                GetCardSetID = foundID   ' 返回找到的卡包ID
                Exit Function
            End If
        End If
        
        ' 收集所有已使用的ID前两位（用于后续新ID生成）
        If Not IsEmpty(ws.Cells(i, 1).Value) Then
            idValue = CStr(ws.Cells(i, 1).Value)
            If Len(idValue) >= 2 Then
                existingIDs(Left(idValue, 2)) = True   ' 记录已使用的两位ID
            End If
        End If
    Next i
    
    ' ========== 新卡包ID生成 ==========
    newID = START_ID   ' 从起始ID开始查找
    
    ' 在10-99范围内查找第一个可用ID
    Do While newID <= END_ID
        Dim newIDStr As String
        newIDStr = Format(newID, "00")   ' 格式化为两位数字符串
        
        ' 检查ID是否未被使用
        If Not existingIDs.Exists(newIDStr) Then
            ' 注释掉的代码：实际使用时可能需要添加新行到工作表
            'ws.Cells(lastRow + 1, 1).Value = newIDStr    ' A列：卡包ID
            'ws.Cells(lastRow + 1, 2).Value = CardSet     ' B列：卡包名称
            
            GetCardSetID = newIDStr   ' 返回新生成的卡包ID
            Exit Function
        End If
        
        newID = newID + 1   ' 检查下一个ID
    Loop
    
    ' ========== ID范围已满处理 ==========
    GetCardSetID = "ID overflow"   ' 所有ID都被使用
    Exit Function

ErrorHandler:
    ' ========== 错误处理 ==========
    GetCardSetID = "Error: " & Err.Description   ' 返回错误信息
End Function
