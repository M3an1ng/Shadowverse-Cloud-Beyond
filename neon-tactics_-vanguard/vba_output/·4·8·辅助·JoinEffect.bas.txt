Attribute VB_Name = "·4·8·辅助·JoinEffect"
' ==================================================
' 函数功能：效果字符串合并器
' 参数：InputEffect - 输入的效果字符串
' 返回：String - 合并后的效果字符串
' 说明：合并效果字符串中的攻击力、生命值和费用修正，将分散的修正值汇总到字符串末尾
' ==================================================
Function JoinEffect(InputEffect As String)
    ' ========== 变量声明 ==========
    Dim i As Integer, j As Integer, k As Integer   ' 循环计数器和位置索引
    Dim Str As String           ' 临时字符串
    Dim Str2 As String          ' 临时字符串
    Dim EffectSplit() As String ' 效果分割数组
    Dim Attack As Integer       ' 攻击力修正值
    Dim Defense As Integer      ' 生命值修正值
    Dim Cost As Integer         ' 费用修正值
    Dim AttackFlag As Boolean   ' 攻击力是否被设为固定值
    Dim DefenseFlag As Boolean  ' 生命值是否被设为固定值
    Dim CostFlag As Boolean     ' 费用是否被设为固定值
    
    ' ========== 输入验证 ==========
    If InputEffect = "" Then Exit Function
    
    ' ========== 效果分割 ==========
    EffectSplit = SplitEffect(InputEffect)
    
    ' ========== 倒序遍历效果 ==========
    ' 从后向前遍历效果数组，处理修正值
    For k = 0 To UBound(EffectSplit)
        i = UBound(EffectSplit) - k   ' 倒序索引
        
        ' 跳过Until效果和包含{}的复杂效果
        If InStr(EffectSplit(i), "Until") = 0 And InStr(EffectSplit(i), "{") = 0 Then
            ' ========== 身材修正处理 ==========
            ' 处理攻击力/生命值修正（格式：+1/+1）
            If InStr(EffectSplit(i), "/") > 0 Then
                j = InStr(EffectSplit(i), "/")   ' 找到分隔符位置
                Str = Left(EffectSplit(i), j - 1)   ' 提取攻击力部分
                
                ' ========== 攻击力修正计算 ==========
                If Left(Str, 1) = "$" Then
                    ' 设为固定值操作符
                    Attack = Attack + Val(Mid(Str, 2, 10))
                    AttackFlag = True   ' 标记攻击力已被设为固定值
                ElseIf Left(Str, 1) = "*" Or Left(Str, 1) = "%" Then
                    ' 遇到乘除操作符，停止向前合并
                    Exit For
                ElseIf Not AttackFlag Then
                    ' 加减操作符，且攻击力未被设为固定值
                    Attack = Attack + Val(Str)
                End If
                
                ' ========== 生命值修正计算 ==========
                Str = Mid(EffectSplit(i), j + 1, 10)   ' 提取生命值部分
                If Left(Str, 1) = "$" Then
                    ' 设为固定值操作符
                    Defense = Defense + Val(Mid(Str, 2, 10))
                    DefenseFlag = True   ' 标记生命值已被设为固定值
                ElseIf Left(Str, 1) = "*" Or Left(Str, 1) = "%" Then
                    ' 遇到乘除操作符，停止向前合并
                    Exit For
                ElseIf Not DefenseFlag Then
                    ' 加减操作符，且生命值未被设为固定值
                    Defense = Defense + Val(Str)
                End If
                
                ' 标记该效果已被处理
                EffectSplit(i) = "\"
                
            ' ========== 费用修正处理 ==========
            ' 处理费用修正（格式：Cost+1）
            ElseIf InStr(EffectSplit(i), "Cost") > 0 Then
                Str = Mid(EffectSplit(i), 1 + Len("Cost"), 10)   ' 提取费用修正部分
                If Left(Str, 1) = "$" Then
                    ' 设为固定值操作符
                    Cost = Cost + Val(Mid(Str, 2, 10))
                    CostFlag = True   ' 标记费用已被设为固定值
                ElseIf Left(Str, 1) = "*" Or Left(Str, 1) = "%" Then
                    ' 遇到乘除操作符，停止向前合并
                    Exit For
                ElseIf Not CostFlag Then
                    ' 加减操作符，且费用未被设为固定值
                    Cost = Cost + Val(Str)
                End If
                
                ' 标记该效果已被处理
                EffectSplit(i) = "\"
            End If
            
        ' ========== Until效果处理 ==========
        ' 遇到Until效果，停止向前合并
        ElseIf InStr(EffectSplit(i), "Until") > 0 And InStr(EffectSplit(i), "{") = 0 Then
            Exit For
        End If
    Next k
    
    ' ========== 清理已处理的效果 ==========
    Str = Join(EffectSplit, "||") & "||"   ' 重新连接效果字符串
    Str = Replace(Str, "\||", "")          ' 移除被标记的效果
    
    ' ========== 整合添加汇总效果 ==========
    ' 添加攻击力/生命值汇总效果
    If Attack <> 0 Or Defense <> 0 Or AttackFlag Or DefenseFlag Then
        If AttackFlag Then
            Str2 = "$" & Attack   ' 固定值格式
        Else
            ' 加减值格式，正数添加+号
            If Attack >= 0 Then Str2 = "+" & Attack Else Str2 = Attack
        End If
        Str2 = Str2 & "/"   ' 添加分隔符
        If DefenseFlag Then
            Str2 = Str2 & "$" & Defense   ' 固定值格式
        Else
            ' 加减值格式，正数添加+号
            If Defense >= 0 Then Str2 = Str2 & "+" & Defense Else Str2 = Str2 & Defense
        End If
        Str = Str & Str2 & "||"   ' 添加到效果字符串
    End If
    
    ' 添加费用汇总效果
    If Cost <> 0 Or CostFlag Then
        If CostFlag Then
            Str2 = "Cost$" & Cost   ' 固定值格式
        Else
            ' 加减值格式，正数添加+号
            If Cost >= 0 Then Str2 = "Cost+" & Cost Else Str2 = "Cost" & Cost
        End If
        Str = Str & Str2 & "||"   ' 添加到效果字符串
    End If
    
    JoinEffect = Str   ' 返回合并后的效果字符串
End Function
