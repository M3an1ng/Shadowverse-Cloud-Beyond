Attribute VB_Name = "·7·1·对局·StartGame"
' ==================================================
' 子过程功能：开始新游戏
' 说明：初始化游戏环境，设置先后手，创建卡组，进行起始换牌
' ==================================================
Sub StartGame()
    ' ========== 优化设置 ==========
    ' 关闭屏幕更新和事件处理以提高性能
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    Application.DisplayAlerts = False
    
    '初始化随机数生成器
    Randomize
    
    ' ========== 变量声明 ==========
    Dim options() As String          ' 选项数组
    Dim Arr As Variant               ' 临时数组
    Dim Brr As Variant               ' 临时数组
    Dim Str As String                ' 临时字符串
    Dim Str2 As String                ' 临时字符串
    Dim FirstMove As String          ' 先手标识
    Dim BattleYourself As Boolean    ' 左右互搏
    Dim Effectrr() As String         ' 效果数组
    Dim i%, j%, k%, m%               ' 循环计数器
    Dim AllData As Variant           ' 所有数据
    Dim OrigDeck As Variant          ' 原始牌组
    Dim EnemyDeck As String          ' 牌组数据
    Dim EnemySwap As String          ' 换牌数据
    
    ' ========== 下载并处理原始卡组 ==========
    '获取牌组
    ReDim OrigDeck(1 To 40)
    Arr = Down("OrigDeck")
    m = 0
    For i = 1 To UBound(Arr, 1)
        If Arr(i, 1) = "" Then Exit For
        Str = IDByName(CStr(Arr(i, 2)))
        EnemyDeck = EnemyDeck & "||" & Str & "|" & Arr(i, 3)
        For j = 1 To Arr(i, 3)
            m = m + 1
            OrigDeck(m) = Str
        Next j
    Next i
    
    ' ========== 先后手选择 ==========
    ' 生成随机数并创建选项
    options = Split(".r1d100=" & Int((100 - 1 + 1) * Rnd + 1) & "（点大的先手）" & SentenceTranslate("|FirstMove|SecondMove|BattleYourself"), "|")
    i = Val(ShowChoice(options, 60, 3))
    
    ' 用户取消选择则退出
    If i = 0 Then GoTo CleanUp
    
    ' 处理用户选择
    If options(i) = Translate("BattleYourself") Then
        FirstMove = "FirstMove"
        BattleYourself = True
        Call Up("BattleYourself", True)      ' 设置自战模式
    Else
        FirstMove = Translate(options(i))
        BattleYourself = False
        Call Up("BattleYourself", False)     ' 设置正常对战模式
    End If
    Call Up("FirstMove", FirstMove)          ' 存储先手信息

    ' ========== 主区域初始化 ==========
    ' 初始化双方玩家状态
    Call Up("EnemyDefense", 20)              ' 敌方当前生命值
    Call Up("EnemyMaxDefense", 20)           ' 敌方最大生命值
    Call Up("EnemyPP", 0)                    ' 敌方当前能量点
    Call Up("EnemyMaxPP", 0)                 ' 敌方最大能量点
    Call Up("EnemyEP", 0)                    ' 敌方当前EP
    Call Up("EnemySEP", 0)                   ' 敌方当前SEP
    
    Call Up("AlliedDefense", 20)             ' 我方当前生命值
    Call Up("AlliedMaxDefense", 20)          ' 我方最大生命值
    Call Up("AlliedPP", 0)                   ' 我方当前能量点
    Call Up("AlliedMaxPP", 0)                ' 我方最大能量点
    Call Up("AlliedEP", 0)                   ' 我方当前EP
    Call Up("AlliedSEP", 0)                  ' 我方当前SEP
    
    ' 初始化游戏区域
    Call Up("EnemyField", Empty)             ' 敌方场地
    Call Up("AlliedField", Empty)            ' 我方场地
    Call Up("Hand", Empty)                   ' 手牌区域
    Call Up("Crest", Empty)                  ' 纹章区域
    Call Up("Deck·Cards", 0)                 ' 牌组卡牌数
    Call Up("Turn", "S0")                    ' 当前回合（F先手）
    Call Up("StepsArea", Empty)              ' 步骤记录区域

    ' ========== 数据区域初始化 ==========
    Call Up("Data", Empty)                   ' 卡牌数据
    Call Up("Steps", Empty)                  ' 游戏步骤记录
    Call Up("CountNum", Empty)               ' 计数数据
    Call Up("Log", Empty)                    ' 对局日志

    ' ========== 创建卡组 ==========
    ' 循环处理双方玩家（自战模式时处理两次）
    For k = 1 To 2
        ' 自战模式特殊处理：第二次循环时选择第二个卡组
        If k = 2 Then
            If FirstMove = "FirstMove" Then FirstMove = "SecondMove" Else FirstMove = "FirstMove" ' 第二个玩家为后手
            If BattleYourself Then
                Call SelectDeck              ' 选择第二个卡组
                
                ' ========== 下载并处理原始卡组 ==========
                ' 根据卡组数据填充临时卡组数组
                ReDim OrigDeck(1 To 40)
                Arr = Down("OrigDeck")
                m = 0
                For i = 1 To UBound(Arr, 1)
                    If Arr(i, 1) = "" Then Exit For
                    Str = IDByName(CStr(Arr(i, 2)))
                    For j = 1 To Arr(i, 3)
                        m = m + 1
                        OrigDeck(m) = Str
                    Next j
                Next i
            Else
InputAgain:
                Str = EnemyDeck
                EnemyDeck = InputBox("1.「剪切」发送给对手" & Chr(13) & "2.「粘贴」对手提供的信息，然后点确定", "交换牌组", Str)
                If EnemyDeck = Str Then
                    Str = MsgBox("对手的牌组与你完全一样，你确定吗？", vbYesNo, "可能粘贴错误")
                    If Str <> vbYes Then GoTo InputAgain
                End If
                
                ReDim OrigDeck(1 To 40)
                m = 0
                Arr = Split(EnemyDeck, "||")
                For i = 1 To UBound(Arr)
                    If Left(Arr(i), 1) <> 1 Then Exit For
                    Brr = Split(Arr(i), "|")
                    For j = 1 To Brr(1)
                        m = m + 1
                        OrigDeck(m) = Brr(0)
                    Next j
                Next i
                EnemySwap = Arr(i) & "||" & Arr(i + 1) & "||" & Arr(i + 2)
            End If
        End If
        
        ' 检查卡组是否满足40张
        If UBound(OrigDeck) < 40 Then
            MsgBox "错误：牌组不满40张", vbExclamation
            GoTo CleanUp
        End If
        
        ' ========== 设置职业 ==========
        If k = 1 Then
            Str = Translate(Down("Class"))
            Call Up("Class", Str, Left(FirstMove, 1) & 1000)  ' 设置职业
            EnemyDeck = Str & EnemyDeck
        Else
            Arr = Split(EnemyDeck, "||")
            Call Up("Class", Arr(0), Left(FirstMove, 1) & 1000)  ' 设置职业
        End If
        
        ' ========== 生成随机序列 ==========
        Brr = Split("|28|29|8|34|13|27|15|16|21|33|23|3|30|6|1|4|20|11|25|17|9|10|12|40|35|26|19|7|24|37|14|5|18|39|32|2|38|36|31|22", "|")            ' 生成40个随机数用于洗牌
        
        ' ========== 创建卡牌数据 ==========
        AllData = Down("Data")               ' 获取现有数据
        ' 找到第一个空行
        For j = 1 To UBound(AllData, 1)
            If AllData(j, 1) = "" Then Exit For
        Next j
        
        ' 创建主战者卡牌
        AllData(j, 1) = Left(FirstMove, 1) & 1000        ' 主战者ID
        If FirstMove = "FirstMove" Then
            AllData(j, 2) = "1000001"                    ' 先手主战者ID
        Else
            AllData(j, 2) = "1000002"                    ' 后手主战者ID
        End If
        AllData(j, 3) = "S0"                             ' 回合信息
        AllData(j, 4) = "Leader"                         ' 区域：主战者
        
        ' 创建牌组卡牌
        For i = 1 To UBound(OrigDeck)
            AllData(j + i, 1) = Left(FirstMove, 1) & 1000 + i    ' 卡牌ID
            AllData(j + i, 2) = OrigDeck(Brr(i))                 ' 卡牌原型ID（随机洗牌）
            'AllData(j + i, 3) = "S0"                             ' 回合信息
            'AllData(j + i, 4) = "Deck"                           ' 区域：牌组
        Next i
        Call Up("Data", AllData)             ' 更新数据
        For i = 1 To UBound(OrigDeck)
            Call UpByMatchID(CStr(AllData(j + i, 1)), "Area", "Deck")
        Next i
        
        ' ========== 创建信仰效果 ==========
        Arr = Translate("Faith")
        AllData = NonEmptyArray(AllData)     ' 清理空行
        ' 遍历所有卡牌检查信仰效果
        For j = 1 To UBound(AllData, 1)
            Str = IDByName(Arr & ":" & DownByID(CStr(AllData(j, 2)), "Name"))
            If Val(Str) > 0 Then             ' 如果存在信仰卡牌
                Effectrr = GetEffect(CStr(AllData(j, 1)))  ' 获取效果
                ' 查找信仰效果
                For i = 1 To UBound(Effectrr)
                    If Left(Effectrr(i), Len("Faith")) = "Faith" Then Exit For
                Next i
                ' 如果找到信仰效果则执行
                If i <= UBound(Effectrr) Then
                    Call GetAbility("||Faith", CStr(AllData(j, 1)))
                    Call Up("Faith", Val(Mid(Effectrr(i), 1 + Len("Faith||"), 10)), CStr(AllData(j, 1)))
                End If
            End If
        Next j
        
        ' ========== 起始换牌阶段 ==========
        ' 从牌组随机获取4张卡牌作为起始手牌选项
        If k = 1 Or BattleYourself Then
            Arr = GetGoal("Random&4-Allied-Deck-Card", Left(FirstMove, 1) & 1000)
            EnemyDeck = EnemyDeck & "||" & PreviousGoal
        Else
            Arr = Split(EnemySwap, "||")
            Arr = Split("&" & Arr(0), "&")
        End If
        
        ReDim options(0 To UBound(Arr))
        options(0) = Translate("CardExchange")           ' 换牌提示
        ' 填充选项数组
        For i = 1 To UBound(Arr)
            options(i) = DownByID(CStr(DownByMatchID(CStr(Arr(i)), "ID")), "Name")
        Next i
        
        ' 显示换牌选择界面（负数允许反选）
        If k = 1 Or BattleYourself Then
            Str = ShowChoice(options, 150, 4, -1)
            EnemyDeck = EnemyDeck & "||" & Str
        Else
            Brr = Split(EnemySwap, "||")
            Str = Brr(1)
        End If
        
        ' 处理换牌：未选择的卡牌放回牌组
        Str2 = Empty
        For i = 1 To UBound(Arr)
            If InStr(Str, i) = 0 Then
                Call UpByMatchID(CStr(Arr(i)), "Area", "Hand")  ' 选择的卡牌加入手牌
            Else
                Str2 = Str2 & "&" & Arr(i)
                Call UpByMatchID(CStr(Arr(i)), "Area", "Wait")  ' 选择的卡牌加入手牌
            End If
        Next i
        
        ' ========== 补充手牌至4张 ==========
        ' 计算需要补充的卡牌数量
        If k = 1 Or BattleYourself Then
            Arr = GetGoal("Random&" & (Len(Str) + 1) / 2 & "-Allied-Deck-Card", Left(FirstMove, 1) & 1000)
            EnemyDeck = EnemyDeck & "||" & PreviousGoal
        Else
            Arr = Split(EnemySwap, "||")
            Arr = Split("&" & Arr(2), "&")
        End If
        For i = 1 To UBound(Arr)
            Call UpByMatchID(CStr(Arr(i)), "Area", "Hand")      ' 补充卡牌加入手牌
        Next i
        
        '不要的牌放回牌组
        Arr = Split(Str2, "&")
        For i = 1 To UBound(Arr)
            Call UpByMatchID(CStr(Arr(i)), "Area", "Deck")
        Next i
        
        If k = 2 Then
            If FirstMove = "FirstMove" Then FirstMove = "SecondMove" Else FirstMove = "FirstMove" ' 第二个玩家为后手
        End If
    Next k

    ' ========== 开始先手回合 ==========
    If FirstMove = "FirstMove" Then
        Call Up("Turn", "F1")
        Call UpSteps("TurnStart", Left(FirstMove, 1) & 1000)
        Call TurnStartFunction
    End If

    ' ========== 刷新界面 ==========
    Call UIRefresh
    Call UpStepsArea

CleanUp:
    ' ========== 恢复应用程序状态 ==========
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
End Sub
