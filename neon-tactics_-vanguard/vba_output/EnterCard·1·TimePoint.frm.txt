' ==================================================
' 事件功能：特殊效果处理 - 法术增幅
' 说明：处理法术增幅效果，提供多种增幅类型选择
' ==================================================
Private Sub Spellboost_Click()
    ' ========== 增幅类型选择 ==========
    ' 询问法术增幅类型
    Dim options() As String
    options = Split(SentenceTranslate("Spellboost|Cost|Attack&Defense|SetX"), "|")
    Dim i%, j%, k%
    i = Val(ShowChoice(options, 90, 3))
    
    ' ========== 费用增幅处理 ==========
    If i = 1 Then
        j = NumInputBox(Translate("Cost"), "Engage")
        TimePoint = "Spellboost|Cost"
        If j >= 0 Then TimePoint = TimePoint & "+" & j Else TimePoint = TimePoint & j
        
    ' ========== 身材增幅处理 ==========
    ElseIf i = 2 Then
        j = NumInputBox(Translate("Attack"), "Engage")
        TimePoint = "Spellboost|"
        If j >= 0 Then TimePoint = TimePoint & "+" & j Else TimePoint = TimePoint & j
        TimePoint = TimePoint & "/"
        j = NumInputBox(Translate("Defense"), "Engage")
        If j >= 0 Then TimePoint = TimePoint & "+" & j Else TimePoint = TimePoint & j
        
    ' ========== X变量设置处理 ==========
    ElseIf i = 3 Then
        j = NumInputBox("SetX", "Engage")
        TimePoint = "Spellboost|X=" & j
    End If
    
    ' ========== 直接退出处理 ==========
    ' 法术增幅不需要进入能力选择，直接设置效果并退出
    CardEffect = TimePoint
    Unload EnterCard·1·TimePoint
End Sub

    ' ========== 时间点构建 ==========

    Else

End Sub

' ==================================================
' 事件功能：常驻效果处理
' 说明：处理各种常驻效果选择
' ==================================================
Private Sub Permannent_Click()
    ' ========== 类型验证 ==========
    ' 如果是法术且没有父效果则退出
    If DownByID("1000000", "Type") = "Spell" And Len(FatherEffect) = 0 Then Exit Sub
    
    ' ========== 常驻效果选择 ==========
    Dim options() As String
    options = Split(SentenceTranslate("Permannent|" & _
        "CantBeDestroyed|" & _
        "CantStrike|" & _
        "CanAttackTimes|" & _
        "LeaveBanish|" & _
        "CantPlay|" & _
        "CantSelectOther|" & _
        "CantTakeMoreDamage|" & _
        "TakeDamage|" & _
        "DoDamage|" & _
        "NoEP|" & _
        "NoSEP|" & _
        "EngageCost|" & _
        "IncreaseModes"), "|")
    Dim i%
    i = Val(ShowChoice(options, 200, 2))
    
    ' ========== 效果设置 ==========
    CardEffect = CardEffect & Translate(options(i))
    
    ' ========== 特殊参数处理 ==========
    ' 可攻击次数需要额外参数
    If options(i) = Translate("CanAttackTimes") Then
        CardEffect = CardEffect & "&" & NumInputBox(Translate("CanAttackTimes"), "Engage")
    ElseIf options(i) = Translate("CantTakeMoreDamage") Then
        CardEffect = CardEffect & "&" & NumInputBox(Translate("CantTakeMoreDamage"), "Engage")
    ElseIf options(i) = Translate("TakeDamage") Then
        CardEffect = CardEffect & "&" & NumInputBox(Translate("TakeDamage"), "Engage")
    ElseIf options(i) = Translate("DoDamage") Then
        CardEffect = CardEffect & "&" & NumInputBox(Translate("DoDamage"), "Engage")
    ElseIf options(i) = Translate("EngageCost") Then
        CardEffect = CardEffect & "&" & NumInputBox(Translate("EngageCost"), "Engage")
        
    ElseIf options(i) = Translate("CantStrike") Then
        EnterCard·4·Goal.Tag = "All|Enemy|Field|Follower&Leader"
        EnterCard·4·Goal.Show
        CardEffect = CardEffect & "|" & Goal
    End If
    
    ' ========== 直接退出处理 ==========
    ' 常驻效果不需要进入能力选择，直接退出
    Unload EnterCard·1·TimePoint
End Sub

    ' ========== 效果设置 ==========
    CardEffect = CardEffect & "Barrier"
    
    ' ========== 直接退出处理 ==========
    Unload EnterCard·1·TimePoint
End Sub

' ========== 分割线：窗体初始化和工具函数 ==========

' ==================================================
' 事件功能：窗体激活初始化
' 说明：窗体显示时清空时间点变量
' ==================================================
Private Sub UserForm_Activate()
    ' ========== 变量初始化 ==========
    ' 清空时间点变量
    TimePoint = Empty
End Sub

' ==================================================
' 函数功能：统一退出处理
' 说明：记录效果并打开能力选择窗体
' ==================================================
Function UnloadForm()
    ' ========== 效果记录 ==========
    ' 将时间点添加到效果字符串
    CardEffect = CardEffect & TimePoint & "||"
    
    ' ========== 窗体切换 ==========
    ' 卸载本窗体并打开能力选择窗体
    Unload EnterCard·1·TimePoint
    EnterCard·3·Ability.Show
End Function
