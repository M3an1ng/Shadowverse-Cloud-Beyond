' 窗体初始化事件
Private Sub UserForm_Initialize()
    ' 设置默认大小
    Me.Width = 600
    Me.Height = 500
    ' 设置图片控件为自动调整大小
    Me.imgPreview.PictureSizeMode = fmPictureSizeModeAutoSize
End Sub

' 窗体激活时调整大小以适应图片
Private Sub UserForm_Activate()
    AdjustFormToImage
End Sub

' 调整窗体大小以适应图片
Private Sub AdjustFormToImage()
    On Error Resume Next
    
    Dim imgWidth As Double
    Dim imgHeight As Double
    Dim maxWidth As Double
    Dim maxHeight As Double
    Dim scaleFactor As Double
    
    ' 获取图片原始尺寸
    imgWidth = Me.imgPreview.Width
    imgHeight = Me.imgPreview.Height
    
    ' 如果图片尺寸为0（可能未加载），则退出
    If imgWidth = 0 Or imgHeight = 0 Then Exit Sub
    
    ' 设置最大显示尺寸（可根据需要调整）
    maxWidth = 1800
    maxHeight = 900

    ' 如果图片尺寸超过最大值，则按比例缩放
    If imgWidth > maxWidth Or imgHeight > maxHeight Then
        scaleFactor = Application.WorksheetFunction.Min(maxWidth / imgWidth, maxHeight / imgHeight)
        imgWidth = imgWidth * scaleFactor
        imgHeight = imgHeight * scaleFactor
    End If
    
    ' 设置窗体大小（加上边框和标题栏的空间）
    Me.Width = imgWidth + 30
    Me.Height = imgHeight + 50
    
    ' 设置图片控件大小
    Me.imgPreview.Width = imgWidth
    Me.imgPreview.Height = imgHeight
    
    ' 居中显示
    Me.imgPreview.Left = (Me.InsideWidth - imgWidth) / 2
    Me.imgPreview.Top = (Me.InsideHeight - imgHeight) / 2
    
    ' 确保窗体不会超出屏幕
    If Me.Width > Application.Width Then
        Me.Width = Application.Width * 0.9
    End If
    If Me.Height > Application.Height Then
        Me.Height = Application.Height * 0.9
    End If
    
    ' 重新居中窗体
    Me.StartUpPosition = 0
    Me.Left = (Application.Left + Application.Width - Me.Width) / 2
    Me.Top = (Application.Top + Application.Height - Me.Height) / 2
End Sub

' 双击图片打开文本窗体
Private Sub imgPreview_DblClick(ByVal Cancel As MSForms.ReturnBoolean)

    '打开文本窗体
    EffectForm.Caption = ImageDisplayForm.Caption
    Unload ImageDisplayForm
    EffectForm.Show
End Sub
