Attribute VB_Name = "·7·5·对局·MatchEvolve"
' ==================================================
' 过程功能：处理从者进化操作的入口函数
' 说明：检查进化条件并执行进化操作
' 包含回合检查、进化点检查、进化状态验证等逻辑
' ==================================================
Sub MatchEvolve()
    ' ========== 优化设置 ==========
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    Application.DisplayAlerts = False
    
    ' ========== 变量声明 ==========
    Dim i%
    Dim Arr As Variant
    Dim Turn As String
    Dim MatchID As String
    
    ' ========== 回合权限检查 ==========
    ' 不是自己的回合直接退出
    Turn = Down("Turn")
    If Left(Turn, 1) <> Left(Down("FirstMove"), 1) Then GoTo CleanUp
    If Left(Turn, 1) = "F" Then
        If Val(Mid(Turn, 2, 10)) < 5 Then GoTo CleanUp
    Else
        If Val(Mid(Turn, 2, 10)) < 4 Then GoTo CleanUp
    End If
    
    ' ========== 战场选择检查 ==========
    ' 是否选择了战场
    Arr = IsSelected("AlliedField")
    If IsEmpty(Arr) Then GoTo CleanUp
    ' ========== 进化状态检查 ==========
    ' 是否已进化（检查按钮文本状态）
    If Sheets("Main").Shapes("EvolveBtn").TextFrame.Characters.Text = Translate("Evolved") Then GoTo CleanUp
    If Sheets("Main").Shapes("SuperEvolveBtn").TextFrame.Characters.Text = Translate("SuperEvolved") Then GoTo CleanUp
    
    ' ========== 战场信息获取 ==========
    ' 下载战场信息
    Arr = Down("Field", Left(Down("FirstMove"), 1) & 1000)
    ' 获取MatchID，如果为空则刷新并退出
    If Selection.Column - 2 > UBound(Arr, 1) Or Arr(1, 1) = "" Then
        Call UIRefresh
        GoTo CleanUp
    End If
    MatchID = Arr(Selection.Column - 2, 1)
    
    ' ========== 随从检查 ==========
    If DownByID(DownByMatchID(MatchID, "ID"), "Type") <> "Follower" Then GoTo CleanUp
    
    ' ========== 已进化状态检查 ==========
    ' 检查是否已经进化（注释标明不严谨实现）
    If InStr(Arr(Selection.Column - 2, 5), "Evolved||") > 0 Then GoTo CleanUp
    
    ' ========== 进化点检查 ==========
    ' 是否有EP（进化点）
    Arr = GetEffect(MatchID)
    For i = 1 To UBound(Arr)
        If Arr(i) = "NoEP" Then Exit For
    Next i
    If i > UBound(Arr) Then
        If Val(Down("AlliedEP")) <= 0 Then GoTo CleanUp
        Arr = EvolveFunction(MatchID)
    Else
        Arr = EvolveFunction(MatchID, True)
    End If
    
    ' ========== 执行进化 ==========
    'Arr = EvolveFunction(MatchID)
    If Not Arr Then GoTo CleanUp

    ' ========== 界面刷新 ==========
    Call UIRefresh
    Call UpStepsArea
    
CleanUp:
    ' ========== 恢复应用程序状态 ==========
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Application.Calculation = xlCalculationAutomatic
    Application.DisplayAlerts = True
End Sub

' ==================================================
' 函数功能：执行具体的进化逻辑
' 参数：MatchID - 要进化的从者MatchID
' 说明：处理进化效果、消耗进化点、触发进化时能力等
' 包括进化失败的回滚处理
' ==================================================
Function EvolveFunction(MatchID As String, Optional NoEP As Boolean = False) As Boolean
    ' ========== 变量声明 ==========
    Dim i%, j%, k%
    Dim Arr As Variant
    Dim Flag As Boolean
    Dim Effect As String
    Dim EffectSaved As String
    Dim Effectrr() As String
    Dim EvolveFlag As Boolean
    EvolveFunction = True
    
    ' ========== 初始化设置 ==========
    ' 清空能力队列
    QueueAbility = Empty
    QueueMatchID = Empty
    
    ' ========== 基础进化效果应用 ==========
    ' 应用+2/+2与突进效果，标记为已进化
    EffectSaved = DownByMatchID(MatchID, "Effect")
    Effect = EffectSaved & "{Rush}||Evolved||"
    Call UpByMatchID(MatchID, "Effect", Effect)
    Call GetAbility("||+2/+2|ThisCard", MatchID)
    
    ' ========== 进化点消耗 ==========
    ' 消耗1点EP（进化点）
    If Not NoEP Then Call Up("EP", Down("EP", MatchID) - 1, MatchID)
    Call UpSteps("Evolve", MatchID, DownByID(DownByMatchID(MatchID, "ID"), "Name"))
    
    ' ========== 进化效果处理 ==========
    ' 遍历并触发进化相关效果
    Flag = True
    Effectrr = GetEffect(MatchID)
    For i = 1 To UBound(Effectrr)
        If Left(Effectrr(i), Len("Evolve")) = "Evolve" And InStr(Effect, "Remove|Evolve") = 0 Then
            Flag = GetTimePoint(Effectrr(i), MatchID)
            EvolveFlag = True
        End If
    Next i
    
    ' ========== 进化失败处理 ==========
    ' 如果选择失败，回滚进化操作
    If Not Flag Then
        If Not NoEP Then Call Up("EP", Down("EP", MatchID) + 1, MatchID)
        Call EraseSteps("Evolve")
        Call UpByMatchID(MatchID, "Effect", EffectSaved)
        QueueAbility = Empty
        QueueMatchID = Empty
        EvolveFunction = False
        Exit Function
    End If
    
    ' ========== 界面更新 ==========
    ' 更改进化按钮文本状态
    Sheets("Main").Shapes("EvolveBtn").TextFrame.Characters.Text = Translate("Evolved")
    
    ' ========== 能力触发处理 ==========
    ' 触发进场时进化效果
    Call GetAction("Evolve", MatchID, True)
    
    '奥义
    Arr = Down("Hand", MatchID)
    For i = 1 To UBound(Arr)
        Effectrr = SplitEffect(CStr(Arr(i, 5)))
        For j = 0 To UBound(Effectrr)
            If Left(Effectrr(j), Len("Skybound&")) = "Skybound&" Then
                k = Val(Mid(Effectrr(j), 1 + Len("Skybound&"), 10))
                k = k + 1
                Effectrr(j) = "Skybound&" & k
                Exit For
            End If
        Next j
        Arr(i, 5) = Join(Effectrr, "||")
        If Arr(i, 5) <> "" Then Arr(i, 5) = Arr(i, 5) & "||"
        Call UpByMatchID(CStr(Arr(i, 1)), "Effect", CStr(Arr(i, 5)))
    Next i
    
    ' 其他卡牌响应进化事件
    Call GetAction("Evolve", MatchID, "0")
    If EvolveFlag Then Call GetAction("Keyword&Evolve", MatchID)
    
    ' ========== 清理和刷新 ==========
    ' 清空能力队列
    Call EraseQueue
End Function
