Attribute VB_Name = "·4·3·辅助·UpByMatchID"
' ==================================================
' 函数功能：根据匹配ID更新数据记录
' 参数：idValue - 匹配ID值
'        HeaderName - 要更新的字段名称
'        NewValue - 新的字段值
' 返回：Boolean - 操作是否成功
' 说明：处理卡牌区域移动、效果更新等复杂逻辑
' ==================================================
Function UpByMatchID(idValue As String, HeaderName As String, NewValue As Variant) As Boolean
    ' ========== 变量声明 ==========
    Dim i As Integer, j As Integer, k As Integer, m As Integer   ' 循环计数器
    Dim Arr As Variant          ' 临时数组
    Dim Brr As Variant          ' 临时数组
    Dim AllData As Variant      ' 全部数据数组
    Dim EffectSplit() As String ' 效果分割数组
    Dim Effectrr() As String    ' 效果数组
    
    UpByMatchID = True   ' 默认操作成功
    
    ' ========== 数据获取 ==========
    ' 获取全部Data区域数据
    AllData = Down("Data")
    
    ' ========== 数据遍历处理 ==========
    ' 遍历所有数据记录查找匹配的ID
    For i = 1 To UBound(AllData)
        If AllData(i, 1) = idValue Then
            ' ========== 不同字段类型处理 ==========
            If HeaderName = "ID" Then
                ' 更新ID字段
                AllData(i, 2) = NewValue
                
            ElseIf HeaderName = "Turn" Then
                ' 更新回合字段
                AllData(i, 3) = NewValue
                
            ElseIf HeaderName = "Area" Then
                ' ========== 区域移动处理 ==========
                ' 复制当前记录到最后一行
                AllData(UBound(AllData, 1), 1) = AllData(i, 1)
                AllData(UBound(AllData, 1), 2) = AllData(i, 2)
                AllData(UBound(AllData, 1), 3) = Down("Turn")
                
                ' ========== 区域容量检查 ==========
                ' 判断目标区域是否有足够空间
                If NewValue = "Hand" And AllData(i, 4) <> "Hand" Then
                    ' 手牌区域检查（最多9张）
                    Arr = Down("Hand", idValue)
                    If UBound(Arr, 1) >= 9 Then
                        NewValue = "HandDestroy"   ' 手牌已满，标记为销毁
                        UpByMatchID = False        ' 操作失败
                        Call Up("Shadows", Down("Shadows", idValue) + 1, idValue) ' 墓场+1
                    End If
                    
                ElseIf NewValue = "Field" And AllData(i, 4) <> "Field" Then
                    ' 场地区域检查（最多5个）
                    Arr = Down("Field", idValue)
                    If UBound(Arr, 1) >= 5 Then
                        NewValue = "FieldDestroy"   ' 场地已满，标记为销毁
                        UpByMatchID = False         ' 操作失败
                    End If
                    
                ElseIf NewValue = "Crest" Or NewValue = "Faith" Then
                    ' 纹章/信仰区域检查（最多5个）
                    Arr = Down("Crest", idValue)
                    If UBound(Arr, 1) >= 5 Then
                        NewValue = "CrestDestroy"   ' 纹章已满，标记为销毁
                        UpByMatchID = False         ' 操作失败
                    End If
                    For j = 1 To UBound(Arr)
                        If Arr(j, 2) = AllData(i, 2) Then
                            NewValue = "CrestDestroy"   ' 纹章已满，标记为销毁
                            UpByMatchID = False         ' 操作失败
                            Exit For
                        End If
                    Next j
                End If
                
                ' ========== 新区域设置 ==========
                AllData(UBound(AllData, 1), 4) = NewValue   ' 设置新区域
                AllData(i, 1) = Empty                       ' 清空原记录ID
                    
                If UpByMatchID Then
                    ' ========== 效果清理 ==========
                    ' 分割效果字符串进行处理
                    EffectSplit = SplitEffect(AllData(i, 5))
                    
                    ' 遍历所有效果，清理特定效果
                    For j = 0 To UBound(EffectSplit)
                        If EffectSplit(j) = "When" Then
                            EffectSplit(j) = "\"
                        ElseIf EffectSplit(j) = "ActivateInHand-When" Then
                            EffectSplit(j) = "\"
                        ElseIf EffectSplit(j) = "ActivateInDeck-When" Then
                            EffectSplit(j) = "\"
                        ElseIf EffectSplit(j) = "Ambush" Then
                            EffectSplit(j) = "\"
                        ElseIf EffectSplit(j) = "Bane" Then
                            EffectSplit(j) = "\"
                        ElseIf EffectSplit(j) = "Ward" Then
                            EffectSplit(j) = "\"
                        ElseIf EffectSplit(j) = "Intimidate" Then
                            EffectSplit(j) = "\"
                        ElseIf EffectSplit(j) = "Aura" Then
                            EffectSplit(j) = "\"
                        ElseIf EffectSplit(j) = "Barrier" Then
                            EffectSplit(j) = "\"
                        ElseIf EffectSplit(j) = "CantSelectOther" Then
                            EffectSplit(j) = "\"
                        ElseIf EffectSplit(j) = "IncreaseModes" Then
                            EffectSplit(j) = "\"
                        ElseIf Left(EffectSplit(j), Len("EngageCost")) = "EngageCost" Then
                            EffectSplit(j) = "\"
                        ElseIf Left(EffectSplit(j), Len("Cost")) = "Cost" Then
                            EffectSplit(j) = "\"
                        Else
                            Arr = Split(EffectSplit(j), "|")
                            If UBound(Arr) >= 1 Then
                                If Right(Arr(0), Len("Until")) = "Until" And Left(Arr(1), Len("Cost")) = "Cost" Then EffectSplit(j) = "\"
                            End If
                        End If
                    Next j
                    
                    ' ========== 效果字符串重组 ==========
                    AllData(i, 5) = Join(EffectSplit, "||")                    ' 合并效果
                    If AllData(i, 5) <> "" Then AllData(i, 5) = AllData(i, 5) & "||"   ' 添加分隔符
                    AllData(i, 5) = Replace(AllData(i, 5), "\||", "")          ' 清理标记的效果
                    
                    ' ========== 触发效果处理 ==========
                    ' 获取卡牌的所有效果
                    Effectrr = GetEffect(idValue)
                    
                    ' 当进入场地、纹章、信仰区域时
                    If NewValue = "Field" Or NewValue = "Crest" Or NewValue = "Faith" Then
                        For j = 1 To UBound(Effectrr)
                            If Effectrr(j) = "" Then Exit For
                            If Left(Effectrr(j), Len("When")) = "When" Then
                                AllData(i, 5) = AllData(i, 5) & "When||"   ' 添加触发效果
                                Exit For
                            End If
                        Next j
                        
                    ' 当进入手牌区域时
                    ElseIf NewValue = "Hand" Then
                        For j = 1 To UBound(Effectrr)
                            If Effectrr(j) = "" Then Exit For
                            If Left(Effectrr(j), Len("ActivateInHand-When")) = "ActivateInHand-When" Then
                                If InStr(AllData(i, 5), "ActivateInHand-When||") = 0 Then AllData(i, 5) = AllData(i, 5) & "ActivateInHand-When||" ' 添加手牌触发效果
                            End If
                            If InStr(Effectrr(j), "Skybound") > 0 Then
                                If InStr(AllData(i, 5), "Skybound&") = 0 Then AllData(i, 5) = AllData(i, 5) & "Skybound&0||" ' 添加奥义
                            End If
                        Next j
                        
                    ' 当进入牌组区域时
                    ElseIf NewValue = "Deck" Then
                        For j = 1 To UBound(Effectrr)
                            If Effectrr(j) = "" Then Exit For
                            If Left(Effectrr(j), Len("ActivateInDeck-When")) = "ActivateInDeck-When" Then
                                If InStr(AllData(i, 5), "ActivateInDeck-When||") = 0 Then AllData(i, 5) = AllData(i, 5) & "ActivateInDeck-When||" ' 添加牌组触发效果
                                Exit For
                            End If
                        Next j
                    End If
                    
                    ' ========== 常驻词条处理 ==========
                    ' 当进入场地区域时，添加常驻效果
                    If NewValue = "Field" Then
                        For j = 1 To UBound(Effectrr)
                            If Effectrr(j) = "" Then Exit For
                            If Effectrr(j) = "Ambush" _
                            Or Effectrr(j) = "Bane" _
                            Or Effectrr(j) = "Ward" _
                            Or Effectrr(j) = "Intimidate" _
                            Or Effectrr(j) = "Aura" _
                            Or Effectrr(j) = "Barrier" _
                            Or Effectrr(j) = "CantSelectOther" _
                            Then
                                AllData(i, 5) = AllData(i, 5) & Effectrr(j) & "||"   ' 添加常驻效果
                            End If
                        Next j
                    End If
                    
                    ' 当进入场地区域时，添加常驻效果
                    If NewValue = "Field" Or NewValue = "Crest" Then
                        For j = 1 To UBound(Effectrr)
                            If Effectrr(j) = "" Then Exit For
                            If Effectrr(j) = "IncreaseModes" _
                            Or Left(Effectrr(j), Len("EngageCost")) = "EngageCost" _
                            Then
                                AllData(i, 5) = AllData(i, 5) & Effectrr(j) & "||"   ' 添加常驻效果
                            End If
                        Next j
                    End If
                    
                    ' ========== 倒数处理 ==========
                    ' 当进入场地或纹章区域时，处理倒数效果
                    If NewValue = "Field" Or NewValue = "Crest" Then
                        j = Val(DownByID(CStr(AllData(i, 2)), "Countdown"))   ' 获取倒数数值
                        If j > 0 Then AllData(i, 5) = AllData(i, 5) & "Countdown&" & j & "||"   ' 添加倒数效果
                    End If
                    
                     ' ========== 土之印处理 ==========
                    ' 当进入场地区域时，处理土之印特性
                    If NewValue = "Field" Then
                        If DownByID(CStr(AllData(i, 2)), "Trait") = Translate("EarthSigil") Then
                            ' 查找场上已有的土之印
                            Arr = Down("Field", idValue)
                            For j = 1 To UBound(Arr, 1)
                                If InStr(Arr(j, 5), "EarthSigil&") > 0 Then Exit For
                            Next j
                            
                            ' 如果找到已有的土之印，则消失它
                            If j <= UBound(Arr, 1) Then
                                Brr = SplitEffect(Arr(j, 5))
                                For k = 0 To UBound(Brr)
                                    If Left(Brr(k), Len("EarthSigil&")) = "EarthSigil&" Then
                                        m = Val(Mid(Brr(k), 1 + Len("EarthSigil&"), 10)) + 1   ' 土之印计数+1
                                        Exit For
                                    End If
                                Next k
                                QueueAbility = QueueAbility & "\" & "||Banish|Goal&" & CStr(Arr(j, 1))   ' 添加到消失队列
                                QueueMatchID = QueueMatchID & "\" & CStr(AllData(i, 1))                  ' 记录匹配ID
                            Else
                                m = 1   ' 第一个土之印
                            End If
                            AllData(i, 5) = AllData(i, 5) & "EarthSigil&" & m & "||"   ' 添加土之印效果
                        End If
                    End If
                End If
                
                ' ========== 新记录效果复制 ==========
                AllData(UBound(AllData, 1), 5) = AllData(i, 5)   ' 复制效果到新记录
                AllData = NonEmptyArray(AllData, True)           ' 清理空记录
                
            ElseIf HeaderName = "Effect" Then
                ' ========== 效果字段更新 ==========
                AllData(i, 5) = NewValue   ' 直接更新效果字段
            End If
            Exit For   ' 找到匹配记录后退出循环
        End If
    Next i
    
    ' ========== 数据保存 ==========
    Call Up("Data", AllData)   ' 更新Data区域
End Function
