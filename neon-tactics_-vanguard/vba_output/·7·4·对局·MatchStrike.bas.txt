Attribute VB_Name = "·7·4·对局·MatchStrike"
' ==================================================
' 过程功能：处理从者攻击/护符交战的入口函数
' 说明：根据用户选择的目标执行攻击或交战操作
' 包含回合检查、攻击权限判断、目标选择等逻辑
' ==================================================
Function MatchStrike()
    ' ========== 变量声明 ==========
    Dim i%, j%, k%
    Dim Str As String
    Dim Arr As Variant
    Dim MatchID As String, ID As String, CardType As String, Turn As String, Effect As String
    Dim IsStrike As Integer
    Dim Effectrr() As String
    Dim EffectSplit() As String
    Dim options() As String
    
    Dim CantStrike As String
    
    ' ========== 战场信息获取 ==========
    ' 下载战场信息
    Arr = Down("Field", Left(Down("FirstMove"), 1) & 1000)
    ' 获取MatchID，如果为空则刷新并退出
    If Selection.Column - 2 > UBound(Arr, 1) Or Arr(1, 1) = "" Then
        Call UIRefresh
        GoTo CleanUp
    End If
    MatchID = Arr(Selection.Column - 2, 1)
    
    ' ========== 卡牌信息获取 ==========
    ' 获取ID和卡牌类型
    ID = DownByMatchID(MatchID, "ID")
    CardType = DownByID(ID, "Type")
    
    ' ========== 从者攻击处理 ==========
    If CardType = "Follower" Then
        ' 获取回合和效果信息
        Turn = DownByMatchID(MatchID, "Turn")
        Effect = DownByMatchID(MatchID, "Effect")
        Effectrr = GetEffect(MatchID)
        
        ' ========== 攻击权限判断 ==========
        ' 判断自己是否可攻击
        IsStrike = 0
        ' 不是登场回合则可以攻击主战者
        If Turn <> Down("Turn") Then IsStrike = 2
        
        ' 遍历效果找突进、疾驰、无法攻击
        For i = 1 To UBound(Effectrr)
            If Effectrr(i) = "Rush" Then
                If IsStrike = 0 Then IsStrike = 1
            ElseIf Effectrr(i) = "Storm" Then
                IsStrike = 2
            ElseIf Left(Effectrr(i), Len("CantStrike|")) = "CantStrike|" Then
                CantStrike = Mid(Effectrr(i), 1 + Len("CantStrike|"), 100)
                Call GetGoal("Enemy-Field-" & CantStrike, MatchID)
                CantStrike = PreviousGoal
            End If
        Next i
        
        ' ========== 攻击次数检查 ==========
        If IsStrike > 0 Then
            ' 遍历效果找风怒（可攻击次数）
            k = 1
            For i = 1 To UBound(Effectrr)
                If Left(Effectrr(i), Len("CanAttackTimes&")) = "CanAttackTimes&" Then
                    j = Val(Mid(Effectrr(i), 1 + Len("CanAttackTimes&"), 10))
                    If j > k Then k = j
                End If
            Next i
            j = k
            If j = 0 Then j = 1
            
            ' 遍历状态找已攻击次数
            i = InStr(Effect, "Striked&")
            If i > 0 Then i = Val(Mid(Effect, i + Len("Striked&"), 10))
            If j <= i Then IsStrike = 0
        End If
        
        ' 不能攻击则直接退出
        If IsStrike = 0 Then GoTo CleanUp
        
        ' ========== 敌方战场分析 ==========
        ' 下载敌方战场信息
        If Left(MatchID, 1) = "F" Then Str = "S" Else Str = "F"
        Arr = Down("Field", Str & 1000)
        
        ' ========== 敌方状态分析 ==========
        ' 遍历查找护符、守护、潜行、威慑等状态
        Flag = False
        For i = 1 To UBound(Arr, 1)
            If Arr(i, 1) = 0 Then
                Arr(i, 5) = 0
            ElseIf DownByID(CStr(Arr(i, 2)), "Type") = "Amulet" Then
                Arr(i, 5) = 0
            Else
                EffectSplit = SplitEffect(Arr(i, 5))
                Arr(i, 5) = 2
                For j = 0 To UBound(EffectSplit)
                    If EffectSplit(j) = "Ward" Or InStr(EffectSplit(j), "{Ward}") Then
                        If Arr(i, 5) > 1 Then Arr(i, 5) = 3
                    ElseIf EffectSplit(j) = "Remove|Ward" Then
                        If Arr(i, 5) = 3 Then Arr(i, 5) = 2
                    ElseIf EffectSplit(j) = "Ambush" Or InStr(EffectSplit(j), "{Ambush}") Or _
                            EffectSplit(j) = "Intimidate" Or InStr(EffectSplit(j), "{Intimidate}") Then
                        Arr(i, 5) = 1
                    ElseIf EffectSplit(j) = "Remove|Ambush" Or _
                            EffectSplit(j) = "Remove|Intimidate" Then
                        If Arr(i, 5) = 1 Then Arr(i, 5) = 2
                    End If
                Next j
                If Arr(i, 5) = 3 Then Flag = True
            End If
        Next i
        
        ' ========== 目标选择界面构建 ==========
        Str = Translate("Strike")
        For i = 1 To UBound(Arr, 1)
            If Flag And Arr(i, 5) = 3 And InStr(CantStrike, Arr(i, 1)) = 0 Then
                Str = Str & "|" & DownByID(CStr(Arr(i, 2)), "Name")
            ElseIf Not Flag And Arr(i, 5) = 2 And InStr(CantStrike, Arr(i, 1)) = 0 Then
                Str = Str & "|" & DownByID(CStr(Arr(i, 2)), "Name")
            Else
                Arr(i, 1) = Empty
            End If
        Next i
        Arr = NonEmptyArray(Arr)
        
        ' 加入主战者选项
        If IsStrike = 2 And Not Flag And InStr(CantStrike, "1000") = 0 Then Str = Str & "|" & Translate("Leader")
        
        ' ========== 用户选择处理 ==========
        ' 弹出选项供用户选择
        i = 0
        options = Split(Str, "|")
        If UBound(options) > 0 Then i = Val(ShowChoice(options, 150, 6))
        If i = 0 Then GoTo CleanUp
        
        ' ========== 攻击执行 ==========
        StrikeMatchID = MatchID
        ' 如果攻击了主战者
        If options(i) = Translate("Leader") Then
            If Left(MatchID, 1) = "F" Then Str = "S" & 1000 Else Str = "F" & 1000
            'ClashMatchID = Empty
        Else
            Str = Arr(i, 1)
            'ClashMatchID = Str
        End If
        Call StrikeFunction(MatchID, Str)
    Else
        ' ========== 护符交战处理 ==========
        Arr = EngageFunction(MatchID)
        If Not Arr Then GoTo CleanUp
    End If

    ' ========== 界面刷新 ==========
    Call UIRefresh
    Call UpStepsArea
    
CleanUp:
End Function

' ==================================================
' 函数功能：执行具体的攻击交战逻辑
' 参数：MatchID - 攻击方MatchID, InputClash - 被攻击方MatchID
' 说明：处理攻击时的各种效果触发、伤害计算、状态更新等
' 包括屏障、毁灭、虹吸等特殊效果的实现
' ==================================================
Function StrikeFunction(MatchID As String, InputClash As String)
    ' ========== 变量声明 ==========
    Dim i%, j%, k%
    Dim Damage%
    Dim StrikeAttack%
    Dim StrikeDefense%
    Dim StrikeID As String
    Dim StrikeEffect As String
    Dim StrikeEffectrr() As String
    Dim StrikeEffectSplit() As String
    Dim StrikeArea As String
    Dim ClashAttack%
    Dim ClashDefense%
    Dim ClashID As String
    Dim ClashEffect As String
    Dim ClashEffectrr() As String
    Dim ClashEffectSplit() As String
    Dim ClashArea As String
    Dim KeywordStrikeFlag As Boolean
    Dim KeywordClashFlagA As Boolean
    Dim KeywordClashFlagB As Boolean
    
    ' ========== 初始化设置 ==========
    ' 清空能力队列
    QueueAbility = Empty
    QueueMatchID = Empty
    If Mid(InputClash, 2, 10) <> "1000" Then ClashMatchID = InputClash Else ClashMatchID = Empty
    
    ' ========== 数据下载 ==========
    ' 下载攻击方和被攻击方的效果信息
    StrikeEffect = DownByMatchID(MatchID, "Effect")
    StrikeEffectSplit = SplitEffect(StrikeEffect)
    StrikeEffectrr = GetEffect(MatchID)
    StrikeID = DownByMatchID(MatchID, "ID")
    
    ClashEffect = DownByMatchID(InputClash, "Effect")
    ClashEffectSplit = SplitEffect(ClashEffect)
    ClashEffectrr = GetEffect(InputClash)
    ClashID = DownByMatchID(InputClash, "ID")
    
    ' ========== 攻击次数记录 ==========
    ' 更新已攻击次数
    For i = 0 To UBound(StrikeEffectSplit)
        If Left(StrikeEffectSplit(i), Len("Striked&")) = "Striked&" Then Exit For
    Next i
    If i <= UBound(StrikeEffectSplit) Then
        j = Val(Mid(StrikeEffectSplit(i), 1 + Len("Striked&"), 10))
        StrikeEffectSplit(i) = "Striked&" & j + 1
        StrikeEffect = Join(StrikeEffectSplit, "||") & "||"
    Else
        StrikeEffect = StrikeEffect & "Striked&" & 1 & "||"
        StrikeEffectSplit = SplitEffect(StrikeEffect)
    End If
    Call UpByMatchID(MatchID, "Effect", StrikeEffect)
    
    ' ========== 关键字效果触发 ==========
    ' 攻击时关键字效果处理
    For i = 1 To UBound(StrikeEffectrr)
        If Left(StrikeEffectrr(i), Len("Strike")) = "Strike" And InStr(StrikeEffect, "Remove|Strike") = 0 Then
            Call GetTimePoint(StrikeEffectrr(i), MatchID)
            KeywordStrikeFlag = True
        ElseIf Mid(InputClash, 2, 10) <> "1000" And Left(StrikeEffectrr(i), Len("Clash")) = "Clash" And InStr(StrikeEffect, "Remove|Clash") = 0 Then
            Call GetTimePoint(StrikeEffectrr(i), MatchID)
            KeywordClashFlagA = True
        End If
    Next i
    
    ' 交战时关键字效果处理
    If Mid(InputClash, 2, 10) <> "1000" Then
        For i = 1 To UBound(ClashEffectrr)
            If Left(ClashEffectrr(i), Len("Clash")) = "Clash" And InStr(ClashEffect, "Remove|Clash") = 0 Then
                Call GetTimePoint(ClashEffectrr(i), InputClash)
                KeywordClashFlagB = True
            End If
        Next i
    End If
    
    ' ========== 状态移除 ==========
    ' 攻击后失去潜行状态
    For i = 0 To UBound(StrikeEffectSplit)
        If StrikeEffectSplit(i) = "Ambush" Or InStr(StrikeEffectSplit(i), "{Ambush}") > 0 Then
            StrikeEffectSplit(i) = "\"
            StrikeEffect = Join(StrikeEffectSplit, "||") & "||"
            StrikeEffect = Replace(StrikeEffect, "\||", "")
            Call UpByMatchID(MatchID, "Effect", StrikeEffect)
        End If
    Next i
    
    ' ========== 能力触发处理 ==========
    ' 其他卡牌响应攻击和交战事件
    Call GetAction("Strike", MatchID, True)
    If Mid(InputClash, 2, 10) <> "1000" Then
        Call GetAction("Clash", MatchID, True)
        Call GetAction("Clash", InputClash, True)
    End If
    Call GetAction("Strike", MatchID, DownByID(StrikeID, "Name"))
    If Mid(InputClash, 2, 10) <> "1000" Then
        Call GetAction("Clash", MatchID, DownByID(StrikeID, "Name"))
        Call GetAction("Clash", InputClash, DownByID(ClashID, "Name"))
    End If
    If KeywordStrikeFlag Then Call GetAction("Keyword&Strike", MatchID)
    If KeywordClashFlagA Then Call GetAction("Keyword&Clash", MatchID)
    If KeywordClashFlagB Then Call GetAction("Keyword&Clash", InputClash)
    
    ' 清空能力队列
    Call EraseQueue
    
    ' ========== 数据重新下载 ==========
    ' 重新下载更新后的效果信息
    StrikeEffect = DownByMatchID(MatchID, "Effect")
    StrikeEffectSplit = SplitEffect(StrikeEffect)
    StrikeEffectrr = GetEffect(MatchID)
    
    ClashEffect = DownByMatchID(InputClash, "Effect")
    ClashEffectSplit = SplitEffect(ClashEffect)
    ClashEffectrr = GetEffect(InputClash)
    
    ' ========== 状态验证 ==========
    ' 如果攻击方或被攻击方不在场，则直接退出
    StrikeArea = DownByMatchID(MatchID, "Area")
    ClashArea = DownByMatchID(InputClash, "Area")
    If StrikeArea <> "Field" Or (ClashArea <> "Field" And ClashArea <> "Leader") Then
        Call UIRefresh
        Exit Function
    End If
    
    ' ========== 主战者攻击处理 ==========
    ' 如果对主战者进行攻击
    If Mid(InputClash, 2, 10) = "1000" Then
        Flag = False
        For i = 0 To UBound(ClashEffectSplit)
            If InStr(ClashEffectSplit(i), "{Barrier}") > 0 Then
                ' 屏障效果处理
                Flag = True
                ClashEffectSplit(i) = "\"
                ClashEffect = Join(ClashEffectSplit, "||") & "||"
                ClashEffect = Replace(ClashEffect, "\||", "")
                Call UpByMatchID(InputClash, "Effect", ClashEffect)
            ElseIf ClashEffectSplit(i) = "Remove|Barrier" Then
                Flag = False
                Exit For
            End If
        Next i
        If Flag Then
            'Call GetAction("LostBarrier", InputClash, 1)
        Else
            ' 计算伤害并更新主战者防御力
            StrikeAttack = GetModel("Attack", StrikeID, StrikeEffect)
            Damage = StrikeAttack
            For i = 1 To UBound(ClashEffectrr)
                If Left(ClashEffectrr(i), Len("CantTakeMoreDamage&")) = "CantTakeMoreDamage&" Then
                    j = Val(Mid(ClashEffectrr(i), 1 + Len("CantTakeMoreDamage&"), 10))
                    If Damage > j Then Damage = j
                ElseIf Left(ClashEffectrr(i), Len("TakeDamage&")) = "TakeDamage&" Then
                    j = Val(Mid(ClashEffectrr(i), 1 + Len("TakeDamage&"), 10))
                    Damage = Damage + j
                End If
            Next i
            If Damage < 0 Then Damage = 0
            
            Call Up("Defense", Down("Defense", InputClash) - Damage, InputClash)
            Call Up("BloodPool", Down("BloodPool", InputClash) + Damage, InputClash) '血池+
        End If
        
    ' ========== 从者攻击处理 ==========
    Else
        StrikeAttack = GetModel("Attack", StrikeID, StrikeEffect)
        ClashAttack = GetModel("Attack", ClashID, ClashEffect)
        
        ' ========== 攻击方伤害处理 ==========
        Flag = False
        For i = 0 To UBound(StrikeEffectSplit)
            If StrikeEffectSplit(i) = "Barrier" Or InStr(StrikeEffectSplit(i), "{Barrier}") > 0 Then
                ' 屏障效果处理
                Flag = True
                StrikeEffectSplit(i) = "\"
                StrikeEffect = Join(StrikeEffectSplit, "||") & "||"
                StrikeEffect = Replace(StrikeEffect, "\||", "")
                Call UpByMatchID(MatchID, "Effect", StrikeEffect)
            ElseIf StrikeEffectSplit(i) = "Remove|Barrier" Then
                Flag = False
                Exit For
            End If
        Next i
        
        If Flag Then
            ' 屏障被打破时的能力触发
            'Call GetAction("LostBarrier", MatchID, True)
            'Call GetAction("LostBarrier", MatchID, DownByID(ClashID, "Name"))
        ElseIf InStr(StrikeEffect, "SuperEvolved||") > 0 Then
            ' 超进化处理（注释标明不严谨实现）
        ElseIf ClashAttack > 0 Then
            '计算伤害
            Damage = ClashAttack
            For i = 1 To UBound(ClashEffectrr)
                If Left(ClashEffectrr(i), Len("DoDamage&")) = "DoDamage&" Then
                    j = Val(Mid(ClashEffectrr(i), 1 + Len("DoDamage&"), 10))
                    Damage = Damage + j
                End If
            Next i
            For i = 1 To UBound(StrikeEffectrr)
                If Left(StrikeEffectrr(i), Len("TakeDamage&")) = "TakeDamage&" Then
                    j = Val(Mid(StrikeEffectrr(i), 1 + Len("TakeDamage&"), 10))
                    Damage = Damage + j
                End If
            Next i
            For i = 1 To UBound(StrikeEffectrr)
                If Left(StrikeEffectrr(i), Len("CantTakeMoreDamage&")) = "CantTakeMoreDamage&" Then
                    j = Val(Mid(StrikeEffectrr(i), 1 + Len("CantTakeMoreDamage&"), 10))
                    If Damage > j Then Damage = j
                End If
            Next i
            If Damage < 0 Then Damage = 0
            
            ' 受伤记录处理
            For i = 0 To UBound(StrikeEffectSplit)
                If Left(StrikeEffectSplit(i), Len("Damaged&")) = "Damaged&" Then Exit For
            Next i
            If i <= UBound(StrikeEffectSplit) Then
                j = Val(Mid(StrikeEffectSplit(i), 1 + Len("Damaged&"), 10))
                StrikeEffectSplit(i) = "Damaged&" & j + Damage
                StrikeEffect = Join(StrikeEffectSplit, "||") & "||"
            Else
                StrikeEffect = StrikeEffect & "Damaged&" & Damage & "||"
                StrikeEffectSplit = SplitEffect(StrikeEffect)
            End If
            Call UpByMatchID(MatchID, "Effect", StrikeEffect)
            
            ' 检查是否被破坏
            StrikeDefense = GetModel("Defense", StrikeID, StrikeEffect)
            If StrikeDefense <= 0 Then Call DestroyFunction(MatchID)
        End If
        
        ' 毁灭效果检查
        Flag = False
        For i = 0 To UBound(ClashEffectSplit)
            If ClashEffectSplit(i) = "Bane" Or InStr(ClashEffectSplit(i), "{Bane}") > 0 Then Flag = True
            If ClashEffectSplit(i) = "Remove|Bane" Then
                Flag = False
                Exit For
            End If
        Next i
        If Flag Then Call GetAbility("||Destroy|ThisCard", MatchID)
        
        ' 受伤未破坏时的能力触发
        If DownByMatchID(MatchID, "Area") = "Field" Then
            Call GetAction("Damaged", MatchID, True)
            Call GetAction("Damaged", MatchID, ClashAttack)
        End If
        
        ' ========== 被攻击方伤害处理 ==========
        Flag = False
        For i = 0 To UBound(ClashEffectSplit)
            If ClashEffectSplit(i) = "Barrier" Or InStr(ClashEffectSplit(i), "{Barrier}") > 0 Then
                ' 屏障效果处理
                Flag = True
                ClashEffectSplit(i) = "\"
                ClashEffect = Join(ClashEffectSplit, "||") & "||"
                ClashEffect = Replace(ClashEffect, "\||", "")
                Call UpByMatchID(InputClash, "Effect", ClashEffect)
            ElseIf ClashEffectSplit(i) = "Remove|Barrier" Then
                Flag = False
                Exit For
            End If
        Next i
        
        If Flag Then
            ' 屏障被打破时的能力触发
            'Call GetAction("LostBarrier", InputClash, True)
            'Call GetAction("LostBarrier", InputClash, DownByID(StrikeID, "Name"))
        ElseIf StrikeAttack > 0 Then
            '计算伤害
            Damage = StrikeAttack
            For i = 1 To UBound(StrikeEffectrr)
                If Left(StrikeEffectrr(i), Len("DoDamage&")) = "DoDamage&" Then
                    j = Val(Mid(StrikeEffectrr(i), 1 + Len("DoDamage&"), 10))
                    Damage = Damage + j
                End If
            Next i
            For i = 1 To UBound(ClashEffectrr)
                If Left(ClashEffectrr(i), Len("TakeDamage&")) = "TakeDamage&" Then
                    j = Val(Mid(ClashEffectrr(i), 1 + Len("TakeDamage&"), 10))
                    Damage = Damage + j
                End If
            Next i
            For i = 1 To UBound(ClashEffectrr)
                If Left(ClashEffectrr(i), Len("CantTakeMoreDamage&")) = "CantTakeMoreDamage&" Then
                    j = Val(Mid(ClashEffectrr(i), 1 + Len("CantTakeMoreDamage&"), 10))
                    If Damage > j Then Damage = j
                End If
            Next i
            If Damage < 0 Then Damage = 0
            
            ' 受伤记录处理
            For i = 0 To UBound(ClashEffectSplit)
                If Left(ClashEffectSplit(i), Len("Damaged&")) = "Damaged&" Then Exit For
            Next i
            If i <= UBound(ClashEffectSplit) Then
                j = Val(Mid(ClashEffectSplit(i), 1 + Len("Damaged&"), 10))
                ClashEffectSplit(i) = "Damaged&" & j + Damage
                ClashEffect = Join(ClashEffectSplit, "||") & "||"
            Else
                ClashEffect = ClashEffect & "Damaged&" & Damage & "||"
                ClashEffectSplit = SplitEffect(ClashEffect)
            End If
            Call UpByMatchID(InputClash, "Effect", ClashEffect)
            
            ' 检查是否被破坏
            ClashDefense = GetModel("Defense", ClashID, ClashEffect)
            If ClashDefense <= 0 Then Call DestroyFunction(InputClash)
        End If
        
        ' 毁灭效果检查
        Flag = False
        For i = 0 To UBound(StrikeEffectSplit)
            If StrikeEffectSplit(i) = "Bane" Or InStr(StrikeEffectSplit(i), "{Bane}") > 0 Then Flag = True
            If StrikeEffectSplit(i) = "Remove|Bane" Then
                Flag = False
                Exit For
            End If
        Next i
        If Flag Then Call GetAbility("||Destroy|ThisCard", InputClash)
        
        ' 受伤未破坏时的能力触发
        If DownByMatchID(InputClash, "Area") = "Field" Then
            Call GetAction("Damaged", MatchID, True)
            Call GetAction("Damaged", MatchID, StrikeAttack)
        ElseIf InStr(StrikeEffect, "SuperEvolved||") > 0 Then
            Call GetAbility("||Damage|Enemy-Leader|1", MatchID)
        End If
    End If
    
    ' ========== 虹吸效果处理 ==========
    For i = 1 To UBound(StrikeEffectrr)
        If StrikeEffectrr(i) = "Drain" And InStr(StrikeEffect, "Remove|Strike") = 0 Then
            Call GetAbility("||Restore|Allied-Leader|" & StrikeAttack, MatchID)
            Exit For
        End If
    Next i
    
    ' ========== 清理和刷新 ==========
    ' 清空能力队列
    Call EraseQueue
End Function
