Attribute VB_Name = "·8·3·对局·DestroyFunction"
' ==================================================
' 函数功能：处理卡牌破坏逻辑
' 参数：MatchID - 要被破坏的卡牌MatchID
' 说明：处理卡牌被破坏时的各种效果和状态变更
' 包括不可破坏效果检查、区域变更、谢幕曲触发等
' ==================================================
Function DestroyFunction(MatchID As String)
    ' ========== 变量声明 ==========
    Dim i%, j%
    Dim Str As String
    Dim Area As String
    Dim Effect As String
    Dim Effectrr() As String
    Dim EffectSplit() As String
    Dim LastWordsFlag As Boolean
    
    ' ========== 效果信息获取 ==========
    ' 下载卡牌的当前效果信息
    Effect = DownByMatchID(MatchID, "Effect")
    Effectrr = GetEffect(MatchID)
    EffectSplit = SplitEffect(Effect)
    
    ' ========== 区域状态检查 ==========
    ' 如果本来就不在战场或纹章区，则空发（不执行破坏）
    Area = DownByMatchID(MatchID, "Area")
    If Area <> "Field" And Area <> "Crest" Then Exit Function
    
    ' ========== 不可破坏效果检查 ==========
    ' 检查不会被破坏的效果和特殊处理
    For i = 1 To UBound(Effectrr)
        If Effectrr(i) = "CantBeDestroyed" Then
            ' 检查防御力或倒计时，如果大于0则不能被破坏
            j = GetModel("Defense", DownByMatchID(MatchID, "ID"), Effect)
            If j > 0 Then Exit Function
            j = GetModel("Countdown", DownByMatchID(MatchID, "ID"), Effect)
            If j > 0 Then Exit Function
        ElseIf Effectrr(i) = "LeaveBanish" Then
            ' 放逐效果处理：将卡牌区域改为放逐并触发相关能力
            Call UpByMatchID(MatchID, "Area", "Banish")
            Call GetAction("Banish", MatchID, True)
            Call GetAction("Banish", MatchID, DownByID(DownByMatchID(MatchID, "ID"), "Name"))
            Call GetAction("LeaveField", MatchID, True)
            Call GetAction("LeaveField", MatchID)
            Exit Function
        End If
    Next i
    
    ' ========== 特殊状态检查 ==========
    ' 检查土之印和超进化状态
    Str = Down("Turn")
    For i = 0 To UBound(EffectSplit)
        If Left(EffectSplit(i), Len("EarthSigil&")) = "EarthSigil&" Then
            ' 土之印状态：不能被破坏
            Exit Function
        ElseIf EffectSplit(i) = "SuperEvolved" And Left(Str, 1) = Left(MatchID, 1) Then
            ' 超进化状态：检查防御力，如果大于0则不能被破坏
            j = GetModel("Defense", DownByMatchID(MatchID, "ID"), Effect)
            If j > 0 Then Exit Function
        End If
    Next i
    
    ' ========== 区域变更处理 ==========
    ' 将卡牌区域更改为破坏区
    Call UpByMatchID(MatchID, "Area", "Destroy")
    'Call UpByMatchID(MatchID, "Effect", "")
    
    ' ========== 墓场计数更新 ==========
    ' 如果是从战场破坏，则墓场计数+1
    If Area = "Field" Then Call Up("Shadows", Down("Shadows", MatchID) + 1, MatchID)
    
    ' ========== 谢幕曲效果处理 ==========
    ' 触发卡牌的谢幕曲效果
    LastWordsFlag = False
    For i = 1 To UBound(Effectrr)
        If Left(Effectrr(i), Len("LastWords")) = "LastWords" And InStr(Effect, "Remove|LastWords") = 0 Then
            Call GetTimePoint(Effectrr(i), MatchID)
            LastWordsFlag = True
        End If
    Next i
    
    ' ========== 离开战场能力触发 ==========
    ' 触发离开战场时的能力
    Call GetAction("LeaveField", MatchID, True)
    
    ' ========== 其他能力触发处理 ==========
    ' 其他卡牌响应破坏事件和离开战场事件
    Call GetAction("Destroy", MatchID, DownByID(DownByMatchID(MatchID, "ID"), "Name"))
    Call GetAction("LeaveField", MatchID)
    If LastWordsFlag Then Call GetAction("Keyword&LastWords", MatchID)
End Function
